# src/townlet/config/experiment_config.py
"""ExperimentConfig DTO for experiment.yaml (v2.1)."""

from pathlib import Path
from pydantic import BaseModel, Field
import yaml


class ExperimentMetadata(BaseModel):
    """Metadata section of experiment.yaml."""
    name: str = Field(..., min_length=1, description="Experiment name")
    description: str = Field(..., description="Human-readable description")
    author: str = Field(default="HAMLET Team", description="Author name")
    created: str = Field(..., description="Creation date (YYYY-MM-DD)")
    tags: list[str] = Field(default_factory=list, description="Search tags")


class ExperimentSection(BaseModel):
    """Root experiment section."""
    version: str = Field(..., pattern=r"^\d+\.\d+$", description="Config version")
    metadata: ExperimentMetadata
    curriculum_levels: list[str] = Field(
        ...,
        min_length=1,
        description="List of curriculum level directory names"
    )


class ExperimentConfig(BaseModel):
    """Root config for experiment.yaml (v2.1 hierarchical structure)."""

    experiment: ExperimentSection

    @classmethod
    def from_yaml(cls, path: Path) -> "ExperimentConfig":
        """Load and validate experiment.yaml."""
        with open(path) as f:
            data = yaml.safe_load(f)
        return cls(**data)
```

# src/townlet/config/stratum_config.py
"""StratumConfig DTO for stratum.yaml (v2.1)."""

from pathlib import Path
from pydantic import BaseModel, Field, field_validator
import yaml


class GridSubstrate(BaseModel):
    """Grid substrate configuration."""
    topology: str = Field(..., description="square or hexagonal")
    width: int = Field(..., ge=1, le=100, description="Grid width")
    height: int = Field(..., ge=1, le=100, description="Grid height")

    # Behavioral parameters (MANDATORY but NOT BREAKING)
    boundary: str = Field(..., description="clamp, wrap, bounce, sticky")
    distance_metric: str = Field(..., description="manhattan, euclidean, chebyshev")
    observation_encoding: str = Field(..., description="relative, scaled, absolute")

    @field_validator('topology')
    def validate_topology(cls, v):
        valid = {'square', 'hexagonal'}
        if v not in valid:
            raise ValueError(f"topology must be one of {valid}, got {v}")
        return v

    @field_validator('boundary')
    def validate_boundary(cls, v):
        valid = {'clamp', 'wrap', 'bounce', 'sticky'}
        if v not in valid:
            raise ValueError(f"boundary must be one of {valid}, got {v}")
        return v

    @field_validator('distance_metric')
    def validate_distance_metric(cls, v):
        valid = {'manhattan', 'euclidean', 'chebyshev'}
        if v not in valid:
            raise ValueError(f"distance_metric must be one of {valid}, got {v}")
        return v

    @field_validator('observation_encoding')
    def validate_observation_encoding(cls, v):
        valid = {'relative', 'scaled', 'absolute'}
        if v not in valid:
            raise ValueError(f"observation_encoding must be one of {valid}, got {v}")
        return v


class SubstrateSection(BaseModel):
    """Substrate configuration."""
    type: str = Field(..., description="grid, grid3d, gridnd, continuous, continuousnd, aspatial")
    grid: GridSubstrate | None = None

    @field_validator('type')
    def validate_type(cls, v):
        valid = {'grid', 'grid3d', 'gridnd', 'continuous', 'continuousnd', 'aspatial'}
        if v not in valid:
            raise ValueError(f"type must be one of {valid}, got {v}")
        return v


class StratumSection(BaseModel):
    """Root stratum section."""
    version: str = Field(..., pattern=r"^\d+\.\d+$")
    substrate: SubstrateSection
    vision_support: str = Field(..., description="both, global, partial, none")
    temporal_support: str = Field(..., description="enabled, disabled")

    @field_validator('vision_support')
    def validate_vision_support(cls, v):
        valid = {'both', 'global', 'partial', 'none'}
        if v not in valid:
            raise ValueError(f"vision_support must be one of {valid}, got {v}")
        return v

    @field_validator('temporal_support')
    def validate_temporal_support(cls, v):
        valid = {'enabled', 'disabled'}
        if v not in valid:
            raise ValueError(f"temporal_support must be one of {valid}, got {v}")
        return v


class StratumConfig(BaseModel):
    """Root config for stratum.yaml (v2.1)."""

    stratum: StratumSection

    @classmethod
    def from_yaml(cls, path: Path) -> "StratumConfig":
        """Load and validate stratum.yaml."""
        with open(path) as f:
            data = yaml.safe_load(f)
        return cls(**data)
```

## Task 11: Create RawConfigsV21 Loader

**Files:**
- Create: `src/townlet/universe/raw_configs_v21.py`
- Test: Manual test loading all configs

**Important:** All DTOs follow the section-root pattern:
- `ExperimentConfig.experiment`, `StratumConfig.stratum`, `EnvironmentConfig.environment`
- `ActionsConfig.actions`, `AgentConfig.agent`
- `CurriculumConfig.curriculum`, `BarsConfig.bars`, `AffordancesConfig.affordances`, `TrainingConfig.training`

This means field access always goes through the root section (e.g., `level.bars.bars.meters`, not `level.bars.meters`).

**Step 1: Create dataclass structure**

```python
# src/townlet/universe/raw_configs_v21.py
"""Raw config loader for v2.1 hierarchical structure."""

from dataclasses import dataclass
from pathlib import Path

from townlet.config.experiment_config import ExperimentConfig
from townlet.config.stratum_config import StratumConfig
from townlet.config.environment_config import EnvironmentConfig
from townlet.config.actions_config import ActionsConfig
from townlet.config.agent_config import AgentConfig
from townlet.config.curriculum_config import CurriculumConfig
from townlet.config.bars_config import BarsConfig
from townlet.config.affordances_config import AffordancesConfig
from townlet.config.training_config import TrainingConfig
from townlet.universe.errors import CompilationErrorCollector


@dataclass(frozen=True)
class CurriculumLevel:
    """All configs for a single curriculum level."""
    name: str
    curriculum: CurriculumConfig
    bars: BarsConfig
    affordances: AffordancesConfig
    training: TrainingConfig


@dataclass(frozen=True)
class RawConfigsV21:
    """Container of all config DTOs for v2.1 hierarchical structure."""

    # Shared experiment-level configs
    experiment: ExperimentConfig
    stratum: StratumConfig
    environment: EnvironmentConfig
    actions: ActionsConfig
    agent: AgentConfig

    # Curriculum levels
    levels: dict[str, CurriculumLevel]

    # Provenance
    experiment_dir: Path

    def __post_init__(self):
        """Validate vocabulary consistency across all levels."""
        if not self.levels:
            raise ValueError(f"No curriculum levels found in {self.experiment_dir}")

        # Vocabulary validation goes here (Task 11)

    @classmethod
    def from_experiment_dir(cls, experiment_dir: Path) -> "RawConfigsV21":
        """Load all configs from v2.1 hierarchical structure."""
        experiment_dir = Path(experiment_dir).resolve()
        errors = CompilationErrorCollector(stage="Stage 1: Load v2.1 Configs")

        # Load shared configs (per-file error handling for clarity)
        shared_files = [
            ("experiment.yaml", ExperimentConfig),
            ("stratum.yaml", StratumConfig),
            ("environment.yaml", EnvironmentConfig),
            ("actions.yaml", ActionsConfig),
            ("agent.yaml", AgentConfig),
        ]

        experiment = stratum = environment = actions = agent = None
        for filename, config_cls in shared_files:
            try:
                config_path = experiment_dir / filename
                config = config_cls.from_yaml(config_path)
                if filename == "experiment.yaml":
                    experiment = config
                elif filename == "stratum.yaml":
                    stratum = config
                elif filename == "environment.yaml":
                    environment = config
                elif filename == "actions.yaml":
                    actions = config
                elif filename == "agent.yaml":
                    agent = config
            except Exception as e:
                errors.add(
                    f"Failed to load {filename}: {e}",
                    code="LOAD_ERROR",
                    location=str(config_path)
                )

        errors.check_and_raise()

        # Load curriculum levels
        levels_dir = experiment_dir / "levels"
        if not levels_dir.exists():
            errors.add(
                f"Missing levels/ directory",
                code="MISSING_LEVELS_DIR",
                location=str(experiment_dir)
            )
            errors.check_and_raise()

        levels = {}
        for level_dir in sorted(levels_dir.iterdir()):
            if not level_dir.is_dir():
                continue

            level_name = level_dir.name
            try:
                curriculum = CurriculumConfig.from_yaml(level_dir / "curriculum.yaml")
                bars = BarsConfig.from_yaml(level_dir / "bars.yaml")
                affordances = AffordancesConfig.from_yaml(level_dir / "affordances.yaml")
                training = TrainingConfig.from_yaml(level_dir / "training.yaml")

                levels[level_name] = CurriculumLevel(
                    name=level_name,
                    curriculum=curriculum,
                    bars=bars,
                    affordances=affordances,
                    training=training
                )
            except Exception as e:
                errors.add(
                    f"Failed to load level {level_name}: {e}",
                    code="LEVEL_LOAD_ERROR",
                    location=level_name
                )

        errors.check_and_raise()

        return cls(
            experiment=experiment,
            stratum=stratum,
            environment=environment,
            actions=actions,
            agent=agent,
            levels=levels,
            experiment_dir=experiment_dir
        )
```

## Task 12: Add Vocabulary Validation

**Files:**

**Step 1: Implement vocabulary consistency check**

```python
def __post_init__(self):
    """Validate vocabulary consistency across all levels."""
    if not self.levels:
        raise ValueError(f"No curriculum levels found in {self.experiment_dir}")

    # Get canonical vocabulary from environment.yaml
    env_meter_names = {m.name for m in self.environment.environment.meters}
    env_affordance_names = {a.name for a in self.environment.environment.affordances}

    # Validate each level has same vocabulary (note: section-root pattern)
    for level_name, level in self.levels.items():
        level_meter_names = {m.name for m in level.bars.bars.meters}
        level_affordance_names = {a.name for a in level.affordances.affordances.affordances}

        if level_meter_names != env_meter_names:
            missing = env_meter_names - level_meter_names
            extra = level_meter_names - env_meter_names
            raise ValueError(
                f"Meter vocabulary mismatch in {level_name}/bars.yaml\n"
                f"  Expected: {sorted(env_meter_names)}\n"
                f"  Actual: {sorted(level_meter_names)}\n"
                f"  Missing: {sorted(missing) if missing else 'none'}\n"
                f"  Extra: {sorted(extra) if extra else 'none'}\n"
                f"\nAll levels must have same meters as environment.yaml"
            )

        if level_affordance_names != env_affordance_names:
            missing = env_affordance_names - level_affordance_names
            extra = level_affordance_names - env_affordance_names
            raise ValueError(
                f"Affordance vocabulary mismatch in {level_name}/affordances.yaml\n"
                f"  Expected: {sorted(env_affordance_names)}\n"
                f"  Actual: {sorted(level_affordance_names)}\n"
                f"  Missing: {sorted(missing) if missing else 'none'}\n"
                f"  Extra: {sorted(extra) if extra else 'none'}\n"
                f"\nAll levels must have same affordances as environment.yaml"
            )
```
