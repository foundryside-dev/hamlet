```python
def _phase_0_validate_yaml_syntax(self, experiment_dir: Path) -> None:
    """Phase 0 – validate all YAML files can be parsed before compilation begins."""
    errors = CompilationErrorCollector(stage="Phase 0: YAML Syntax Validation")

    # Shared config files (v2.1 structure)
    shared_files = [
        "experiment.yaml",
        "stratum.yaml",
        "environment.yaml",
        "actions.yaml",
        "agent.yaml",
    ]

    for file_name in shared_files:
        file_path = experiment_dir / file_name
        if not file_path.exists():
            errors.add(
                f"{file_name}: File not found",
                code="MISSING_FILE",
                location=str(file_path)
            )
            continue

        try:
            with file_path.open() as handle:
                yaml.safe_load(handle)
        except yaml.YAMLError as exc:
            error_msg = str(exc)
            if hasattr(exc, "problem_mark"):
                mark = exc.problem_mark
                problem = getattr(exc, "problem", None) or "syntax error"
                error_msg = f"line {mark.line + 1}, column {mark.column + 1}: {problem}"
            errors.add(error_msg, code="YAML_SYNTAX_ERROR", location=file_name)

    # Curriculum level files
    levels_dir = experiment_dir / "levels"
    if not levels_dir.exists():
        errors.add(
            "Missing levels/ directory",
            code="MISSING_LEVELS_DIR",
            location=str(experiment_dir)
        )
    else:
        for level_dir in levels_dir.iterdir():
            if not level_dir.is_dir():
                continue

            level_files = ["curriculum.yaml", "bars.yaml", "affordances.yaml", "training.yaml"]
            for file_name in level_files:
                file_path = level_dir / file_name
                if not file_path.exists():
                    errors.add(
                        f"{level_dir.name}/{file_name}: File not found",
                        code="MISSING_FILE",
                        location=str(file_path)
                    )
                    continue

                try:
                    with file_path.open() as handle:
                        yaml.safe_load(handle)
                except yaml.YAMLError as exc:
                    error_msg = str(exc)
                    if hasattr(exc, "problem_mark"):
                        mark = exc.problem_mark
                        problem = getattr(exc, "problem", None) or "syntax error"
                        error_msg = f"line {mark.line + 1}, column {mark.column + 1}: {problem}"
                    errors.add(
                        error_msg,
                        code="YAML_SYNTAX_ERROR",
                        location=f"{level_dir.name}/{file_name}"
                    )

    if errors.errors:
        errors.add_hint("Check YAML indentation (use spaces, not tabs)")
        errors.add_hint("Ensure lists use proper '- item' syntax")
        errors.check_and_raise()
```

```python
observation_fields = []

# 1. Position (from substrate)
if raw_configs.stratum.stratum.substrate.type in ["grid", "grid3d"]:
    observation_fields.append(ObservationField(
        name="obs_position",
        dims=2,  # Or 3 for grid3d
        curriculum_active=True  # Always active
    ))

# 2. Spatial views (Support/Active pattern)
vision_support = raw_configs.stratum.stratum.vision_support
active_vision = level.curriculum.active_vision

if vision_support in ["both", "global"]:
    # grid_encoding field exists
    grid_dims = (
        raw_configs.stratum.stratum.substrate.grid.width *
        raw_configs.stratum.stratum.substrate.grid.height
    )
    observation_fields.append(ObservationField(
        name="obs_grid_encoding",
        dims=grid_dims,
        curriculum_active=(active_vision == "global")  # Active only if selected
    ))

if vision_support in ["both", "partial"]:
    # local_window field exists
    vision_range = level.curriculum.vision_range
    window_size = int(vision_range * 2 + 1)
    observation_fields.append(ObservationField(
        name="obs_local_window",
        dims=window_size ** 2,
        curriculum_active=(active_vision == "partial")  # Active only if selected
    ))

# 3. Meters (always active)
for meter in level.bars.bars.meters:
    observation_fields.append(ObservationField(
        name=f"obs_{meter.name}",
        dims=1,
        curriculum_active=True
    ))

# 4. Affordances (always active)
affordance_count = len(level.affordances.affordances.affordances)
observation_fields.append(ObservationField(
    name="obs_affordance_at_position",
    dims=affordance_count + 1,  # +1 for "none"
    curriculum_active=True
))

# 5. Temporal (Support/Active pattern)
temporal_support = raw_configs.stratum.stratum.temporal_support
active_temporal = level.curriculum.active_temporal

if temporal_support == "enabled":
    observation_fields.extend([
        ObservationField(
            name="obs_time_sin",
            dims=1,
            curriculum_active=active_temporal
        ),
        ObservationField(
            name="obs_time_cos",
            dims=1,
            curriculum_active=active_temporal
        ),
    ])

# 6. VFS variables (always active)
for variable in raw_configs.environment.environment.variables:
    observation_fields.append(ObservationField(
        name=f"obs_{variable.name}",
        dims=variable.dims,
        curriculum_active=True
    ))

observation_spec = ObservationSpec(fields=observation_fields)
```
