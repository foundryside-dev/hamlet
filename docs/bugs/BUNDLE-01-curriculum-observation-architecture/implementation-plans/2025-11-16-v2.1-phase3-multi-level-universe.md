## Task 2: Add get_level() Method

**Files:**
- Modify: `src/townlet/universe/compiled.py`

**Step 1: Add get_level() accessor method**

Add after the LevelMetadata class definition:

```python
def get_level(self, level_name: str) -> "CompiledUniverse.LevelMetadata":
    """
    Get metadata for a specific curriculum level.

    Args:
        level_name: Name of curriculum level (e.g., "L1_full_observability")

    Returns:
        LevelMetadata for the requested level

    Raises:
        ValueError: If multi-level not enabled or level not found
    """
    if self.all_levels is None:
        raise ValueError(
            "This CompiledUniverse was not compiled with multi-level support. "
            "Use compile(experiment_dir, primary_level=...) instead of compile(config_dir)."
        )

    if level_name not in self.all_levels:
        raise ValueError(
            f"Level '{level_name}' not found. "
            f"Available levels: {list(self.all_levels.keys())}"
        )

    return self.all_levels[level_name]
```

**Step 2: Add available_levels property**

Add after get_level() method:

```python
@property
def available_levels(self) -> list[str]:
    """Get list of available curriculum levels (sorted alphabetically)."""
    if self.all_levels is None:
        return []
    return sorted(self.all_levels.keys())
```


With:

```python
def create_environment(
    self,
    num_agents: int = 1,
    device: str = "cpu",
    level_name: str | None = None
) -> VectorizedHamletEnv:
    """
    Create vectorized environment from compiled universe.

    Args:
        num_agents: Number of parallel agents
        device: Torch device ("cpu" or "cuda")
        level_name: For multi-level compilation, which level to use
                   (uses primary level if None)

    Returns:
        VectorizedHamletEnv configured for the selected level
    """
    # If level_name specified, use that level's metadata
    if level_name is not None:
        if self.all_levels is None:
            raise ValueError(
                f"level_name='{level_name}' specified but this CompiledUniverse "
                "was not compiled with multi-level support"
            )
        level_metadata = self.get_level(level_name)
        observation_spec = level_metadata.observation_spec
        action_metadata = level_metadata.action_metadata
    else:
        # Use primary level metadata
        observation_spec = self.observation_spec
        action_metadata = self.action_metadata

    # Create environment using selected level's metadata
    return VectorizedHamletEnv(
        observation_spec=observation_spec,
        action_metadata=action_metadata,
        meter_metadata=self.meter_metadata,
        affordance_metadata=self.affordance_metadata,
        num_agents=num_agents,
        device=device
    )
```
