"""Tests for multi-level compilation features (Config v2.1)."""

from pathlib import Path
import pytest
from townlet.universe.compiler import UniverseCompiler


def test_compile_all_levels():
    """Test that all 5 curriculum levels are compiled in single pass."""
    compiler = UniverseCompiler()
    compiled = compiler.compile(
        Path("configs/default_curriculum"),
        use_cache=False
    )

    assert compiled.all_levels is not None
    assert len(compiled.all_levels) == 5

    expected_levels = [
        "L0_0_minimal",
        "L0_5_dual_resource",
        "L1_full_observability",
        "L2_partial_observability",
        "L3_temporal_mechanics"
    ]

    for level_name in expected_levels:
        assert level_name in compiled.all_levels, f"Missing level: {level_name}"


def test_primary_level_selection():
    """Test that primary_level parameter selects correct level for primary metadata."""
    compiler = UniverseCompiler()

    # Compile with L1 as primary
    compiled_l1 = compiler.compile(
        Path("configs/default_curriculum"),
        primary_level="L1_full_observability",
        use_cache=False
    )

    assert compiled_l1.metadata.level_name == "L1_full_observability"

    # Compile with L2 as primary
    compiled_l2 = compiler.compile(
        Path("configs/default_curriculum"),
        primary_level="L2_partial_observability",
        use_cache=False
    )

    assert compiled_l2.metadata.level_name == "L2_partial_observability"


def test_default_primary_level():
    """Test that primary_level defaults to first level alphabetically."""
    compiler = UniverseCompiler()
    compiled = compiler.compile(
        Path("configs/default_curriculum"),
        use_cache=False
    )

    # Should default to L0_0_minimal (first alphabetically)
    assert compiled.metadata.level_name == "L0_0_minimal"


def test_switch_levels_at_runtime():
    """Test that we can get different level metadata from same CompiledUniverse."""
    compiler = UniverseCompiler()
    compiled = compiler.compile(
        Path("configs/default_curriculum"),
        use_cache=False
    )

    l1 = compiled.get_level("L1_full_observability")
    l2 = compiled.get_level("L2_partial_observability")

    # Both should have observation_spec
    assert l1.observation_spec is not None
    assert l2.observation_spec is not None

    # Both should have action_metadata
    assert l1.action_metadata is not None
    assert l2.action_metadata is not None


def test_vocabulary_consistency_across_levels():
    """Test that all levels have same meters/affordances vocabulary."""
    compiler = UniverseCompiler()
    compiled = compiler.compile(
        Path("configs/default_curriculum"),
        use_cache=False
    )

    # Get meter names from each level
    level_meters = {}
    for level_name, level_meta in compiled.all_levels.items():
        meter_names = [f.name for f in level_meta.observation_spec.fields if f.name.startswith("obs_meter_")]
        level_meters[level_name] = set(meter_names)

    # All levels should have same meter vocabulary
    first_level = list(level_meters.keys())[0]
    expected_meters = level_meters[first_level]

    for level_name, meters in level_meters.items():
        assert meters == expected_meters, f"{level_name} has different meter vocabulary"


def test_invalid_primary_level_raises_error():
    """Test that invalid primary_level raises clear error."""
    compiler = UniverseCompiler()

    with pytest.raises(ValueError, match="Primary level 'INVALID' not found"):
        compiler.compile(
            Path("configs/default_curriculum"),
            primary_level="INVALID",
            use_cache=False
        )


def test_experiment_dir_set_correctly():
    """Test that experiment_dir is set in CompiledUniverse."""
    compiler = UniverseCompiler()
    compiled = compiler.compile(
        Path("configs/default_curriculum"),
        use_cache=False
    )

    assert compiled.experiment_dir is not None
    assert compiled.experiment_dir.name == "default_curriculum"
```

"""Tests for v2.1 config DTOs."""

from pathlib import Path
import pytest
from townlet.config.experiment import ExperimentConfig
from townlet.config.stratum import StratumConfig
from townlet.config.environment import EnvironmentConfig
from townlet.config.curriculum import CurriculumConfig
from townlet.config.raw_configs_v21 import RawConfigsV21


def test_experiment_config_loads():
    """Test ExperimentConfig loads from experiment.yaml."""
    config = ExperimentConfig.from_yaml(
        Path("configs/default_curriculum/experiment.yaml")
    )

    assert config.experiment.name == "default_curriculum"
    assert config.experiment.version == "1.0"


def test_stratum_config_loads():
    """Test StratumConfig loads from stratum.yaml."""
    config = StratumConfig.from_yaml(
        Path("configs/default_curriculum/stratum.yaml")
    )

    assert config.stratum.substrate_type in ["grid", "grid3d", "continuous", "aspatial"]


def test_environment_config_loads():
    """Test EnvironmentConfig loads from environment.yaml."""
    config = EnvironmentConfig.from_yaml(
        Path("configs/default_curriculum/environment.yaml")
    )

    assert config.environment.meters is not None
    assert config.environment.affordances is not None


def test_curriculum_config_loads():
    """Test CurriculumConfig loads from level's curriculum.yaml."""
    config = CurriculumConfig.from_yaml(
        Path("configs/default_curriculum/levels/L1_full_observability/curriculum.yaml")
    )

    assert config.curriculum.level_metadata.name == "L1_full_observability"


def test_raw_configs_v21_loads_all():
    """Test RawConfigsV21 loads entire experiment."""
    raw = RawConfigsV21.from_experiment_dir(
        Path("configs/default_curriculum")
    )

    assert raw.experiment.experiment.name == "default_curriculum"
    assert len(raw.levels) == 5

    expected_levels = ["L0_0_minimal", "L0_5_dual_resource", "L1_full_observability", "L2_partial_observability", "L3_temporal_mechanics"]
    for level_name in expected_levels:
        assert level_name in raw.levels


def test_vocabulary_validation():
    """Test that RawConfigsV21 enforces vocabulary consistency."""
    # This should pass (default_curriculum has consistent vocabulary)
    raw = RawConfigsV21.from_experiment_dir(
        Path("configs/default_curriculum")
    )

    # All levels should have same meter names
    env_meter_names = [m.name for m in raw.environment.environment.meters]

    for level_name, level in raw.levels.items():
        level_meter_names = [m.name for m in level.bars.bars.meters]
        assert set(level_meter_names) == set(env_meter_names), \
            f"{level_name} has different meter vocabulary"
```
