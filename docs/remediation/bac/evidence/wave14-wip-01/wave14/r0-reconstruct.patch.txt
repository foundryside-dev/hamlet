+def training_config(*, strategy="static", mode="stage_band", rate=0.01, omitted=False):
+    data = load_training_v2_config(LEVEL).model_dump()
+    data["population"]["size"] = 1
+    data["curriculum"]["strategy"] = strategy
+    data["curriculum"]["adversarial"].update(
+        adaptation_rate=rate,
+        min_difficulty=0.0,
+        max_difficulty=1.0,
+        min_steps_at_stage=1,
+        survival_advance_threshold=0.7,
+        survival_retreat_threshold=0.3,
+        entropy_gate=0.5,
+    )
+    data["bac_options"]["curriculum"] = None if omitted else {"commit_mode": "completion", "difficulty_mode": mode}
+    return TrainingV2Config.model_validate(data)
+
+
+def resolve(training):
+    return op.resolve_bac_options(training.bac_options, training=training)
+
+
+def effective(training, roster=ROSTER):
+    brain = ef.BrainConfig.model_validate(yaml.safe_load((PACK / "brain.yaml").read_text()))
+    drive = yaml.safe_load((LEVEL / "drive.yaml").read_text())["drive"]["intrinsic"]
+    return ef.resolve_effective_run(
+        brain,
+        training,
+        primary_level="L0_0_minimal",
+        origins={"training": "test-training"},
+        drive_intrinsic=ef.IntrinsicStrategyConfig.model_validate(drive),
+        declared_meter_names=roster,
+    )
+
+
+def manager(training, *, roster=ROSTER, via_effective=False):
+    if via_effective:
+        policy = effective(training, roster)
+        options, static = policy.options, policy.static_policy
+    else:
+        options = resolve(training).options
+        static = op.resolve_static_baseline_policy(declared_meter_names=roster)
+    result = build_curriculum(training, curriculum_options=options.curriculum, static_policy=static, device=torch.device("cpu"))
+    result.initialize_population(1)
+    return result
+
+
+def event(seq, *, survival=10, horizon=10, entropy=0.0):
+    return LaneOutcome(
+        outcome=EpisodeCompletion(
+            agent_idx=0,
+            reason="cap",
+            survival_time=survival,
+            final_observation=torch.tensor([0.2]),
+            final_meters=torch.tensor([0.8]),
+        ),
+        batch_episode=seq,
+        planned_horizon=horizon,
+        reward_total=1.0,
+        reward_extrinsic=1.0,
+        reward_intrinsic=0.0,
+        reward_shaping=0.0,
+        policy_entropy=entropy,
+        world_id="declared-test-world",
+        world_episode_seq=seq,
+    )
+
+
+def difficulty(curriculum):
+    return curriculum.get_batch_decisions(None, ["test-agent"])[0].difficulty_level

+    payload = copy.deepcopy(current.to_dict())
+    if damage == "missing_policy":
+        payload.pop("static_policy")
+    elif damage == "roster":

+    elif damage == "difficulty":
+        payload["static_policy"]["difficulty_level"] = 0.9
+    elif damage == "reward":
+        payload["static_policy"]["reward_mode"] = "forged"
+    elif damage == "hash":
+        payload["execution_hash"] = "0" * 64
+    elif damage == "role":
+        payload["field_roles"]["static_policy.active_meters"]["status"] = "active"
+    else:
+        payload["training"]["bac_options"]["curriculum"]["commit_mode"] = "legacy_step"
+    with pytest.raises((ValueError, KeyError)):
+        ef.EffectiveRunConfig.from_dict(payload)
