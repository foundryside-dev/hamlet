def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=R, text=True).strip()

m = json.loads((W / 'wave32/r1-candidate.json').read_text())
assert git('rev-parse', 'HEAD') == m['base_commit']
assert git('write-tree') == m['candidate_tree']
assert not git('diff', '--name-only')
for group in ('source_files', 'configs', 'artifacts'):
    for p, h in m[group].items():
        assert digest(R / p) == h, p
receipt = json.loads((W / 'wave32/r1-checkpoint-receipt.json').read_text())
assert receipt['candidate_tree'] == m['candidate_tree']
assert receipt['results']['coherent']['counts']['passed'] == 951
assert receipt['results']['coherent']['counts']['skipped'] == 3

}

    assert digest(W / 'wave32-preparation' / name) == h
execution = {p: git('rev-parse', m['candidate_tree'] + ':' + p) for p in ('src', 'tests', 'configs', 'scripts')}
out = D / 'evidence/wave32-wip-01'
out.mkdir(exist_ok=True)
artifacts = {}

def copy(source, dest):
    dest.parent.mkdir(parents=True, exist_ok=True)
    if dest.exists():
        assert digest(source) == digest(dest), str(dest)
    else:
        dest.write_bytes(source.read_bytes())
    assert digest(source) == digest(dest)
    artifacts[dest.relative_to(R).as_posix()] = {'sha256': digest(dest), 'original': str(source)}

        if folder.is_dir():
            for p in sorted(folder.rglob('*')):
                if p.is_file() and p.suffix in ('.json', '.log'):
                    copy(p, out / 'qualification' / folder.name / p.relative_to(folder))
    for p in sorted(Q.glob(wave + '-r*.json')):
        copy(p, out / 'qualification' / p.name)
for folder in ('wave29', 'wave29-preparation', 'wave30-preparation', 'wave31', 'wave31-preparation', 'wave32', 'wave32-preparation'):

    copy(W / name, out / ('execution-' + name))
