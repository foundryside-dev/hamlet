    assert source.is_file() and not dest.exists(), str(dest)

    assert sha(source) == sha(dest)

for p in sorted(Q.glob('wave34-*.json')):

for folder in ('wave34', 'wave34-preparation', 'wave35-preparation'):

Not actual parent integration or package/full BAC release.', 'publication': 'Existing authorized remote branch only. No new project branch/worktree/repository/source copy/PR/merge/deletion. Private backup bundle excluded.', 'next': 'Capture immutable WIP source, run required earlier full repository execution, causally repair failures before dependent boundary green; then continue finite T18e canonical reward integration.'}

assert not mp.exists()
mp.write_text(json.dumps(manifest, indent=2) + '\n')
p = D / 'STATUS.md'
p.write_text(p.read_text() + '''
