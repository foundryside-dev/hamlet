def put(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data if isinstance(data, bytes) else data.encode())
    return path


def js(path, value):
    return put(path, json.dumps(value, sort_keys=True).encode())


def ref(root, path, root_name):
    p = root / path
    return {"root": root_name, "path": path, "sha256": d.digest(p), "bytes": p.stat().st_size}

    def plan(self, draft=False):
        return d.plan(self.r, self.w, self.q, self.contract_path, self.chain, self.receipt_path, draft=draft)

    def fails(self, fragment):
        with self.assertRaisesRegex(ValueError, fragment):
            self.plan()

    def test_additive_reuse_alias_and_immutable_python(self):
        result = self.plan()
        self.assertEqual(result["status"], "READY FOR ROOT PUBLICATION CHECKS")
        self.assertEqual(result["objects"][d.digest(self.wip13_object)]["action"], "reference-existing")
        self.assertIn("qualification/wave35-r74-failed/failure.log", {r["logical_path"] for r in result["inputs"]})
        self.assertIn("qualification/wave35-r73-unrun.json", {r["logical_path"] for r in result["inputs"]})
        self.assertTrue(any(x["path"].endswith("verify.py.txt") for x in result["objects"].values()))
        self.assertGreater(len(result["inputs"]) - len(result["objects"]), 0)
        self.assertGreaterEqual(sum(x["sha256"] == d.digest(self.wip13_object) for x in result["inputs"]), 1)

    def test_assemble_synthetic_only_and_draft_rejection(self):
        draft = self.plan(draft=True)
        p = js(self.w / "draft-plan.json", draft)
        args = (self.r, self.w, self.q, self.contract_path, self.chain, p, self.w / "assembly.json")
        with self.assertRaisesRegex(ValueError, "draft or invalid"):
            d.assemble(*args)
        p = js(p, self.plan())
        report = d.assemble(*args)
        self.assertEqual(report["status"], "INCOMPLETE PUBLICATION PROPOSAL; ROOT REVIEW REQUIRED")
        self.assertFalse(report["navigation_approved"])
        with self.assertRaises(ValueError):
            d.assemble(*args)
