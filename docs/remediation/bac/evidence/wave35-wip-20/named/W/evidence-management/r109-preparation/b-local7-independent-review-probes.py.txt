def ids(source):
    classes = {n.name:n for n in ast.parse(source).body if isinstance(n, ast.ClassDef)}
    def names(name):
        node = classes[name]
        found = {n.name for n in node.body if isinstance(n, ast.FunctionDef) and n.name.startswith('test_')}
        for base in node.bases:
            if isinstance(base, ast.Name) and base.id in classes:
                found |= names(base.id)
        return found
    return {name + '.' + method for name in classes for method in names(name)}

count = 0

def probe(label, mutation=None, reason=None):
    global count
    owner = m.CudaVerifierTests()
    try:
        f = owner.fixture()
        if mutation:
            mutation(f)
            for trial in f.guards:
                owner.save_guard(f, trial)
                identity = f.job_dir / f'fork-identity-cuda_core_inherited_pid-{trial}.json'
                assert f.guards[trial][1]['identity_sha256'] == hashlib.sha256(identity.read_bytes()).hexdigest()
        try:
            receipt = f.invoke()
        except ValueError as exc:
            assert mutation is not None, str(exc)
            assert str(exc) == reason, (label, str(exc), reason)
            assert not list(f.job_dir.glob('replay-spawn-custody-*.json'))
            outcome = {'refused': str(exc), 'receipt_absent': True}
        else:
            assert mutation is None, ('accepted malformed fixture', label)
            assert len(receipt['fork_guards']) == 3
            outcome = {'accepted': True}
        count += 1
        print(json.dumps({'probe': label, **outcome}))
    finally:
        owner.doCleanups()
