def digest(data):
    return hashlib.sha256(data).hexdigest()


def ref(path):
    data = path.read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': digest(data)}


def write(path, value):

    path.write_text(json.dumps(value))

def paths_of(value, prefix=()):
    if type(value) is dict:
        yield prefix, value
        for key, child in value.items():
            yield from paths_of(child, prefix + (key,))
    elif type(value) is list:
        for index, child in enumerate(value):
            yield from paths_of(child, prefix + (index,))


def at(value, path):
    for key in path:
        value = value[key]
    return value


@pytest.mark.parametrize('row', [0, 1, 2])
@pytest.mark.parametrize('mutation', ['extra', 'missing', 'wrong_type'])
def test_every_record_object_is_closed_and_typed(tmp_path, validator, monkeypatch, row, mutation):
    f = Fixture(tmp_path, row)
    for record_name in ('child', 'worker', 'observed', 'result'):
        original = copy.deepcopy(getattr(f, record_name))
        # JSON-object maps are closed too; remove arbitrary one of their fixed/source-derived keys.
        for path, obj in list(paths_of(original)):
            if not obj:
                continue
            mutant = copy.deepcopy(original)
            target = at(mutant, path)
            if mutation == 'extra':
                target['unauthorized'] = 1
            elif mutation == 'missing':

            else:

                target[key] = None
            desired = copy.deepcopy(mutant)
            setattr(f, record_name, mutant)
            f.sync()
            # Preserve the deliberate mutation while authenticating its exact bytes.
            write(f.paths[record_name], desired)
            if record_name == 'child':
                f.observed['subject_ref'] = ref(f.paths['child'])
                write(f.paths['observed'], f.observed)
            with pytest.raises(ValueError):
                f.invoke(validator, monkeypatch)
            assert not (f.job / 'cold-import-custody.json').exists(), (record_name, path, mutation)
