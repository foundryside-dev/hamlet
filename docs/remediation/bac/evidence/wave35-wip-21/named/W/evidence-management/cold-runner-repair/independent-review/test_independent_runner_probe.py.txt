def load(name):

    spec.loader.exec_module(module)
    return module

@pytest.mark.parametrize('target', ['output', 'parent', 'root', 'manifest', 'spec'])
def test_symlink_paths_refuse(packet, target):
    p = packet
    real = p['path'].parent / ('real-' + target)
    if target in ('output', 'parent'):
        real.mkdir()
        (real / 'preserved').write_bytes(b'untouched')
        link = p['output'] if target == 'output' else p['output'].parent
        link.parent.mkdir(parents=True, exist_ok=True)
        link.symlink_to(real, target_is_directory=True)
    elif target == 'root':
        real.symlink_to(p['root'], target_is_directory=True)
        p['spec']['root'] = str(real)
    elif target == 'manifest':
        real.symlink_to(p['manifest'])
        p['spec']['manifest'] = str(real)
    else:
        real.write_text('{}')
        p['path'].symlink_to(real)
    completed = runner_tests.launch(p)
    assert completed.returncode != 0
    assert not p['guards'].exists()
    assert not (p['output'] / 'cold-role.log').exists()
    if target in ('output', 'parent'):
        assert (real / 'preserved').read_bytes() == b'untouched'

@pytest.mark.parametrize('guard', ['HEAD', 'index', 'diff'])
def test_failed_pre_guard_never_launches(packet, guard):
    binary = Path(packet['env']['PATH'].split(os.pathsep)[0]) / 'git'
    code = binary.read_text()
    if guard == 'HEAD':
        code = code.replace("'a' * 40", "'c' * 40")
    elif guard == 'index':
        code = code.replace("'b' * 40", "'c' * 40")
    else:
        code += "\nif args == ('diff', '--quiet'):\n    sys.exit(1)\n"
    binary.write_text(code)
    completed = runner_tests.launch(packet)
    assert completed.returncode != 0
    assert not (packet['output'] / 'cold-role.log').exists()


def test_simultaneous_attempt_collision_has_one_winner(packet):
    packet['path'].write_text(json.dumps(packet['spec']))
    def start():
        return subprocess.run([sys.executable, '-B', str(runner_tests.RUNNER), str(packet['path'])],
                              env=packet['env'], capture_output=True, text=True)
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
        results = list(pool.map(lambda _: start(), range(2)))
    assert sorted(r.returncode == 0 for r in results) == [False, True]
    assert (packet['output'] / 'cold-role.log').read_text() == 'child reached\n'
    assert len(packet['guards'].read_text().splitlines()) == 6
