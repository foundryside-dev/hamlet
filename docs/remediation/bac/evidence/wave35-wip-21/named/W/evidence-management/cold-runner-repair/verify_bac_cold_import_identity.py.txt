def require(condition, message):
    if not condition:
        raise ValueError(message)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def same(actual, expected):
    if type(actual) is not type(expected):
        return False
    if type(expected) is dict:
        return actual.keys() == expected.keys() and all(same(actual[k], v) for k, v in expected.items())
    if type(expected) is list:
        return len(actual) == len(expected) and all(same(a, b) for a, b in zip(actual, expected))
    return actual == expected


def equal(actual, expected, label):
    require(same(actual, expected), label + ' mismatch')


def exact(obj, fields, label):
    require(type(obj) is dict and set(obj) == set(fields.split()), label + ' schema mismatch')


def integer(value, minimum, label):
    require(type(value) is int and value >= minimum, label + ' must be exact integer')


def string(value, label):
    require(type(value) is str and bool(value), label + ' must be nonempty string')


def sha(value):
    require(type(value) is str and re.fullmatch('[0-9a-f]{64}', value) is not None, 'invalid SHA256')


def duration(value):
    require(type(value) is float and math.isfinite(value) and 0 < value <= 100, 'invalid duration')


def pairs(items):
    obj = {}
    for key, value in items:
        require(key not in obj, 'duplicate JSON key: ' + key)
        obj[key] = value
    return obj


def finite(value):
    if type(value) is float:
        require(math.isfinite(value), 'nonfinite JSON number')
    elif type(value) is dict:
        for child in value.values():
            finite(child)
    elif type(value) is list:
        for child in value:
            finite(child)


def read_json(path):
    def constant(value):
        raise ValueError('nonfinite JSON constant: ' + value)
    value = json.loads(Path(path).read_bytes(), object_pairs_hook=pairs, parse_constant=constant)
    finite(value)
    require(type(value) is dict, 'JSON object required')
    return value


def canonical(value):
    string(value, 'path')
    path = Path(value)
    require(path.is_absolute() and str(path) == value and path.resolve() == path, 'noncanonical path')
    for ancestor in (path, *path.parents):
        require(not ancestor.is_symlink(), 'symlink path refused')
    return path


def reference(path):
    path = canonical(str(path))
    require(path.is_file(), 'reference is not a regular file')
    data = path.read_bytes()
    return dict(path=str(path), bytes=len(data), sha256=digest(data))


def check_ref(value, path=None):
    exact(value, 'path bytes sha256', 'reference')
    integer(value['bytes'], 0, 'reference size')
    sha(value['sha256'])
    actual = canonical(value['path'])
    if path is not None:
        equal(str(actual), str(path), 'reference path')
    equal(value, reference(actual), 'reference bytes')
    return actual

def extract_probe(source):
    try:
        tree = ast.parse(source)
        writes = [n for n in ast.walk(tree) if isinstance(n, ast.Name) and n.id == '_COLD_IMPORT_PROBE'
                  and isinstance(n.ctx, (ast.Store, ast.Del))]
        candidates = [n for n in tree.body if isinstance(n, ast.Assign) and len(n.targets) == 1
                      and isinstance(n.targets[0], ast.Name) and n.targets[0].id == '_COLD_IMPORT_PROBE']
        require(len(writes) == len(candidates) == 1, 'probe assignment/rebinding')
        for binding in ast.walk(tree):
            if isinstance(binding, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.ExceptHandler,
                                    ast.MatchAs, ast.MatchStar)):
                require(binding.name != '_COLD_IMPORT_PROBE', 'probe named rebinding')
            elif isinstance(binding, ast.MatchMapping):
                require(binding.rest != '_COLD_IMPORT_PROBE', 'probe mapping rebinding')
            elif isinstance(binding, ast.arg):
                require(binding.arg != '_COLD_IMPORT_PROBE', 'probe argument rebinding')
            elif isinstance(binding, ast.alias):
                require((binding.asname or binding.name.split('.')[0]) != '_COLD_IMPORT_PROBE', 'probe import rebinding')
            elif isinstance(binding, ast.Subscript) and isinstance(binding.ctx, (ast.Store, ast.Del)):
                require(not isinstance(binding.slice, ast.Constant) or binding.slice.value != '_COLD_IMPORT_PROBE',
                        'probe namespace rebinding')
            elif isinstance(binding, ast.Attribute) and isinstance(binding.ctx, (ast.Store, ast.Del)):
                require(binding.attr != '_COLD_IMPORT_PROBE', 'probe attribute rebinding')
        node = candidates[0]
        require(isinstance(node.value, ast.Constant) and type(node.value.value) is str and node.value.value.isascii(), 'probe literal type')
        segment = ast.get_source_segment(source, node.value)
        tokens = [t for t in tokenize.generate_tokens(io.StringIO(segment).readline)
                  if t.type not in (tokenize.NL, tokenize.NEWLINE, tokenize.ENDMARKER)]
        require(len(tokens) == 1 and tokens[0].type == tokenize.STRING, 'probe adjacent/dynamic literals')
        return node.value.value
    except (SyntaxError, tokenize.TokenError) as error:
        raise ValueError('invalid probe source') from error


def fault_sites(probe):
    tree = ast.parse(probe)
    parents = {child: parent for parent in ast.walk(tree) for child in ast.iter_child_nodes(parent)}
    def function(node):
        while node in parents:
            node = parents[node]
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                return node.name
        return '<module>'
    boundaries = [n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name == '_cold_boundary']
    require(len(boundaries) == 1, 'unique cold boundary required')
    assertions = [n for n in ast.walk(boundaries[0]) if isinstance(n, ast.Assert) and
                  isinstance(n.msg, ast.Constant) and n.msg.value == 'cold import precondition violated']
    raises = [n for n in ast.walk(tree) if isinstance(n, ast.Raise) and isinstance(n.exc, ast.Call)
              and isinstance(n.exc.func, ast.Name) and n.exc.func.id == 'RuntimeError'
              and len(n.exc.args) == 1 and isinstance(n.exc.args[0], ast.Constant)
              and n.exc.args[0].value == 'cold import CUDA sentinel touched']
    require(len(assertions) == len(raises) == 1, 'unique fault site required')
    found = {}
    for mode, site in (('preflight_preloaded', assertions[0]), ('preflight_cuda_touch', raises[0])):
        branches = [n for n in ast.walk(tree) if isinstance(n, ast.If) and any(
            isinstance(v, ast.Constant) and v.value == mode for v in ast.walk(n.test))]
        calls = []
        for branch in branches:
            for stmt in branch.body:
                for n in ast.walk(stmt):
                    if mode == 'preflight_preloaded':
                        match = isinstance(n, ast.Call) and isinstance(n.func, ast.Name) and n.func.id == '_cold_boundary'
                    else:
                        match = isinstance(n, ast.Attribute) and ast.unparse(n) == 'torch.cuda.is_initialized'
                    if match:
                        calls.append(n)
        require(len(calls) == 1, 'unique designated fault caller required')
        found[mode] = dict(site_function=function(site), site_line=site.lineno,
                           caller_function=function(calls[0]), caller_line=calls[0].lineno)
    return found


def check_repository(root, manifest):
    # Same established guard protocol; no old spawn subject/profile is called.
    require(subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip() == manifest['base_commit'], 'HEAD mismatch')
    with (root / '.superpowers/bac-qualification-index.lock').open('a') as lock:
        fcntl.flock(lock.fileno(), fcntl.LOCK_EX)
        try:
            tree = subprocess.check_output(['git', 'write-tree'], cwd=root, text=True).strip()
            require(tree == manifest['candidate_tree'], 'index tree mismatch')
        finally:
            fcntl.flock(lock.fileno(), fcntl.LOCK_UN)
    require(subprocess.run(['git', 'diff', '--quiet'], cwd=root, check=False).returncode == 0, 'unstaged changes')
    registry = subprocess.check_output(['git', 'worktree', 'list', '--porcelain'], cwd=root, text=True)
    equal(registry, manifest['worktree_registry'], 'worktree registry')


def check_inventory(root, manifest):
    allowed = {}
    for category in ('source_files', 'configs', 'artifacts'):
        inventory = manifest[category]
        require(type(inventory) is dict, 'invalid inventory')
        for relative, expected in inventory.items():
            string(relative, 'inventory path')
            path = Path(relative)
            require(not path.is_absolute() and '..' not in path.parts and str(path) == relative, 'unsafe inventory path')
            target = canonical(str(root / path))
            require(target.is_relative_to(root), 'inventory escape')
            sha(expected)
            equal(reference(target)['sha256'], expected, 'inventory hash')
            if category == 'source_files' and relative.startswith('src/townlet/') and relative.endswith('.py'):
                module = relative[4:-3].replace('/', '.')
                if module.endswith('.__init__'):
                    module = module[:-9]
                require(module not in allowed, 'duplicate module origin')
                allowed[module] = dict(path=str(target), sha256=expected)
    return allowed

def check_fault(child, allowed, probe, stderr):
    preload = child['mode'] == 'preflight_preloaded'
    equal(child['phase'], 'before_preentry_assert' if preload else 'before_sentinel_raise', 'fault phase')
    equal(child['cuda_calls'], 0 if preload else 1, 'fault CUDA count')
    inventory_names(child['modules'], allowed)
    if preload:
        require({'townlet', CORE} <= set(child['modules']), 'preload witness missing custody')
        require(not {FLAT, PER}.intersection(child['modules']), 'preload imported producer')
    else:
        equal(child['modules'], [], 'CUDA fault must be cold')
    check_origins(child['source_modules'], child['modules'], allowed)
    check_threads(child['threads'], child['pid'])
    equal(child['environment'], ENV, 'fault environment')
    sites = fault_sites(probe)[child['mode']]
    for key, value in sites.items():
        equal(child[key], value, 'fault ' + key)
    text = stderr.read_text(encoding='utf-8')
    lines = text.splitlines()
    require(text.count('Traceback (most recent call last):') == 1, 'one complete traceback required')
    exception = ('AssertionError: cold import precondition violated' if preload else
                 'RuntimeError: cold import CUDA sentinel touched')
    require(bool(lines) and lines[-1] == exception, 'terminal fault exception mismatch')
    header = lines.index('Traceback (most recent call last):')
    require(header == 0, 'unexpected traceback preamble')
    frames = []
    for line in lines[1:-1]:
        match = re.fullmatch(r'  File "<string>", line ([1-9][0-9]*), in (.+)', line)
        if match:
            frames.append((int(match[1]), match[2]))
        else:
            require(line.startswith('    ') and bool(frames), 'unexpected traceback chain content')
    require(len(frames) >= 2, 'fault caller frame missing')
    equal(frames[-2:], [(sites['caller_line'], sites['caller_function']), (sites['site_line'], sites['site_function'])], 'fault terminal frames')


def check_worker(worker, job, root, runtime, manifest, manifest_ref, test_ref, plugin_ref):
    exact(worker, 'schema role node pid ppid start_ticks root cwd launch_python binary binary_sha256 orig_argv proc_argv command_sha256 '
                  'manifest_ref test_ref plugin_ref base_commit candidate_tree python_threads os_threads', 'worker')
    for key in ('pid', 'ppid', 'start_ticks', 'os_threads'):
        integer(worker[key], 1, 'worker ' + key)
    require(worker['pid'] != worker['ppid'], 'worker self parent')
    expected = dict(schema='cuda-core-cold-worker-v2', role=job['role'], node=job['node'], root=str(root), cwd=str(root),
                    launch_python=runtime['launch_python'], binary=runtime['binary'], binary_sha256=runtime['binary_sha256'],
                    orig_argv=command(root, job['node']), proc_argv=command(root, job['node']),
                    command_sha256=command_hash(command(root, job['node'])), manifest_ref=manifest_ref, test_ref=test_ref,
                    plugin_ref=plugin_ref, base_commit=manifest['base_commit'], candidate_tree=manifest['candidate_tree'])
    for key, value in expected.items():
        equal(worker[key], value, 'worker ' + key)
    threads = worker['python_threads']
    require(type(threads) is list and bool(threads), 'worker Python thread list')
    for thread in threads:
        exact(thread, 'name ident native_id', 'worker thread')
        string(thread['name'], 'worker thread name')
        integer(thread['ident'], 1, 'worker ident')
        integer(thread['native_id'], 1, 'worker native id')
    equal(threads, sorted(threads, key=lambda t: (t['name'], t['ident'], t['native_id'])), 'worker sorted threads')
    require(len({t['ident'] for t in threads}) == len(threads) == len({t['native_id'] for t in threads}), 'duplicate worker thread')
    require(worker['os_threads'] >= len(threads), 'worker OS/Python count')
    main = [t for t in threads if t['name'] == 'MainThread']
    require(len(main) == 1 and main[0]['native_id'] == worker['pid'], 'worker main PID linkage')


def validated(spec_path, role):
    require(type(role) is str and role in {r[0] for r in ROWS}, 'unknown cold role')
    job = next(j for j in JOBS if j['role'] == role)
    spec_path = canonical(str(spec_path))
    spec_ref = reference(spec_path)
    spec = read_json(spec_path)
    exact(spec, 'schema root_run_specs manifest_ref tool_seal_ref root output jobs', 'cold spec')
    equal(spec['schema'], 'bac-cold-import-spec-v2', 'cold schema')
    equal(spec['jobs'], JOBS, 'closed ordered jobs')
    root, output = canonical(spec['root']), canonical(spec['output'])
    role_dirs = [path for path in output.iterdir() if path.is_dir() or path.is_symlink()]
    require(all(path.name in {r[0] for r in ROWS} and not path.is_symlink() for path in role_dirs), 'unexpected role directory')
    run = canonical(str(output / role / 'run'))
    directory = canonical(str(run / role))
    require(directory.is_dir(), 'missing job directory')
    frozen = [spec_ref]
    for path in directory.glob('cold-import-*'):
        frozen.append(reference(path))
    for path in (run / (role + '-result.json'), run / (role + '.log')):
        frozen.append(reference(path))
    manifest_path = check_ref(spec['manifest_ref'])
    manifest = read_json(manifest_path)
    for key in ('base_commit', 'candidate_tree'):
        require(type(manifest[key]) is str and re.fullmatch('[0-9a-f]{40}', manifest[key]), 'invalid candidate identity')
    allowed = check_inventory(root, manifest)
    test_ref = reference(root / TEST)
    equal(test_ref['sha256'], manifest['source_files'][TEST], 'test source')
    probe = extract_probe((root / TEST).read_text(encoding='utf-8'))
    fault_sites(probe)
    seal_path = check_ref(spec['tool_seal_ref'])
    seal = read_json(seal_path)
    exact(seal, 'schema tools runtime', 'tool seal')
    equal(seal['schema'], 'bac-cold-import-tool-seal-v2', 'tool seal schema')
    exact(seal['tools'], ' '.join(TOOLS), 'tool inventory')
    for name, ref in seal['tools'].items():
        check_ref(ref, Path(__file__).resolve().parent / name)
    equal(seal['tools'][TOOLS[0]]['sha256'], OLD_SHA, 'old strict verifier')
    frozen.extend([spec['manifest_ref'], spec['tool_seal_ref'], *seal['tools'].values(), *spec['root_run_specs'].values()])
    runtime = seal['runtime']
    exact(runtime, 'launch_python binary binary_sha256 python_version torch_path torch_version numpy_path numpy_version', 'sealed runtime')
    equal(runtime['launch_python'], str(root / '.venv/bin/python'), 'venv launch spelling')
    for key in ('python_version', 'torch_version', 'numpy_version'):
        string(runtime[key], key)
    binary_ref = reference(canonical(runtime['binary']))
    equal(binary_ref['sha256'], runtime['binary_sha256'], 'interpreter binary')
    frozen.append(binary_ref)
    launch_path = Path(runtime['launch_python'])
    canonical(str(launch_path.parent))
    require(launch_path.is_file() and launch_path.resolve() == Path(runtime['binary']), 'launch interpreter binary mismatch')
    for key in ('torch_path', 'numpy_path'):
        require(canonical(runtime[key]).is_file(), 'runtime origin absent')
    exact(spec['root_run_specs'], ' '.join(r[0] for r in ROWS), 'root run specs')
    run_specs = {}
    for item in JOBS:
        path = check_ref(spec['root_run_specs'][item['role']])
        require(str(path) not in {str(p) for p in run_specs.values()}, 'duplicate run spec')
        harness_spec = read_json(path)
        equal(harness_spec, dict(root=str(root), manifest=str(manifest_path), output=str(output / item['role'] / 'run'),
                                workers=1, jobs=[dict(name=item['role'], args=command(root, item['node'])[1:])]), 'one-job harness spec')
        run_specs[item['role']] = path
    positive = job['mode'] == 'qualify'
    subject_name = 'cold-import-identity.json' if positive else 'cold-import-fault-witness.json'
    names = {'cold-import-worker.json', subject_name, 'cold-import-observation.json', 'cold-import-stdout.log', 'cold-import-stderr.log'}
    equal({p.name for p in directory.glob('cold-import-*')}, names, 'exclusive cold file inventory')
    require(all((directory / name).is_file() and not (directory / name).is_symlink() for name in names), 'invalid cold file')
    result_path = run / (role + '-result.json')
    result = read_json(result_path)
    exact(result, 'name exit_code seconds command log sha256 candidate_tree base_commit manifest_sha256 harness_sha256', 'root result')
    equal(result['exit_code'], 0, 'root result exit')
    require(type(result['seconds']) in (int, float) and math.isfinite(result['seconds']) and result['seconds'] > 0, 'root duration')
    log = run / (role + '.log')
    expected = dict(name=role, command=command(root, job['node']), log=str(log), sha256=reference(log)['sha256'],
                    candidate_tree=manifest['candidate_tree'], base_commit=manifest['base_commit'],
                    manifest_sha256=spec['manifest_ref']['sha256'], harness_sha256=HARNESS_SHA)
    for key, value in expected.items():
        equal(result[key], value, 'root result ' + key)
    worker_path = directory / 'cold-import-worker.json'
    worker_ref = reference(worker_path)
    worker = read_json(worker_path)
    check_worker(worker, job, root, runtime, manifest, spec['manifest_ref'], test_ref, seal['tools']['bac_cold_worker_identity.py'])
    markers = re.findall(r'^BAC_COLD_WORKER_SHA256=(.*)$', log.read_text(encoding='utf-8'), flags=re.MULTILINE)
    equal(markers, [worker_ref['sha256']], 'worker log marker')
    subject_path = directory / subject_name
    child = read_json(subject_path)
    exact(child, COMMON + ' ' + (POSITIVE if positive else FAULT), 'child')
    for key in ('pid', 'ppid'):
        integer(child[key], 1, key)
    require(child['pid'] not in (worker['pid'], worker['ppid']), 'child PID collision')
    nonce = child['nonce']
    require(type(nonce) is str and re.fullmatch('[0-9a-f]{32}', nonce), 'invalid nonce')
    argv = [runtime['launch_python'], '-I', '-c', probe, job['mode'], job['entry'], str(root), str(directory), nonce]
    expected = dict(schema='cuda-core-cold-import-identity-v2' if positive else 'cuda-core-cold-import-fault-witness-v2',
                    profile='linux-exec-cold-import-v2', role=role, mode=job['mode'], entry=job['entry'], ppid=worker['pid'],
                    worker_ref=worker_ref, root=str(root), cwd=str(root), orig_argv=argv, proc_argv=argv,
                    command_sha256=command_hash(argv), script_sha256=digest(probe.encode('utf-8')), manifest_ref=spec['manifest_ref'],
                    test_ref=test_ref, base_commit=manifest['base_commit'], candidate_tree=manifest['candidate_tree'])
    expected.update({key: runtime[key] for key in ('launch_python', 'binary', 'binary_sha256', 'python_version')})
    for key, value in expected.items():
        equal(child[key], value, 'child ' + key)
    observation_path = directory / 'cold-import-observation.json'
    observed = read_json(observation_path)
    exact(observed, 'schema role mode entry nonce parent_pid child_pid worker_start_ticks worker_ref launch_argv command_sha256 '
                    'script_sha256 subject_ref stdout_ref stderr_ref exitcode timed_out termination reaped seconds', 'observation')
    expected = dict(schema='cuda-core-cold-import-observation-v2', role=role, mode=job['mode'], entry=job['entry'], nonce=nonce,
                    parent_pid=worker['pid'], child_pid=child['pid'], worker_start_ticks=worker['start_ticks'], worker_ref=worker_ref,
                    launch_argv=argv, command_sha256=child['command_sha256'], script_sha256=child['script_sha256'],
                    subject_ref=reference(subject_path), stdout_ref=reference(directory / 'cold-import-stdout.log'),
                    stderr_ref=reference(directory / 'cold-import-stderr.log'), exitcode=0 if positive else 1,
                    timed_out=False, termination='none', reaped=True)
    for key, value in expected.items():
        equal(observed[key], value, 'observation ' + key)
    duration(observed['seconds'])
    if positive:
        check_positive(child, allowed, runtime)
    else:
        check_fault(child, allowed, probe, directory / 'cold-import-stderr.log')
    check_repository(root, manifest)
    check_inventory(root, manifest)
    for sealed_ref in frozen:
        check_ref(sealed_ref)
    equal({p.name for p in directory.glob('cold-import-*')}, names, 'final cold inventory')
    require(launch_path.resolve() == Path(runtime['binary']), 'launch interpreter drift')
    receipt = dict(schema='cuda-core-cold-import-custody-v2', status='COLD IMPORT POSITIVE VERIFIED' if positive else
                   'COLD IMPORT EXPECTED REJECTION VERIFIED', role=role, mode=job['mode'], entry=job['entry'],
                   spec_ref=reference(spec_path), root_run_spec_ref=reference(run_specs[role]), root_run_result_ref=reference(result_path),
                   root_run_log_ref=reference(log), tool_seal_ref=spec['tool_seal_ref'], worker_ref=worker_ref,
                   subject_ref=reference(subject_path), observation_ref=reference(observation_path), stdout_ref=observed['stdout_ref'],
                   stderr_ref=observed['stderr_ref'], base_commit=manifest['base_commit'], candidate_tree=manifest['candidate_tree'],
                   manifest_ref=spec['manifest_ref'], parent_pid=worker['pid'], child_pid=child['pid'],
                   script_sha256=child['script_sha256'], command_sha256=child['command_sha256'], scope=SCOPE)
    return receipt, directory / 'cold-import-custody.json'


def verify(spec_path, role):
    try:
        receipt, destination = validated(spec_path, role)
        with destination.open('x', encoding='utf-8') as stream:
            stream.write(json.dumps(receipt, indent=2) + '\n')
        return receipt
    except (KeyError, IndexError, TypeError, OSError, SyntaxError) as error:
        raise ValueError('invalid cold-import evidence: ' + str(error)) from error


if __name__ == '__main__':
    require(len(sys.argv) == 3, 'COLD_SPEC ROLE required')
    print(json.dumps(verify(sys.argv[1], sys.argv[2]), indent=2))
