        protocol.validate_record(kind, value)
    assert protocol.validate_spec(records["spec"]) is None


@pytest.mark.parametrize("change", [
    lambda s: s.update(schema="bac-hardware-v0"), lambda s: s.update(schema="future"),
    lambda s: s.update(extra=1), lambda s: s.update(job_concurrency=True),
    lambda s: s["budget"].update(cpu_slots=float("nan")),
    lambda s: s["jobs"].append(copy.deepcopy(s["jobs"][0])),
    lambda s: s["jobs"][0].update(output="/synthetic/output/attempt/../escaped"),
    lambda s: s["jobs"][0].update(output="/synthetic/output/attempt-other/job"),
    lambda s: s["jobs"][0].update(output="/synthetic/output/attempt"),
    lambda s: s["tools"].pop(), lambda s: s["tools"].append(copy.deepcopy(s["tools"][0])),
    lambda s: s["jobs"][0]["environment"].update(CUDA_VISIBLE_DEVICES="0"),
    lambda s: s["jobs"][0]["environment"].update(OMP_NUM_THREADS="2"),
    lambda s: s["jobs"][0].update(required_collected_ids=["x", "x"]),
    lambda s: s["jobs"][0].update(argv=["/wrong/python"]),
    lambda s: s["jobs"][0].update(cwd="/outside"),
    lambda s: s["jobs"][0]["roles"][0].update(intra_op=2),
    lambda s: s["jobs"][0]["roles"][0].update(role="invented-role"),
    lambda s: s["jobs"][0]["obligations"][0].update(kind="invented-obligation"),
    lambda s: s["jobs"][0].update(obligations=[]),
    lambda s: s.update(profile="cuda"), lambda s: s["jobs"][0].update(xdist_workers=16),
])
def test_closed_schema_refusals(protocol, records, change):
    value = copy.deepcopy(records["spec"])
    change(value)
    with pytest.raises(ValueError):
        protocol.validate_spec(value)

@pytest.mark.parametrize("disposition,code,signal", [("completed-failed", 1, 0), ("interrupted", None, 15),

    result = records["job-result"]
    result.update(disposition=disposition, exit_code=code, signal=signal, reason="synthetic refusal or termination")
