@pytest.mark.parametrize('mode', ['leaf', 'parent', 'fifo', 'directory'])
def test_unsafe_reference_forms(protocol, ref, tmp_path, mode):
    target = tmp_path / 'unsafe'
    if mode == 'leaf':
        target.symlink_to(ref['path'])
    elif mode == 'parent':
        target.symlink_to(tmp_path, target_is_directory=True)
        target /= 'inert'
    elif mode == 'fifo':
        os.mkfifo(target)
    else:
        target.mkdir()
    with pytest.raises(ValueError):
        protocol.file_reference(target)


def test_reference_parent_replacement_during_hash_rejected(protocol, tmp_path, monkeypatch):
    parent = tmp_path / 'parent'
    parent.mkdir()
    target = parent / 'artifact'
    target.write_bytes(b'original')
    old_read = protocol.os.read
    changed = False

    def replacing_read(fd, size):
        nonlocal changed
        data = old_read(fd, size)
        if not changed:
            changed = True
            parent.rename(tmp_path / 'retained-original-parent')
            parent.mkdir()
            target.write_bytes(b'replacement')
        return data

    monkeypatch.setattr(protocol.os, 'read', replacing_read)
    with pytest.raises(ValueError):
        protocol.file_reference(target)
