"""Independent inert probes; never import application/Torch or discover hardware."""
import hashlib
import os
from pathlib import Path
import pytest
import capture_bac_hardware_v1 as tool
import test_capture_bac_hardware_v1 as fixtures


@pytest.mark.parametrize('field', ['host', 'boot_id'])
def test_wrong_allocation_identity_refuses_preflight(field):
    obs, req = fixtures.observation(), fixtures.request()
    allocation = dict(host=obs['identity']['host'], boot_id=obs['identity']['boot_id'],
        authorized_cpu_ids=[0, 1, 2, 3],
        capacity=dict(cpu_slots=4, ram_bytes=1000, scratch_bytes=1000, scratch_inodes=20),
        overhead=dict(cpu_slots=1, ram_bytes=100, scratch_bytes=100, scratch_inodes=1))
    allocation[field] = 'wrong-allocation-identity'
    with pytest.raises(ValueError):
        tool.discovery_preflight(obs, req, allocation)


def test_persistent_pool_lock_replacement_refuses_at_release(tmp_path):
    root = tmp_path / 'leases'; root.mkdir()
    req = dict(lease_root=str(root), jobs=[])
    allocation = dict(pool_id='pool', allocation_id='one', partition_basis='exclusive-pool')
    with pytest.raises(ValueError):
        with tool.leases(req, allocation):
            path = root / (hashlib.sha256(b'pool:pool').hexdigest() + '.lock')
            replacement = root / 'replacement'; replacement.write_text('inert replacement')
            os.replace(replacement, path)


def test_capture_refuses_persistent_pool_lock_replacement_after_discovery(tmp_path, monkeypatch):
    candidate, request_path, source, req, env, registry = fixtures.capture_fixture(tmp_path)
    fixtures.stub_capture_observations(monkeypatch, tool, req, env, registry)
    original = tool.launch_discovery
    def replace_after_discovery(*args):
        result = original(*args)
        import json
        allocation = json.loads((tmp_path / 'allocation.json').read_text())
        path = Path(req['lease_root']) / (hashlib.sha256(('pool:' + allocation['pool_id']).encode()).hexdigest() + '.lock')
        replacement = path.parent / 'replacement'; replacement.write_text('inert replacement')
        os.replace(replacement, path)
        return result
    monkeypatch.setattr(tool, 'launch_discovery', replace_after_discovery)
    destination = tmp_path / 'capture'
    result = tool.capture(candidate, request_path, destination)
    assert result['disposition'] == 'refused', result
    assert result['spec'] is None
    assert not (destination / 'spec.json').exists()
