Synthetic observations exercise arithmetic only; actual qualification requires a
root-authorized source freeze and a separate observed discovery process. Neither
reference hashes nor admission establish actual test execution or CUDA behavior.
"""

PROTOCOL_SHA256 = "81b7e420987d1b8b0452c798445d7eb14f6157fd16e72961bf03f0479f1f9ea7"
DISCOVERY_FIELDS = {
    "schema", "profile", "identity", "at", "argv", "interpreter", "cwd", "inherited_visibility", "effective_visibility", "tmpdir",
    "cgroup_root", "cgroup_leaf", "ancestry_complete", "cgroups", "affinity", "meminfo", "scratch_free_bytes", "scratch_free_inodes",
    "cpu_topology", "numa_topology", "gpu_topology", "competing_workloads", "torch_build", "cuda_runtime", "driver", "devices",
    "probe_errors", "commands", "python_version", "dependencies", "torch_origin",
}
DEVICE_FIELDS = {"logical_index", "gpu_uuid", "mig_uuid", "physical_parent_uuid", "model", "vram_bytes", "isolation"}


def require(condition, reason):
    if not condition:
        raise ValueError(reason)


def canonical(value):
    require(type(value) is str and value.startswith("/") and "\x00" not in value and "\\" not in value, "absolute path required")
    require(value == "/" or all(p not in ("", ".", "..") for p in value[1:].split("/")), "noncanonical path")
    return Path(value)


def text_list(value):
    return type(value) is list and all(type(v) is str and v.strip() and "\x00" not in v for v in value)


def validate_discovery(obs):
    require(type(obs) is dict and set(obs) == DISCOVERY_FIELDS, "discovery exact fields required")
    require(obs["schema"] == "bac-hardware-discovery-v1" and obs["profile"] in ("cpu", "cuda"), "discovery version/profile")
    protocol.validate_record("identity", obs["identity"])
    require(type(obs["at"]) is str and re.fullmatch(r"\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d\.\d{6}Z", obs["at"]), "discovery timestamp")
    datetime.datetime.strptime(obs["at"], "%Y-%m-%dT%H:%M:%S.%fZ")
    for key in ("interpreter", "cwd", "tmpdir", "cgroup_root", "cgroup_leaf"):
        canonical(obs[key])
    require(type(obs["ancestry_complete"]) is bool, "discovery ancestry flag")
    require(text_list(obs["argv"]) and obs["argv"], "discovery argv")
    for key in ("meminfo", "cpu_topology", "numa_topology", "gpu_topology", "torch_build", "cuda_runtime", "driver", "inherited_visibility"):
        require(obs[key] is None or type(obs[key]) is str and "\x00" not in obs[key], "discovery text fact: " + key)
    require(type(obs["effective_visibility"]) is str, "discovery visibility")
    for key in ("scratch_free_bytes", "scratch_free_inodes"):
        require(obs[key] is None or type(obs[key]) is int and obs[key] >= 0, "discovery unsigned fact: " + key)
    affinity = obs["affinity"]
    require(affinity is None or type(affinity) is list and affinity and all(type(v) is int and v >= 0 for v in affinity)
            and len(set(affinity)) == len(affinity), "discovery affinity")
    require(obs["competing_workloads"] is None or text_list(obs["competing_workloads"]), "discovery competing workloads")
    require(text_list(obs["probe_errors"]), "discovery probe errors")
    require(type(obs["cgroups"]) is list and obs["cgroups"], "discovery cgroups required")
    for group in obs["cgroups"]:
        require(type(group) is dict and set(group) == {"path", "cpu_max", "cpuset", "memory_max", "memory_current"}, "cgroup fields")
        canonical(group["path"])
        require(all(group[k] is None or type(group[k]) is str for k in ("cpu_max", "cpuset", "memory_max", "memory_current")), "cgroup facts")
    require(type(obs["devices"]) is list, "discovery devices")
    for device in obs["devices"]:
        require(type(device) is dict and set(device) == DEVICE_FIELDS, "discovery device fields")
        for key in ("logical_index", "vram_bytes"):
            require(type(device[key]) is int and device[key] >= (1 if key == "vram_bytes" else 0), "device numeric fact")
        for key in ("gpu_uuid", "physical_parent_uuid", "model", "isolation"):
            require(type(device[key]) is str and device[key].strip(), "device identity fact")
        require(device["mig_uuid"] is None or type(device["mig_uuid"]) is str and device["mig_uuid"], "MIG identity fact")
    if obs["torch_origin"] is not None:
        protocol.validate_record("reference", obs["torch_origin"])
    require(type(obs["python_version"]) is str and obs["python_version"], "discovery Python version")
    require(type(obs["dependencies"]) is list, "discovery dependencies")
    for dependency in obs["dependencies"]:
        protocol.validate_record("dependency", dependency)
    require(type(obs["commands"]) is list, "discovery command inventory")
    for command in obs["commands"]:
        require(type(command) is dict and set(command) == {"argv", "executable", "identity", "start", "end", "exit_code", "stdout", "stderr", "errors"}, "discovery command fields")
        require(text_list(command["argv"]) and command["argv"] and type(command["exit_code"]) is int
                and type(command["stdout"]) is str and type(command["stderr"]) is str and text_list(command["errors"]), "discovery command types")
        protocol.validate_record("reference", command["executable"])
        if command["identity"] is not None:
            protocol.validate_record("identity", command["identity"])
        require(command["argv"][0] == command["executable"]["path"] and command["end"] >= command["start"], "command binding/time")


def cpuset(raw):
    require(type(raw) is str and raw.strip(), "cpuset unavailable")
    values = set()
    for part in raw.strip().split(","):
        require(re.fullmatch(r"\d+(?:-\d+)?", part) is not None, "invalid cpuset")
        endpoints = [int(x) for x in part.split("-")]
        low, high = endpoints[0], endpoints[-1]
        require(0 <= low <= high <= 1048575, "invalid cpuset range")
        current = set(range(low, high + 1))
        require(not current & values, "duplicate cpuset")
        values |= current
    return values


def assess_resources(request, obs, provenance):
    """Derive P0 admission fields from closed observations; never certify origin."""
    validate_discovery(obs)
    protocol.validate_record("reference", provenance)
    reasons = list(obs["probe_errors"])
    if reasons:
        reasons.insert(0, "required discovery probe failed")

    def fact(value, reason="", state="unavailable"):
        return {"state": "measured" if value is not None else state, "value": value,
                "reason": "" if value is not None else reason, "provenance": provenance}

    def numeric(raw, label):
        require(type(raw) is str and re.fullmatch(r"\d+", raw.strip()), label + " unavailable or malformed")
        return int(raw.strip())

    def observed(key):
        value = obs[key]
        if value is None:
            reasons.append(key + " unavailable")
        return fact(value, key + " unavailable")

    require(obs["profile"] == request["profile"], "discovery profile mismatch")
    fields = {key: observed(key) for key in ("affinity", "scratch_free_bytes", "scratch_free_inodes", "competing_workloads")}
    fields.update({key: fact(obs[key], key + " not observed", "unsupported")
                   for key in ("cpu_topology", "numa_topology", "gpu_topology")})
    fields.update(tmpdir=obs["tmpdir"], requested_visibility=request["requested_visibility"],
                  effective_visibility=obs["effective_visibility"], devices=[], cgroups=[])
    root, leaf = Path(obs["cgroup_root"]), Path(obs["cgroup_leaf"])
    if not leaf.is_relative_to(root) or not obs["ancestry_complete"]:
        reasons.append("cgroup ancestry is hidden or unsupported")
    expected = []
    if leaf.is_relative_to(root):
        current = leaf
        while True:
            expected.append(str(current))
            if current == root:
                break
            current = current.parent
        expected.reverse()
    if [g["path"] for g in obs["cgroups"]] != expected:
        reasons.append("cgroup ancestry incomplete or duplicate")
    cpus = set(obs["affinity"] or [])
    quotas, memory_limits, headrooms, currents = [], [], [], []
    cpu_known, memory_known = obs["affinity"] is not None, True
    for group in obs["cgroups"]:
        converted = {"path": group["path"]}
        try:
            require(type(group["cpu_max"]) is str, "cpu controller unavailable")
            parts = group["cpu_max"].split()
            require(len(parts) == 2, "cpu quota/period malformed")
            period = numeric(parts[1], "cpu period")
            require(period > 0, "cpu period must be positive")
            converted["cpu_period_us"] = fact(period)
            if parts[0] == "max":
                converted["cpu_quota_us"] = fact(None, "cpu.max is max", "unlimited")
            else:
                quota = numeric(parts[0], "cpu quota")
                require(quota > 0, "cpu quota must be positive")
                quotas.append(quota / period)
                converted["cpu_quota_us"] = fact(quota)
        except (ValueError, OverflowError) as exc:
            cpu_known = False
            reasons.append(str(exc))
            converted.update(cpu_quota_us=fact(None, str(exc)), cpu_period_us=fact(None, str(exc)))
        try:
            subset = cpuset(group["cpuset"])
            cpus &= subset
            converted["cpuset"] = fact(sorted(subset))
        except ValueError as exc:
            cpu_known = False
            reasons.append(str(exc))
            converted["cpuset"] = fact(None, str(exc))
        try:
            current_memory = numeric(group["memory_current"], "memory current")
            currents.append(current_memory)
            if group["memory_max"] == "max":
                converted["memory_limit_bytes"] = fact(None, "memory.max is max", "unlimited")
            else:
                limit = numeric(group["memory_max"], "memory limit")
                memory_limits.append(limit)
                headrooms.append(max(0, limit - current_memory))
                converted["memory_limit_bytes"] = fact(limit)
        except ValueError as exc:
            memory_known = False
            reasons.append(str(exc))
            converted["memory_limit_bytes"] = fact(None, str(exc))
        fields["cgroups"].append(converted)
    capacity = min([len(cpus), *quotas]) if cpu_known else None
    fields["effective_cpu_capacity"] = fact(capacity, "cpu/affinity/cpuset facts insufficient")
    host_total = host_available = None
    try:
        require(type(obs["meminfo"]) is str, "MemTotal/MemAvailable unavailable")
        memory = {}
        for line in obs["meminfo"].splitlines():
            match = re.fullmatch(r"(MemTotal|MemAvailable):\s*(\d+) kB", line)
            if match:
                require(match[1] not in memory, "duplicate memory observation")
                memory[match[1]] = int(match[2]) * 1024
        require(set(memory) == {"MemTotal", "MemAvailable"}, "MemTotal/MemAvailable unavailable")
        host_total, host_available = memory["MemTotal"], memory["MemAvailable"]
        require(0 < host_total and host_available <= host_total, "host memory observation inconsistent")
    except ValueError as exc:
        memory_known = False
        reasons.append(str(exc))
    limited = min([host_total, *memory_limits]) if host_total is not None and memory_known else None
    fields.update(host_ram_bytes=fact(host_total, "MemTotal unavailable"), limited_ram_bytes=fact(limited, "memory facts insufficient"),
                  ram_current_bytes=fact(currents[-1] if memory_known and currents else None, "memory current unavailable"))
    available = {"cpu_slots": math.floor(capacity) if capacity is not None else 0,
                 "ram_bytes": min([host_available, *headrooms]) if memory_known and host_available is not None else 0,
                 "scratch_bytes": obs["scratch_free_bytes"] or 0, "scratch_inodes": obs["scratch_free_inodes"] or 0}
    budget = {key: max(0, value - request["reserve"][key]) for key, value in available.items()}
    for key, minimum in request["minimum"].items():
        count = len(request["jobs"]) if key in ("scratch_bytes", "scratch_inodes") else request["requested_job_concurrency"]
        required = sum(sorted((job["budget"][key] for job in request["jobs"]), reverse=True)[:count])
        if budget[key] < max(minimum, required):
            reasons.append(key + " insufficient after reserve and concurrent job reservations")
    for job in request["jobs"]:
        if job["budget"]["cpu_slots"] < sum((r["count"] + r["replacement_limit"]) * r["peak_cpu_slots"] for r in job["roles"]):
            reasons.append("job CPU reservation misses concurrent roles/replacement overlap")
    if request["profile"] == "cpu":
        if obs["effective_visibility"] != "" or obs["devices"]:
            reasons.append("CPU discovery must hide CUDA")
        fields["effective_visibility"] = ""
    else:
        if not obs["cuda_runtime"] or not obs["torch_build"] or "cpu" in obs["torch_build"].lower():
            reasons.append("CUDA-capable Torch build/runtime unavailable")
        if not obs["driver"]:
            reasons.append("CUDA driver unavailable")
        if obs["inherited_visibility"] is not None and obs["inherited_visibility"] != request["requested_visibility"]:
            reasons.append("inherited CUDA visibility differs; expansion/remapping is not authorized")
        if obs["effective_visibility"] != request["requested_visibility"]:
            reasons.append("effective CUDA visibility differs from request")
        if obs["competing_workloads"]:
            reasons.append("competing GPU workloads preclude physical-exclusive admission")
        devices = obs["devices"]
        if not devices:
            reasons.append("CUDA device mapping unavailable")
        if len({d["gpu_uuid"] for d in devices}) != len(devices):
            reasons.append("duplicate GPU UUID")
        if [d["logical_index"] for d in devices] != list(range(len(devices))):
            reasons.append("logical device mapping is not contiguous")
        for device in devices:
            if device["gpu_uuid"] not in request["authorized_devices"]:
                reasons.append("GPU UUID is outside authorized allocation")
            if device["mig_uuid"] is not None or device["gpu_uuid"].startswith("MIG-"):
                reasons.append("MIG mapping unsupported; physical parent isolation unproved")
            if device["physical_parent_uuid"] != device["gpu_uuid"] or device["isolation"] != "physical-exclusive":
                reasons.append("physical parent mapping unsupported")
        for job in request["jobs"]:
            if job["profile"] == "cuda":
                raw_devices = [{k: v for k, v in d.items() if k != "mapping_provenance"} for d in job["devices"]]
                if raw_devices != devices or job["environment"].get("CUDA_VISIBLE_DEVICES") != obs["effective_visibility"]:
                    reasons.append("job device allocation/visibility differs from measured mapping")
        # Invalid raw mappings remain in discovery; refused P0 records contain no
        # invalid device objects that could accidentally look usable downstream.
        if not reasons:
            fields["devices"] = [dict(d, mapping_provenance=provenance) for d in devices]
    fields.update(reasons=list(dict.fromkeys(reasons)), admitted_budget=budget)
    return fields


def cgroup_paths(mountinfo, membership):
    """Only an unambiguous full cgroup-v2 mount is supported; never guess v1."""
    mounts = []
    for line in mountinfo.splitlines():
        sides = line.split(" - ")
        require(len(sides) == 2, "malformed mountinfo")
        before, after = sides[0].split(), sides[1].split()
        require(len(before) >= 6 and len(after) >= 3, "malformed mountinfo fields")
        if after[0] == "cgroup2":
            require(before[3] == "/", "cgroup ancestry hidden by mount root")
            mounts.append(canonical(before[4]))
    require(len(mounts) == 1, "unambiguous cgroup-v2 controller mount required")
    entries = membership.splitlines()
    require(len(entries) == 1 and entries[0].startswith("0::"), "unsupported or ambiguous cgroup membership")
    member = canonical(entries[0][3:])
    return mounts[0], mounts[0] / member.relative_to("/")


def utc():
    return datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%fZ")


def read_kernel(path):
    return Path(path).read_text()


def kernel_pids():
    return sorted(int(p.name) for p in Path("/proc").iterdir() if p.name.isdigit())


def process_identity(pid):
    raw = read_kernel(f"/proc/{pid}/stat")
    fields = raw.rsplit(")", 1)[1].split()
    return {"pid": pid, "ppid": int(fields[1]), "start_identity": fields[19],
            "host": socket.gethostname(), "boot_id": read_kernel("/proc/sys/kernel/random/boot_id").strip()}


def dependency_observations(environment):
    observed = []
    for dependency in environment["dependencies"]:
        if dependency["name"] == "nvidia-smi":
            continue  # Version and executable observed by the pinned driver query.
        dist = metadata.distribution(dependency["name"])
        suffix = "torch/__init__.py" if dependency["name"] == "torch" else ".dist-info/METADATA"
        matches = [p for p in dist.files or [] if str(p).endswith(suffix)]
        require(len(matches) == 1, "dependency distribution origin is ambiguous")
        origin = Path(dist.locate_file(matches[0])).resolve(strict=True)
        observed.append({"name": dependency["name"], "version": dist.version, "reference": protocol.file_reference(origin)})
    return observed


def collect_discovery(request, environment, inherited_visibility):
    """Called only inside the admitted separate discovery process; reads facts."""
    errors = []

    def observe(call, reason):
        try:
            return call()
        except (OSError, ValueError, KeyError, IndexError) as exc:
            errors.append(reason + ": " + str(exc))
            return None

    obs = {"schema": "bac-hardware-discovery-v1", "profile": request["profile"], "at": utc(),
           "identity": process_identity(os.getpid()), "argv": [str(Path(sys.executable).resolve()), *sys.argv],
           "interpreter": str(Path(sys.executable).resolve()), "cwd": str(Path.cwd()),
           "inherited_visibility": inherited_visibility, "effective_visibility": os.environ.get("CUDA_VISIBLE_DEVICES", ""),
           "tmpdir": str(Path(os.environ["TMPDIR"]).resolve(strict=True)),
           "cgroup_root": "/sys/fs/cgroup", "cgroup_leaf": "/sys/fs/cgroup", "ancestry_complete": False,
           "cgroups": [], "affinity": observe(lambda: sorted(os.sched_getaffinity(0)), "affinity"),
           "meminfo": observe(lambda: read_kernel("/proc/meminfo"), "memory"),
           "cpu_topology": observe(lambda: read_kernel("/proc/cpuinfo"), "CPU topology"),
           "numa_topology": None, "gpu_topology": None, "torch_origin": None, "torch_build": None, "cuda_runtime": None, "driver": None,
           "devices": [], "probe_errors": errors, "commands": [], "python_version": platform.python_version(),
           "dependencies": observe(lambda: dependency_observations(environment), "dependency origin/version") or []}
    require(obs["tmpdir"] == request["scratch_root"], "discovery TMPDIR differs from authorized scratch root")
    try:
        obs["numa_topology"] = read_kernel("/sys/devices/system/node/online")
    except OSError:
        pass  # Optional topology remains explicitly unsupported.
    try:
        root, leaf = cgroup_paths(read_kernel("/proc/self/mountinfo"), read_kernel("/proc/self/cgroup"))
        require(os.readlink("/proc/self/ns/cgroup") == os.readlink("/proc/1/ns/cgroup"), "cgroup namespace hides ancestors")
        require(leaf != root, "host cgroup ancestry cannot be established from namespace root membership")
        obs.update(cgroup_root=str(root), cgroup_leaf=str(leaf), ancestry_complete=True)
        paths = [leaf]
        while paths[-1] != root:
            paths.append(paths[-1].parent)
        for path in reversed(paths):
            group = {"path": str(path)}
            for key, name in [("cpu_max", "cpu.max"), ("cpuset", "cpuset.cpus.effective"),
                              ("memory_max", "memory.max"), ("memory_current", "memory.current")]:
                group[key] = observe(lambda path=path, name=name: read_kernel(path / name).strip(), "cgroup " + key)
            obs["cgroups"].append(group)
    except (OSError, ValueError) as exc:
        errors.append("cgroup ancestry: " + str(exc))
    if not obs["cgroups"]:
        obs["cgroups"] = [{"path": obs["cgroup_root"], "cpu_max": None, "cpuset": None, "memory_max": None, "memory_current": None}]
    usage = observe(lambda: os.statvfs(obs["tmpdir"]), "scratch filesystem")
    obs["scratch_free_bytes"] = usage.f_bavail * usage.f_frsize if usage is not None else None
    obs["scratch_free_inodes"] = usage.f_favail if usage is not None else None
    def workloads():
        found = []
        for pid in kernel_pids():
            if pid != os.getpid():
                try:
                    found.append(read_kernel(f"/proc/{pid}/stat").strip())
                except FileNotFoundError:
                    continue  # Process departed; no fabricated workload observation.
        return found
    obs["competing_workloads"] = observe(workloads, "competing workloads")
    if request["profile"] == "cuda":
        observe(lambda: collect_gpu(obs, environment), "CUDA discovery")
    return obs


def run_query(executable, tail, commands):
    protocol.validate_record("reference", executable)
    argv, started, errors = [executable["path"], *tail], utc(), []
    child = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                             text=True, env={"PATH": "/usr/bin:/bin", "LC_ALL": "C"})
    identity = None
    try:
        identity = process_identity(child.pid)
        require(identity["pid"] == child.pid and identity["ppid"] == os.getpid(), "driver child identity mismatch")
    except (OSError, ValueError, IndexError) as exc:
        errors.append("driver child identity unavailable: " + str(exc))
    try:
        stdout, stderr = child.communicate(timeout=30)
    except subprocess.TimeoutExpired:
        child.kill()
        stdout, stderr = child.communicate(timeout=30)
        errors.append("driver query exceeded fixed 30 second limit")
    commands.append({"argv": argv, "executable": executable, "identity": identity, "start": started, "end": utc(),
                     "exit_code": child.returncode, "stdout": stdout, "stderr": stderr, "errors": errors})
    protocol.validate_record("reference", executable)
    require(child.returncode == 0 and not errors, "driver query failed: " + "; ".join(errors))
    return stdout


def load_torch(reference):
    protocol.validate_record("reference", reference)
    spec = importlib.util.find_spec("torch")
    require(spec is not None and spec.origin is not None and protocol.file_reference(Path(spec.origin).resolve()) == reference,
            "Torch import origin differs from pinned dependency")
    torch = importlib.import_module("torch")
    require(protocol.file_reference(Path(torch.__file__).resolve()) == reference, "loaded Torch origin differs")
    return torch


def collect_gpu(obs, environment):
    dependencies = {d["name"]: d for d in environment["dependencies"]}
    require(len(dependencies) == len(environment["dependencies"]) and {"torch", "nvidia-smi"} <= dependencies.keys(),
            "CUDA requires exact pinned Torch and driver executable dependencies")
    executable = dependencies["nvidia-smi"]["reference"]
    query = lambda args: run_query(executable, args, obs["commands"])
    version = query(["--version"])
    match = re.search(r"^NVIDIA-SMI version\s*:\s*(\S+)\s*$", version, re.MULTILINE)
    require(match is not None and match[1] == dependencies["nvidia-smi"]["version"], "driver executable version mismatch")
    obs["dependencies"].append({"name": "nvidia-smi", "version": match[1], "reference": executable})
    raw = query(["--query-gpu=index,uuid,name,memory.total,driver_version,mig.mode.current", "--format=csv,noheader,nounits"])
    inventory, indices, drivers = {}, set(), set()
    for row in csv.reader(io.StringIO(raw), strict=True):
        row = [v.strip() for v in row]
        require(len(row) == 6, "driver GPU query columns")
        index, gpu, model, memory, driver, mig = row
        require(index.isdecimal() and int(index) not in indices and gpu.startswith("GPU-") and gpu not in inventory, "duplicate/unknown driver GPU identity")
        require(model and memory.isdecimal() and int(memory) > 0 and driver, "driver GPU model/memory/version unavailable")
        require(mig in ("Disabled", "[N/A]", "N/A"), "MIG enabled or mapping unsupported")
        indices.add(int(index)); drivers.add(driver)
        inventory[gpu] = {"model": model, "vram_bytes": int(memory) * 1024**2}
    require(inventory and len(drivers) == 1, "driver inventory/version missing or ambiguous")
    obs["driver"] = next(iter(drivers))
    workload = query(["--query-compute-apps=gpu_uuid,pid,process_name", "--format=csv,noheader,nounits"])
    workloads = []
    for row in csv.reader(io.StringIO(workload), strict=True):
        row = [v.strip() for v in row]
        require(len(row) == 3 and row[0] in inventory and row[1].isdecimal() and int(row[1]) > 0 and row[2], "GPU competing workload unproved")
        workloads.append(",".join(row))
    obs["competing_workloads"] = workloads
    try:
        obs["gpu_topology"] = query(["topo", "-m"])
    except ValueError:
        obs["gpu_topology"] = None  # Optional, with exact failed command retained.
    torch_ref = dependencies["torch"]["reference"]
    torch = load_torch(torch_ref)
    require(protocol.file_reference(Path(torch.__file__).resolve()) == torch_ref, "Torch actual origin mismatch")
    obs.update(torch_origin=torch_ref, torch_build=str(torch.__version__), cuda_runtime=torch.version.cuda)
    require(obs["cuda_runtime"] and torch.cuda.is_available(), "CPU-only Torch or CUDA runtime unavailable")
    visible = obs["effective_visibility"].split(",")
    require(visible and len(set(visible)) == len(visible) and (all(v.isdecimal() for v in visible) or all(v.startswith("GPU-") for v in visible)),
            "unsupported or mixed CUDA visibility")
    count = torch.cuda.device_count()
    require(type(count) is int and count == len(visible) and count > 0, "CUDA visibility/cardinality mismatch")
    devices = []
    for index in range(count):
        props = torch.cuda.get_device_properties(index)
        gpu = props.uuid
        if type(gpu) is bytes and len(gpu) == 16:
            gpu = "GPU-" + str(uuid.UUID(bytes=gpu))
        require(type(gpu) is str and gpu in inventory and gpu not in {d["gpu_uuid"] for d in devices}, "unknown/duplicate Torch GPU UUID")
        require(props.name == inventory[gpu]["model"] and type(props.total_memory) is int
                and 0 < props.total_memory <= inventory[gpu]["vram_bytes"], "Torch/driver model or VRAM mismatch")
        if visible[0].startswith("GPU-"):
            require(gpu == visible[index], "full UUID visibility order mismatch")
        devices.append({"logical_index": index, "gpu_uuid": gpu, "mig_uuid": None, "physical_parent_uuid": gpu,
                        "model": props.name, "vram_bytes": props.total_memory, "isolation": "physical-exclusive"})
    obs["devices"] = devices
