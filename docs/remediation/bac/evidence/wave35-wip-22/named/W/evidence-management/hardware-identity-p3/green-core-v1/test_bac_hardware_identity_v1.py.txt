def test_xdist_environment_probe(request):
    """Discovery only: persist actual framework environment for root contract."""
    destination = os.environ.get('BAC_P3_ENV_PROBE')
    if not destination:
        pytest.skip('explicit inert environment discovery only')
    worker = getattr(request.config, 'workerinput', {})
    output = Path(destination) / (worker.get('workerid', 'controller') + '.json')
    with output.open('x') as stream:
        json.dump({'environment': dict(os.environ), 'workerinput': worker,
                   'pid': os.getpid(), 'ppid': os.getppid(), 'argv': sys.argv}, stream, indent=2)


def _probe_hook(config, phase):
    destination = os.environ.get('BAC_P3_ENV_PROBE')
    if destination:
        worker = getattr(config, 'workerinput', {})
        with (Path(destination) / (worker.get('workerid', 'controller') + '-' + phase + '.json')).open('x') as stream:
            json.dump({'environment': dict(os.environ), 'workerinput': worker,
                       'pid': os.getpid(), 'ppid': os.getppid(), 'argv': sys.argv}, stream, indent=2)


def pytest_configure(config):
    _probe_hook(config, 'configure')


def pytest_sessionfinish(session, exitstatus):
    _probe_hook(session.config, 'finish')


def pytest_configure_node(node):
    destination = os.environ.get('BAC_P3_ENV_PROBE')
    if destination:
        with (Path(destination) / ('node-' + node.gateway.id + '-configure.json')).open('x') as stream:
            json.dump({'workerinput': node.workerinput, 'gateway': node.gateway.id}, stream, indent=2)


def pytest_testnodedown(node, error):
    destination = os.environ.get('BAC_P3_ENV_PROBE')
    if destination:
        with (Path(destination) / ('node-' + node.gateway.id + '-down.json')).open('x') as stream:

def _prepare(capsule, monkeypatch):
    api=_api()
    def git(argv, **kwargs):
        assert kwargs['cwd'] == capsule['root'] or str(kwargs['cwd']) == str(capsule['root'])
        assert argv in [['git','rev-parse','HEAD'],['git','diff-index','--cached','--quiet','b'*40,'--']]
        return subprocess.CompletedProcess(argv,0,stdout='a'*40+'\n' if 'rev-parse' in argv else '',stderr='')
    monkeypatch.setattr(api.subprocess,'run',git)
    return api


def test_process_custody(capsule,monkeypatch):
    api=_prepare(capsule,monkeypatch); c=capsule['context']
    start=_read(api.record_process(c,'start')); end=_read(api.record_process(c,'end'))
    assert start['identity']['pid']==os.getpid()
    assert start['identity']['ppid']==os.getppid()
    assert start['identity']['start_identity']==Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()[19]
    assert start['identity']==end['identity']
    assert start['argv']==[v.decode() for v in Path('/proc/self/cmdline').read_bytes().split(b'\0') if v]
    assert end['elapsed_ns']['value']==end['monotonic_ns']-start['monotonic_ns']
    assert end['imports']==[{'name':'inert_module','reference':capsule['blobref']}]
    assert start['environment']==dict(os.environ)
    assert start['interpreter']==protocol.file_reference(Path(sys.executable).resolve())
    assert len(list(capsule['output'].glob('process-*-start.json'))) == 2  # raw and P0 start
    with pytest.raises(ValueError): api.record_process(c,'end')

def test_nonperturbing_observer(capsule,monkeypatch):
    api=_prepare(capsule,monkeypatch)
    torch=types.ModuleType('torch');cuda=types.ModuleType('torch.cuda');cuda._initialized=False
    def forbidden(*a,**k): raise AssertionError('observer called Torch or CUDA')
    torch.__getattr__=forbidden;cuda.__getattr__=forbidden
    for name in ['get_num_threads','get_num_interop_threads','is_anomaly_enabled']:setattr(torch,name,forbidden)
    torch.cuda=cuda;monkeypatch.setitem(sys.modules,'torch',torch);monkeypatch.setitem(sys.modules,'torch.cuda',cuda)
    value=_read(api.record_process(capsule['context'],'start'))
    assert value['torch_observation']=='loaded-without-calls'
    assert value['cuda_observation']=='uninitialized'
    assert all(value[k]['state']=='unavailable' for k in ['intra_op','inter_op','debug'])
    raw=_read(value['intra_op']['provenance'])
    assert raw['cuda_marker'] is False and 'native' in raw['observation_limit']
