"""Actual process witnesses. No application imports, CUDA discovery or RNG work.

Authority is supplied explicitly by root/accepted owner or the root-delegated P2
runtime descriptor. Hashes alone never authenticate that delegation. P4/P5 own
whole-job custody and case/hardware acceptance, including missing final witnesses.
"""
import datetime
import fcntl
import hashlib
import json
import os
from pathlib import Path
import re
import socket
import subprocess
import sys
import time
import types

import bac_hardware_protocol_v1 as protocol

PROTOCOL_SHA = 'adc8efeba368998e2e875a60c54ed3338a5ad2aa7daa710b7609e9ed2119de3c'
CONTEXT_KEYS = {'schema','spec','command','trusted_grant','job_id','role_id','required_imports',
                'worker_roles','environment_delta','parent_start','replacement_start','lifecycle',
                'worker_identity','policy','runtime'}
POLICY_KEYS = {'schema','spec_path','command_path','runtime_path','job_id','role_id',
               'required_imports','worker_roles','environment_delta'}
GRANT_KEYS = {'schema','authorized_root','head','tree','import_roots','inputs','start','end',
              'no_writer_barrier','includes_untracked','authority','allocation_grant','cpu_reservation_evidence'}
_STATE = None


def _require(ok, message):
    if not ok:
        raise ValueError(message)


def _exact(value, keys, label):
    _require(type(value) is dict and set(value) == set(keys), label + ': exact fields required')


def _utc():
    return datetime.datetime.now(datetime.timezone.utc).strftime('%Y-%m-%dT%H:%M:%S.%fZ')


def _pairs(pairs):
    result = {}
    for key, value in pairs:
        _require(key not in result, 'duplicate JSON field')
        result[key] = value
    return result


def _load(reference, kind=None):
    protocol.validate_record('reference', reference)
    value = json.loads(Path(reference['path']).read_text(), object_pairs_hook=_pairs)
    _require(protocol.file_reference(Path(reference['path'])) == reference, 'reference changed while reading')
    if kind:
        protocol.validate_record(kind, value)
    return value


def _directory(path):
    """No-follow directory capability; caller must close. P0 checks spelling."""
    raw = str(path)
    _require(raw.startswith('/') and '..' not in Path(raw).parts, 'unsafe directory')
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY)
    try:
        for part in Path(raw).parts[1:]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = child
        return fd
    except BaseException:
        os.close(fd)
        raise


def _write(output, name, value, kind=None):
    if kind:
        protocol.validate_record(kind, value)
    _require(Path(name).name == name, 'unsafe output filename')
    parent = _directory(output)
    try:
        fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600, dir_fd=parent)
        with os.fdopen(fd, 'wb') as stream:
            stream.write((json.dumps(value, sort_keys=True, indent=2) + '\n').encode())
            stream.flush()
            os.fsync(stream.fileno())
        # Verify namespace still names the retained directory and exact output.
        live = _directory(output)
        try:
            _require(os.fstat(parent).st_ino == os.fstat(live).st_ino and
                     os.fstat(parent).st_dev == os.fstat(live).st_dev, 'output ancestor changed')
        finally:
            os.close(live)
        return protocol.file_reference(Path(output) / name)
    finally:
        os.close(parent)


def _identity(pid=None):
    pid = os.getpid() if pid is None else pid
    stat = Path(f'/proc/{pid}/stat').read_text().rsplit(')', 1)[1].split()
    return dict(pid=pid, ppid=int(stat[1]), start_identity=stat[19],
                host=socket.gethostname(), boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip())


def _process_id(identity):
    return 'p-' + hashlib.sha256(json.dumps(identity, sort_keys=True).encode()).hexdigest()


def _imports(context, candidate, environment, phase):
    allowed = {item['reference']['path']:item['reference'] for group in candidate['inventory'].values() for item in group}
    allowed.update({item['reference']['path']:item['reference'] for item in environment['dependencies']})
    result = []
    names = set()
    for required in context['required_imports']:
        _exact(required, {'name','reference','phases'}, 'required import')
        protocol.validate_record('named-reference', {k:required[k] for k in ('name','reference')})
        _require(required['name'] not in names, 'duplicate import name')
        names.add(required['name'])
        phases = required['phases']
        _require(type(phases) is list and bool(phases) and len(phases)==len(set(phases)) and
                 set(phases) <= {'start','end'}, 'invalid import phases')
        reference = required['reference']
        _require(allowed.get(reference['path']) == reference, 'import not authorized by input/dependency inventory')
        module = sys.modules.get(required['name'])
        if module is None:
            _require(phase not in phases, 'required actual import absent')
            continue
        _require(type(module) is types.ModuleType, 'unrecognized imported module')
        origin = vars(module).get('__file__')
        _require(type(origin) is str and protocol.file_reference(Path(origin)) == reference, 'actual import origin mismatch')
        result.append(dict(name=required['name'], reference=reference))
    # Every loaded source-root file must be frozen, including imports not requested.
    for name, module in list(sys.modules.items()):
        if type(module) is not types.ModuleType:
            continue
        origin = vars(module).get('__file__')
        if type(origin) is str and Path(origin).is_absolute() and Path(origin).is_relative_to(candidate['authorized_root']):
            ref = protocol.file_reference(Path(origin))
            _require(allowed.get(ref['path']) == ref, 'unfrozen loaded source import')
    return result


def _environment(context, job):
    delta = context['environment_delta']
    _exact(delta, {'common','worker'}, 'environment delta')
    for group in delta.values():
        _require(type(group) is dict, 'environment addition map required')
    _require(set(delta['common']) <= {'PYTEST_VERSION','LC_CTYPE'} and
             set(delta['worker']) <= {'COLUMNS','LINES'}, 'unaccepted environment addition')
    expected = dict(job['environment'])
    additions = dict(delta['common'])
    worker = context['worker_identity']
    if worker is not None:
        _worker_identity(worker)
        additions.update(delta['worker'])
        additions.update(PYTEST_XDIST_WORKER=worker['workerid'],
                         PYTEST_XDIST_WORKER_COUNT=str(worker['workercount']),
                         PYTEST_XDIST_TESTRUNUID=worker['testrunuid'])
    _require(not set(expected) & set(additions), 'framework cannot override launch environment')
    expected.update(additions)
    actual = dict(os.environ)
    _require(actual == expected, 'actual environment differs from sealed launch and framework additions')
    return actual


def _worker_identity(value):
    _exact(value, {'workerid','workercount','testrunuid','mainargv'}, 'worker identity')
    _require(type(value['workerid']) is str and re.fullmatch(r'gw[0-9]+', value['workerid']) is not None,
             'invalid actual worker ID')
    _require(type(value['workercount']) is int and value['workercount'] > 0, 'invalid worker count')
    _require(type(value['testrunuid']) is str and re.fullmatch('[0-9a-f]{32}', value['testrunuid']) is not None,
             'invalid testrunuid')
    _require(type(value['mainargv']) is list and value['mainargv'] and
             all(type(v) is str and v for v in value['mainargv']), 'invalid worker argv')


def _source(context, spec, candidate, identity, output, name, at):
    freeze = _load(spec['source_freeze'], 'source-freeze')
    _require(candidate['source_freeze'] == spec['source_freeze'] and
             freeze['grant'] == context['trusted_grant'], 'root source grant binding mismatch')
    grant = _load(context['trusted_grant'])
    _exact(grant, GRANT_KEYS, 'root grant')
    _require(grant['schema'] == 'bac-hardware-source-grant-v1' and grant['no_writer_barrier'] is True and
             grant['includes_untracked'] is True and type(grant['authority']) is str and bool(grant['authority'].strip()),
             'root no-writer authority missing')
    inputs = [item['reference'] for group in candidate['inventory'].values() for item in group]
    _require(len({r['path'] for r in inputs}) == len(inputs), 'duplicate candidate input')
    for key in ('authorized_root','head','tree','import_roots'):
        _require(grant[key] == candidate[key], 'grant candidate mismatch: '+key)
    _require(grant['inputs'] == inputs and freeze['authorized_root'] == spec['authorized_root'] == candidate['authorized_root'],
             'root/inventory authority mismatch')
    _require(grant['start'] <= at and (grant['end'] is None or at <= grant['end']) and freeze['start'] <= at and
             (freeze['end']['state'] == 'unavailable' or at <= freeze['end']['value']), 'source freeze interval inactive')
    protocol.validate_record('reference', grant['allocation_grant'])
    _require(type(grant['cpu_reservation_evidence']) is list and grant['cpu_reservation_evidence'], 'missing CPU authority')
    for ref in grant['cpu_reservation_evidence']:
        protocol.validate_record('reference', ref)
    for ref in inputs:
        protocol.validate_record('reference', ref)
    lockpath = Path(freeze['index_lock'])
    parent = _directory(lockpath.parent)
    try:
        lock = os.open(lockpath.name, os.O_RDWR | os.O_NOFOLLOW, dir_fd=parent)
    finally:
        os.close(parent)
    try:
        fcntl.flock(lock, fcntl.LOCK_EX)
        head = subprocess.run(['git','rev-parse','HEAD'], cwd=Path(spec['authorized_root']),
                              capture_output=True, text=True, check=False)
        tree = subprocess.run(['git','diff-index','--cached','--quiet',candidate['tree'],'--'],
                              cwd=Path(spec['authorized_root']), capture_output=True, text=True, check=False)
        _require(head.returncode == 0 and head.stdout.strip() == candidate['head'] and tree.returncode == 0,
                 'source HEAD/index changed')
    finally:
        fcntl.flock(lock, fcntl.LOCK_UN)
        os.close(lock)
    witness = dict(schema='bac-hardware-v1',kind='source-witness',candidate=spec['candidate'],
                   source_freeze=spec['source_freeze'],head=candidate['head'],tree=candidate['tree'],
                   inventory=inputs,at=at,process_identity=identity)
    return _write(output, name, witness, 'source-witness')


def _torch(role, phase):
    torch = sys.modules.get('torch')
    loaded = type(torch) is types.ModuleType
    missing = dict(intra_op=None, inter_op=None, debug=None)
    if role['role'] == 'fork-guard':
        return missing, 'forbidden-inherited-module-witness', 'forbidden', loaded, None, 'guard-only: no Torch/CUDA calls permitted'
    work = role['observer'] == 'work' or role['torch_policy'] == 'work'
    if not loaded:
        _require(not work or phase == 'start', 'work end requires loaded Torch configuration')
        return missing, 'not-loaded', 'uninitialized', False, None, 'Torch absent; no import or runtime configuration observation'
    cuda = sys.modules.get('torch.cuda')
    _require(type(cuda) is types.ModuleType, 'loaded CUDA module observation unavailable')
    if not work:
        marker = vars(cuda).get('_initialized')
        _require(type(marker) is bool, 'loaded Torch CUDA marker unavailable/unrecognized')
        facts = missing
        observation = 'loaded-without-calls'
        reason = 'loaded Torch initialization marker only, not every native CUDA context; runtime calls prohibited'
    else:
        facts = {}
        for key, getter in [('intra_op','get_num_threads'),('inter_op','get_num_interop_threads'),('debug','is_anomaly_enabled')]:
            fn = vars(torch).get(getter)
            _require(callable(fn), 'required loaded Torch getter unavailable')
            value = fn()
            _require(type(value) is type(role[key]) and value == role[key], 'role configuration mismatch: '+key)
            facts[key] = value
        fn = vars(cuda).get('is_initialized')
        _require(callable(fn), 'loaded CUDA getter unavailable')
        marker = fn()
        _require(type(marker) is bool, 'invalid CUDA state observation')
        observation = 'work'
        reason = 'loaded Torch initialization state only, not independent proof of native device work'
    _require(role['cuda_policy'] != 'no-initialize' or marker is False, 'required uninitialized role already initialized')
    _require(role['torch_policy'] != 'forbidden', 'Torch observation forbidden for role')
    return facts, observation, 'initialized' if marker else 'uninitialized', True, marker, reason


def _policy(path, digest):
    ref = protocol.file_reference(Path(path))
    _require(ref['sha256'] == digest, 'static identity policy digest mismatch')
    policy = _load(ref)
    _exact(policy, POLICY_KEYS, 'identity policy')
    _require(policy['schema'] == 'bac-hardware-identity-policy-v1', 'unknown static policy')
    return ref, policy


def _authority(context, spec, job, identity):
    _require(len(context['policy']) == len(context['runtime']), 'authority transport cardinality')
    if not context['policy']:
        return  # Direct owner API: caller supplies explicit out-of-band authority.
    policy = _load(context['policy'][0])
    _exact(policy, POLICY_KEYS, 'identity policy')
    _require(policy['schema'] == 'bac-hardware-identity-policy-v1', 'wrong policy schema')
    argv = job['argv']
    for key, value in [('--bac-identity-policy',context['policy'][0]['path']),
                       ('--bac-identity-policy-sha256',context['policy'][0]['sha256'])]:
        _require(argv.count(key) == 1 and argv.index(key)+1 < len(argv) and argv[argv.index(key)+1] == value,
                 'policy not pinned by exact job argv')
    for key in ('job_id','required_imports','worker_roles','environment_delta'):
        _require(context[key] == policy[key], 'derived context changed static policy')
    _require(policy['spec_path'] == context['spec']['path'] and
             policy['command_path'] == context['command']['path'] == str(Path(job['output'])/'command.json') and
             policy['runtime_path'] == context['runtime'][0]['path'] == str(Path(job['output'])/'runtime-context.json'),
             'runtime known path mismatch')
    candidate = _load(spec['candidate'], 'candidate')
    _require(not any(entry['path'] == context['policy'][0]['path'] for group in candidate['inventory'].values() for entry in group),
             'static policy cannot occur in candidate inventory')
    runtime = _load(context['runtime'][0])
    _exact(runtime, {'schema','spec','command','trusted_grant','policy','runner_identity','runner_source_start'}, 'runtime descriptor')
    _require(runtime['schema'] == 'bac-hardware-identity-runtime-v1', 'unknown runtime descriptor')
    for key in ('spec','command','trusted_grant'):
        _require(runtime[key] == context[key], 'runtime context binding mismatch')
    _require(runtime['policy'] == context['policy'][0], 'runtime policy mismatch')
    source = _load(runtime['runner_source_start'], 'source-witness')
    _require(source['process_identity'] == runtime['runner_identity'] and source['candidate'] == spec['candidate'] and
             source['source_freeze'] == spec['source_freeze'], 'runner delegation source mismatch')
    if context['worker_identity'] is None and not context['parent_start']:
        _require(context['role_id'] == policy['role_id'] and runtime['runner_identity'] == _identity(identity['ppid']),
                 'runtime not delegated by actual live runner parent')


def _parent(context, spec, job, identity, role):
    parents = []
    replacements = []
    if context['parent_start']:
        start = _load(context['parent_start'][0], 'process')
        _require(start['phase']=='start' and start['spec']==context['spec'] and start['job_id']==job['id'] and
                 start['identity'] == _identity(identity['ppid']), 'actual parent process identity mismatch')
        parents.append(start['identity'])
    if context['worker_identity'] is not None:
        _require(role['role']=='worker' and bool(parents) and bool(context['lifecycle']), 'worker lifecycle missing')
        lifecycle = _load(context['lifecycle'][0])
        _exact(lifecycle, {'schema','phase','controller_start','worker_identity','role_id','predecessor_start',
                           'process_start','error','at','prior_lifecycle'}, 'worker lifecycle')
        _require(lifecycle['schema']=='bac-hardware-worker-lifecycle-v1' and lifecycle['phase']=='configure' and
                 lifecycle['controller_start']==context['parent_start'][0] and lifecycle['worker_identity']==context['worker_identity'] and
                 lifecycle['role_id']==role['id'] and lifecycle['predecessor_start']==context['replacement_start'] and
                 lifecycle['process_start']==[] and lifecycle['error']=='', 'worker configure association mismatch')
        _require(context['worker_identity']['workercount']==job['xdist_workers'], 'worker count mismatch')
    if context['replacement_start']:
        previous = _load(context['replacement_start'][0], 'process')
        _require(role['replacement_limit'] > 0 and previous['phase']=='start' and previous['role_id']==role['id'] and
                 previous['spec']==context['spec'] and previous['identity'] != identity and context['lifecycle'],
                 'unadmitted/ambiguous replacement')
        lifecycle = _load(context['lifecycle'][0])
        _require(len(lifecycle['prior_lifecycle'])==1, 'replacement missing predecessor lifecycle')
        down = _load(lifecycle['prior_lifecycle'][0])
        _require(down['phase']=='down' and down['error'] and down['process_start']==context['replacement_start'] and
                 down['role_id']==role['id'], 'replacement missing actual failed predecessor')
        replacements.append(previous['identity'])
    return parents, replacements


def record_process(context: dict, phase: str) -> dict:
    """Record only observations made by this actual process; preserve partials."""
    try:
        _exact(context, CONTEXT_KEYS, 'identity context')
        _require(context['schema']=='bac-hardware-identity-context-v1' and phase in ('start','end'), 'unknown context/phase')
        for key in ('parent_start','replacement_start','lifecycle','policy','runtime'):
            _require(type(context[key]) is list and len(context[key]) <= 1, 'context reference cardinality')
        _require(protocol.file_reference(Path(protocol.__file__))['sha256']==PROTOCOL_SHA, 'protocol bytes unadopted')
        spec = _load(context['spec'], 'spec')
        jobs = [job for job in spec['jobs'] if job['id']==context['job_id']]
        _require(len(jobs)==1, 'unknown job')
        job = jobs[0]
        roles = [role for role in job['roles'] if role['id']==context['role_id']]
        _require(len(roles)==1, 'unadmitted role')
        role = roles[0]
        tools = {tool['name']:tool['reference'] for tool in spec['tools']}
        _require(tools['identity']==protocol.file_reference(Path(__file__)) and
                 tools['protocol']==protocol.file_reference(Path(protocol.__file__)), 'identity/protocol tool binding mismatch')
        environment = _load(spec['dependencies'], 'environment')
        interpreter = protocol.file_reference(Path(sys.executable).resolve())
        _require(interpreter==environment['interpreter'] and str(Path(sys.executable).resolve())==str(Path(job['interpreter']).resolve()),
                 'actual interpreter mismatch')
        _require(os.getcwd()==job['cwd'], 'actual cwd mismatch')
        command = _load(context['command'], 'command')
        _require(command['spec']==context['spec'] and command['attempt_id']==spec['attempt_id'] and command['job_id']==job['id'],
                 'command identity mismatch')
        for key in ('argv','cwd','environment','seed'):
            _require(command[key]==job[key], 'command launch mismatch')
        _require(command['runner']==tools['runner'], 'command runner mismatch')
        candidate = _load(spec['candidate'], 'candidate')
        identity = _identity()
        pid = _process_id(identity)
        output = Path(job['output'])
        process_path = output/f'process-{pid}-{phase}.json'
        _require(not process_path.exists(), 'duplicate process phase')
        _authority(context,spec,job,identity)
        actual_env = _environment(context,job)
        parents,replacements = _parent(context,spec,job,identity,role)
        now = time.monotonic_ns()
        at = _utc()
        if phase=='end':
            start = _load(protocol.file_reference(output/f'process-{pid}-start.json'),'process')
            _require(start['identity']==identity and start['spec']==context['spec'] and start['role_id']==role['id'] and
                     start['parent_identity']==parents and start['replacement_of']==replacements and now>=start['monotonic_ns'],
                     'end not paired with actual start')
            elapsed = dict(state='measured',value=now-start['monotonic_ns'],reason='')
        else:
            elapsed = dict(state='unavailable',value=None,reason='process start has no elapsed interval')
        imports = _imports(context,candidate,environment,phase)
        facts,torch_observation,cuda_observation,loaded,marker,reason = _torch(role,phase)
        source = _source(context,spec,candidate,identity,output,f'source-{pid}-{phase}.json',at)
        raw = _write(output,f'process-{pid}-observation-{phase}.json',dict(
            schema='bac-hardware-process-observation-v1',context=context,identity=identity,phase=phase,
            environment=actual_env,worker_identity=context['worker_identity'],torch_loaded=loaded,
            cuda_marker=marker,observation_limit=reason,at=at))
        measured = {key:dict(state='measured' if value is not None else 'unavailable',value=value,
                            reason='' if value is not None else reason,provenance=raw) for key,value in facts.items()}
        process = dict(schema='bac-hardware-v1',kind='process',attempt_id=spec['attempt_id'],job_id=job['id'],
            spec=context['spec'],process_id=pid,role_id=role['id'],role=role['role'],phase=phase,identity=identity,
            parent_identity=parents,replacement_of=replacements,interpreter=interpreter,
            argv=[value.decode() for value in Path('/proc/self/cmdline').read_bytes().split(b'\0') if value],
            cwd=os.getcwd(),imports=imports,source_witness=source,environment=actual_env,devices=job['devices'],
            **measured,torch_observation=torch_observation,cuda_observation=cuda_observation,
            at=at,monotonic_ns=now,elapsed_ns=elapsed)
        return _write(output,process_path.name,process,'process')
    except (OSError,KeyError,TypeError,json.JSONDecodeError) as exc:
        raise ValueError('process witness refused: '+str(exc)) from exc


def pytest_addoption(parser):
    group = parser.getgroup('bac-hardware-identity')
    group.addoption('--bac-identity-policy', action='store', default=None)
    group.addoption('--bac-identity-policy-sha256', action='store', default=None)


def pytest_configure(config):
    global _STATE
    path, digest = config.getoption('--bac-identity-policy'), config.getoption('--bac-identity-policy-sha256')
    if path is None and digest is None:
        return
    _require(path is not None and digest is not None, 'both identity policy options required')
    policy_ref, policy = _policy(path,digest)
    runtime_ref = protocol.file_reference(Path(policy['runtime_path']))
    runtime = _load(runtime_ref)
    _exact(runtime, {'schema','spec','command','trusted_grant','policy','runner_identity','runner_source_start'}, 'runtime descriptor')
    context = dict(schema='bac-hardware-identity-context-v1',spec=runtime['spec'],command=runtime['command'],
        trusted_grant=runtime['trusted_grant'],policy=[policy_ref],runtime=[runtime_ref],
        **{key:policy[key] for key in ('job_id','role_id','required_imports','worker_roles','environment_delta')},
        parent_start=[],replacement_start=[],lifecycle=[],worker_identity=None)
    workerinput = getattr(config,'workerinput',None)
    if workerinput is not None:
        actual = {key:workerinput[key] for key in ('workerid','workercount','testrunuid','mainargv')}
        derived = workerinput.get('bac_identity_context')
        _exact(derived,CONTEXT_KEYS,'derived worker context')
        for key in CONTEXT_KEYS - {'role_id','parent_start','replacement_start','lifecycle','worker_identity'}:
            _require(derived[key] == context[key], 'worker changed static context')
        _require(derived['worker_identity'] == actual, 'workerinput actual identity mismatch')
        context = derived
    spec = _load(context['spec'],'spec')
    job = next(job for job in spec['jobs'] if job['id']==context['job_id'])
    _require(type(context['worker_roles']) is list and len(context['worker_roles'])==job['xdist_workers'], 'worker policy count mismatch')
    for role in job['roles']:
        if role['role']=='worker':
            _require(context['worker_roles'].count(role['id'])==role['count'], 'worker policy role count mismatch')
    start = record_process(context,'start')
    _STATE = dict(context=context,spec=spec,job=job,start=start,initial=0,nodes={},failed=[],replacements={},report_keys=set())


def _worker_start(worker_id):
    matches = []
    for path in Path(_STATE['job']['output']).glob('process-*-start.json'):
        if '-observation-' in path.name:
            continue
        ref = protocol.file_reference(path)
        value = _load(ref,'process')
        if value['role']=='worker' and value['environment'].get('PYTEST_XDIST_WORKER')==worker_id:
            _require(value['spec']==_STATE['context']['spec'] and value['parent_identity']==[_load(_STATE['start'])['identity']],
                     'worker start outside controller custody')
            matches.append(ref)
    _require(len(matches)==1,'worker start absent/ambiguous; cannot infer replacement predecessor')
    return matches[0]


def _lifecycle(worker, role_id, phase, predecessor, process, prior, error):
    value = dict(schema='bac-hardware-worker-lifecycle-v1',phase=phase,controller_start=_STATE['start'],
        worker_identity=worker,role_id=role_id,predecessor_start=predecessor,process_start=process,error=error,
        at=_utc(),prior_lifecycle=prior)
    return _write(_STATE['job']['output'],f'worker-{worker["workerid"]}-{phase}.json',value)


def pytest_configure_node(node):
    if _STATE is None:
        return
    worker = {key:node.workerinput[key] for key in ('workerid','workercount','testrunuid','mainargv')}
    _worker_identity(worker)
    _require(worker['workerid']==node.gateway.id,'gateway and workerinput mismatch')
    _require(worker['workerid'] not in _STATE['nodes'],'reused worker label')
    prior,predecessor = [],[]
    if _STATE['initial'] < len(_STATE['context']['worker_roles']):
        role_id = _STATE['context']['worker_roles'][_STATE['initial']]
        _STATE['initial'] += 1
    else:
        _require(bool(_STATE['failed']),'replacement has no failed lifecycle predecessor')
        down_ref = _STATE['failed'].pop(0)
        down = _load(down_ref)
        role_id = down['role_id']
        count = _STATE['replacements'].get(role_id,0)+1
        role = next(role for role in _STATE['job']['roles'] if role['id']==role_id)
        _require(count <= role['replacement_limit'],'unadmitted worker replacement')
        _STATE['replacements'][role_id] = count
        predecessor,prior = down['process_start'],[down_ref]
    configure = _lifecycle(worker,role_id,'configure',predecessor,[],prior,'')
    context = dict(_STATE['context'],role_id=role_id,parent_start=[_STATE['start']],
                   replacement_start=predecessor,lifecycle=[configure],worker_identity=worker)
    node.workerinput['bac_identity_context'] = context
    _STATE['nodes'][worker['workerid']] = configure


def pytest_testnodedown(node,error):
    if _STATE is None:
        return
    configure_ref = _STATE['nodes'][node.gateway.id]
    configure = _load(configure_ref)
    start = _worker_start(node.gateway.id)
    down = _lifecycle(configure['worker_identity'],configure['role_id'],'down',configure['predecessor_start'],
                      [start],[configure_ref],'' if error is None else str(error))
    if error is not None:
        _STATE['failed'].append(down)


def pytest_collection_finish(session):
    if _STATE is None:
        return
    if _STATE['job']['xdist_workers'] and _STATE['context']['worker_identity'] is None:
        return
    ids = [item.nodeid for item in session.items]
    _require(len(ids)==len(set(ids)), 'duplicate collected IDs')
    process_id = _load(_STATE['start'])['process_id']
    _write(_STATE['job']['output'],f'collection-{process_id}.json',dict(
        schema='bac-hardware-pytest-collection-v1',process=_STATE['start'],ids=ids))


def pytest_runtest_logreport(report):
    if _STATE is None or (_STATE['job']['xdist_workers'] and _STATE['context']['worker_identity'] is None):
        return
    _require(report.when in ('setup','call','teardown'),'unknown pytest report phase')
    key = (report.nodeid,report.when)
    _require(key not in _STATE['report_keys'],'duplicate case phase report')
    _STATE['report_keys'].add(key)
    outcome = report.outcome
    if hasattr(report,'wasxfail'):
        outcome = 'xfailed' if report.skipped else 'xpassed'
    value = dict(schema='bac-hardware-pytest-report-v1',process=_STATE['start'],case_id=report.nodeid,
                 phase=report.when,outcome=outcome,longrepr=str(report.longrepr) if report.longrepr is not None else '',at=_utc())
    process_id = _load(_STATE['start'])['process_id']
    digest = hashlib.sha256((report.nodeid+'\0'+report.when).encode()).hexdigest()
    _write(_STATE['job']['output'],f'report-{process_id}-{digest}.json',value)


def _completion():
    context,job,spec = _STATE['context'],_STATE['job'],_STATE['spec']
    processes,sources,obligations,inventory,collections,reports = [],[],[],[],[],{}
    for path in sorted(Path(job['output']).glob('*.json')):
        ref = protocol.file_reference(path)
        value = _load(ref)
        inventory.append(ref)
        kind = value.get('kind')
        if kind=='process':
            protocol.validate_record(kind,value)
            _require(value['spec']==context['spec'] and value['job_id']==job['id'],'foreign process evidence')
            processes.append(ref)
        elif kind=='source-witness':
            protocol.validate_record(kind,value)
            _require(value['candidate']==spec['candidate'] and value['source_freeze']==spec['source_freeze'],'foreign source evidence')
            sources.append(ref)
        elif kind=='obligation-evidence':
            protocol.validate_record(kind,value)
            _require(value['spec']==context['spec'] and value['job_id']==job['id'],'foreign obligation evidence')
            obligations.append(ref)
        elif value.get('schema')=='bac-hardware-pytest-collection-v1':
            _exact(value,{'schema','process','ids'},'pytest collection')
            emitter = _load(value['process'],'process')
            _require(emitter['spec']==context['spec'] and emitter['job_id']==job['id'],'foreign collection emitter')
            _require(type(value['ids']) is list and len(value['ids'])==len(set(value['ids'])),'invalid collection')
            collections.append(value['ids'])
        elif value.get('schema')=='bac-hardware-pytest-report-v1':
            _exact(value,{'schema','process','case_id','phase','outcome','longrepr','at'},'pytest report')
            emitter = _load(value['process'],'process')
            _require(emitter['spec']==context['spec'] and emitter['job_id']==job['id'],'foreign report emitter')
            case,phase = value['case_id'],value['phase']
            _require(phase in ('setup','call','teardown'),'unknown report phase')
            row = reports.setdefault(case,dict(case_id=case,setup='not-run',call='not-run',teardown='not-run',evidence=[]))
            _require(row[phase]=='not-run','duplicate case phase across processes')
            row[phase] = value['outcome']
            row['evidence'].append(ref)
    _require(bool(collections) and all(c==collections[0] for c in collections),'missing/differing process collections')
    envelope = dict(schema='bac-hardware-v1',kind='job-evidence',attempt_id=spec['attempt_id'],job_id=job['id'],
        spec=context['spec'],command=context['command'],collected_ids=collections[0],
        outcomes=[reports[key] for key in sorted(reports)],process_records=processes,
        obligation_records=obligations,source_witnesses=sources,inventory=inventory)
    return _write(job['output'],'completion.json',envelope,'job-evidence')


def pytest_sessionfinish(session,exitstatus):
    if _STATE is None:
        return
    record_process(_STATE['context'],'end')
    if _STATE['context']['worker_identity'] is None:
        _completion()
