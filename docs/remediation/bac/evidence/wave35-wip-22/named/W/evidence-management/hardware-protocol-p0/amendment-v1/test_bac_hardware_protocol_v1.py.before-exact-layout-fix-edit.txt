def test_reference_parent_replacement_during_hash_refuses(protocol, tmp_path, monkeypatch):
    parent = tmp_path / "original"
    parent.mkdir()
    target = parent / "data"
    target.write_bytes(b"original bytes")
    original_read = protocol.os.read
    changed = False

    def read_and_replace(fd, length):
        nonlocal changed
        chunk = original_read(fd, length)
        if chunk and not changed:
            changed = True
            parent.rename(tmp_path / "retained")
            parent.mkdir()
            target.write_bytes(b"replacement bytes")
        return chunk

    monkeypatch.setattr(protocol.os, "read", read_and_replace)
    with pytest.raises(ValueError):
        protocol.file_reference(target)
