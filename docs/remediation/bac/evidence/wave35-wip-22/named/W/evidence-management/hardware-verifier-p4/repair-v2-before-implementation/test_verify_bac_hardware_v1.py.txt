def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, sort_keys=True) + "\n")
    return p.file_reference(path)


def obs(value):
    return {"state": "measured", "value": value, "reason": ""}


def absent():
    return {"state": "unavailable", "value": None, "reason": "not observed"}


def record(kind, **fields):
    return {"schema": p.SCHEMA, "kind": kind, **fields}

            self.output / ("process-" + process_id + "-observation-" + phase + ".json"),
            {
                "schema": "bac-hardware-process-observation-v1",
                "context": context,
                "identity": actual_identity,
                "phase": phase,
                "environment": environment,
                "worker_identity": context["worker_identity"],
                "torch_loaded": False,
                "cuda_marker": None,
                "observation_limit": "synthetic only",
                "at": at,
            },
        )
        return write(
            self.output / ("process-" + process_id + "-" + phase + ".json"),
            record(
                "process",
                attempt_id="attempt",
                job_id="job",
                spec=self.spec_ref,
                process_id=process_id,
                role_id=context["role_id"],
                role="worker" if is_worker else "controller",
                phase=phase,
                identity=actual_identity,
                parent_identity=[self.identity] if is_worker else [],
                replacement_of=(
                    [json.loads(Path(context["replacement_start"][0]["path"]).read_text())["identity"]]
                    if context["replacement_start"]
                    else []
                ),
                interpreter=self.interpreter,
                argv=[self.interpreter["path"], "-c", "inert"] if is_worker else self.argv,
                cwd=str(self.src),
                imports=[],
                source_witness=self.source("source-" + process_id + "-" + phase + ".json", actual_identity, at),
                environment=environment,
                devices=self.devices,
                **{k: dict(**absent(), provenance=raw) for k in ("intra_op", "inter_op", "debug")},
                torch_observation="not-loaded",
                cuda_observation="uninitialized",
                at=at,
                monotonic_ns=mono,
                elapsed_ns=absent() if phase == "start" else obs(10 if is_worker else 100)
            ),
        )

    def seal(self):
        files = [x for x in self.output.iterdir() if x.is_file() and x.name not in ("result.json", "completion.json")]
        completion = record(
            "job-evidence",
            **{
                k: self.result[k]
                for k in (
                    "attempt_id",
                    "job_id",
                    "spec",
                    "command",
                    "collected_ids",
                    "outcomes",
                    "process_records",
                    "obligation_records",
                    "source_witnesses",
                )
            },
            inventory=[p.file_reference(x) for x in sorted(files)]
        )
        write(self.output / "completion.json", completion)
        self.result["inventory"] = [p.file_reference(x) for x in sorted(self.output.iterdir()) if x.is_file() and x.name != "result.json"]
        self.result_ref = write(self.output / "result.json", self.result)
        refs = [p.file_reference(x) for x in sorted(self.output.parent.rglob("*")) if x.is_file() and x.name != "seal.json"]
        write(
            self.output.parent / "seal.json",
            record(
                "attempt-seal",
                attempt_id="attempt",
                spec=self.spec_ref,
                start=self.start_ref,
                disposition=self.result["disposition"],
                job_results=[self.result_ref],
                inventory=[*refs, p.file_reference(self.capture / "result.json")],
                at=T4,
                source_freeze=self.freeze,
            ),
        )

    def verify(self, name="receipt.json"):
        return verifier().verify(str(self.spec_path), "job", str(self.root / name))

def coherent_change(evidence, mutate):
    """Alter inert observations, then rehash their reference DAG without stale-reference failures."""
    for path in sorted(evidence.root.rglob("*.json")):
        value = json.loads(path.read_text())
        mutate(path, value)
        write(path, value)
    for _ in range(30):
        changed = False

        def refresh(value):
            if type(value) is dict:
                if set(value) == {"path", "bytes", "sha256"}:
                    return p.file_reference(Path(value["path"]))
                return {k: refresh(v) for k, v in value.items()}
            return [refresh(v) for v in value] if type(value) is list else value

        for path in sorted(evidence.root.rglob("*.json")):
            value = json.loads(path.read_text())
            updated = refresh(value)
            if updated != value:
                write(path, updated)
                changed = True
        if not changed:
            return
    raise AssertionError("synthetic reference DAG did not converge")

def change_worker_identity(evidence, old_worker, field, replacement):
    def mutate(_path, value):
        def visit(item):
            if type(item) is dict:
                if item.get("workerid") == old_worker:
                    item[field] = replacement
                environment_key = {"workerid": "PYTEST_XDIST_WORKER", "testrunuid": "PYTEST_XDIST_TESTRUNUID"}.get(field)
                if environment_key and item.get("PYTEST_XDIST_WORKER") == old_worker:
                    item[environment_key] = replacement
                for child in item.values():
                    visit(child)
            elif type(item) is list:
                for child in item:
                    visit(child)

        visit(value)

    coherent_change(evidence, mutate)
