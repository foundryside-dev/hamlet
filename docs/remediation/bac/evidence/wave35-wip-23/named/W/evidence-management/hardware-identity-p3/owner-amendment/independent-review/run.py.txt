import datetime,hashlib,json,os,subprocess,sys
from pathlib import Path

names=['bac_hardware_identity_v1.py','test_bac_hardware_identity_v1.py','bac_hardware_protocol_v1.py']
def refs():
 return [dict(path=str(q/n),bytes=(q/n).stat().st_size,sha256=hashlib.sha256((q/n).read_bytes()).hexdigest()) for n in names]
def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
