def test_xdist_environment_probe(request):
    """Discovery only: persist actual framework environment for root contract."""
    destination = os.environ.get('BAC_P3_ENV_PROBE')
    if not destination:
        pytest.skip('explicit inert environment discovery only')
    worker = getattr(request.config, 'workerinput', {})
    output = Path(destination) / (worker.get('workerid', 'controller') + '.json')
    with output.open('x') as stream:
        json.dump({'environment': dict(os.environ), 'workerinput': worker,
                   'pid': os.getpid(), 'ppid': os.getppid(), 'argv': sys.argv}, stream, indent=2)


def _probe_hook(config, phase):
    destination = os.environ.get('BAC_P3_ENV_PROBE')
    if destination:
        worker = getattr(config, 'workerinput', {})
        with (Path(destination) / (worker.get('workerid', 'controller') + '-' + phase + '.json')).open('x') as stream:
            json.dump({'environment': dict(os.environ), 'workerinput': worker,
                       'pid': os.getpid(), 'ppid': os.getppid(), 'argv': sys.argv}, stream, indent=2)

def _prepare(capsule, monkeypatch):
    api=_api()
    monkeypatch.delenv('PYTEST_CURRENT_TEST',raising=False)
    def git(argv, **kwargs):
        assert kwargs['cwd'] == capsule['root'] or str(kwargs['cwd']) == str(capsule['root'])
        assert argv in [['git','rev-parse','HEAD'],['git','diff-index','--cached','--quiet','b'*40,'--']]
        return subprocess.CompletedProcess(argv,0,stdout='a'*40+'\n' if 'rev-parse' in argv else '',stderr='')
    monkeypatch.setattr(api,'subprocess',types.SimpleNamespace(run=git))
    return api

def _child_emit(context,connection,crash):
    import bac_hardware_identity_v1 as api
    def git(argv,**kwargs):
        if argv==['git','rev-parse','HEAD']: return subprocess.CompletedProcess(argv,0,stdout='a'*40+'\n')
        assert argv==['git','diff-index','--cached','--quiet','b'*40,'--']
        return subprocess.CompletedProcess(argv,0,stdout='')
    api.subprocess=types.SimpleNamespace(run=git)
    module=types.ModuleType('inert_module');module.__file__=context['required_imports'][0]['reference']['path']
    sys.modules['inert_module']=module
    torch=types.ModuleType('torch');cuda=types.ModuleType('torch.cuda');cuda._initialized=False
    def forbidden(): raise AssertionError('fork guard called Torch')
    torch.get_num_threads=forbidden if context['role_id']=='fork-guard' else lambda:1
    torch.get_num_interop_threads=forbidden if context['role_id']=='fork-guard' else lambda:5
    torch.is_anomaly_enabled=forbidden if context['role_id']=='fork-guard' else lambda:True
    cuda.is_initialized=forbidden if context['role_id']=='fork-guard' else lambda:False
    sys.modules['torch']=torch;sys.modules['torch.cuda']=cuda
    try:
        start=api.record_process(context,'start');connection.send(start)
        if crash: os._exit(17)
        connection.send(api.record_process(context,'end'))
    except BaseException as exc:
        expected=_read(context['spec'])['jobs'][0]['environment']
        connection.send({'error':repr(exc),'env_diff':{k:[expected.get(k),os.environ.get(k)] for k in set(expected)|set(os.environ) if expected.get(k)!=os.environ.get(k)}})
        raise
    finally:
        connection.close()


@pytest.mark.parametrize('kind,method,crash',[('spawn-child','spawn',False),('fork-guard','fork',False),('spawn-child','spawn',True)])
def test_actual_child_process_custody(capsule,monkeypatch,kind,method,crash):
    import multiprocessing
    api=_prepare(capsule,monkeypatch);_strict_role(capsule,kind)
    parent=api.record_process(capsule['context'],'start')
    context=copy.deepcopy(capsule['context']);context['role_id']=kind;context['parent_start']=[parent]
    mp=multiprocessing.get_context(method);receiver,sender=mp.Pipe(duplex=False)
    process=mp.Process(target=_child_emit,args=(context,sender,crash));process.start();sender.close()
    assert receiver.poll(10), 'child never emitted actual start'
    start=receiver.recv();assert 'error' not in start,start
    first=_read(start)
    assert first['identity']['pid']==process.pid and first['identity']['ppid']==os.getpid()
    assert first['identity']!=_read(parent)['identity'] and first['parent_identity']==[_read(parent)['identity']]
    if crash:
        process.join(10);assert process.exitcode==17
        assert not Path(start['path'].replace('-start.json','-end.json')).exists()
    else:
        assert receiver.poll(10);end=receiver.recv();assert 'error' not in end,end
        last=_read(end);process.join(10);assert process.exitcode==0
        assert first['identity']==last['identity']
        if kind=='fork-guard':
            assert last['torch_observation']=='forbidden-inherited-module-witness'
            assert last['cuda_observation']=='forbidden'
        else:
            assert [last[k]['value'] for k in ['intra_op','inter_op','debug']]==[1,5,True]
    receiver.close()

def test_preloaded_fork_recorder_zero_imports_baseline(capsule,monkeypatch):
    import builtins
    api=_prepare(capsule,monkeypatch);_strict_role(capsule,'fork-guard')
    parent=api.record_process(capsule['context'],'start')
    context=dict(capsule['context'],role_id='fork-guard',parent_start=[parent])
    read,write=os.pipe();pid=os.fork()
    if pid==0:
        os.close(read)
        def forbidden(name,*args,**kwargs): raise AssertionError('forbidden import: '+name)
        builtins.__import__=forbidden
        try:
            api.record_process(context,'start');api.record_process(context,'end');message=b'OK'
        except BaseException as exc: message=str(exc).encode()
        os.write(write,message);os._exit(0)
    os.close(write);message=os.read(read,4096);os.close(read);os.waitpid(pid,0)
    assert message==b'OK',message.decode()
