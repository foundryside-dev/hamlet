def _loaded_sources() -> dict[str, Any]:
    """Observe existing modules and source bytes, without a fresh child import."""
    representatives = {
        "env": (sys.modules[VectorizedHamletEnv.__module__], VectorizedHamletEnv.step.__wrapped__.__code__),
        "helper": (parent_custody, parent_custody.EnvCustody.check_process.__code__),
        "probe": (sys.modules[__name__], _fork_refusals.__code__),
    }
    observations = {}
    for label, (module, code) in representatives.items():
        path = Path(module.__file__).resolve()
        # marshal's reference/intern encoding is not a canonical code identity.
        # Observe stable public bytes/fields, retaining exact module/code IDs.
        structure = {
            "bytecode_sha256": hashlib.sha256(code.co_code).hexdigest(),
            "linetable_sha256": hashlib.sha256(code.co_linetable).hexdigest(),
            "exceptiontable_sha256": hashlib.sha256(code.co_exceptiontable).hexdigest(),
            "name": code.co_name, "qualname": code.co_qualname,
            "firstlineno": code.co_firstlineno, "flags": code.co_flags,
            "argcount": code.co_argcount, "posonlyargcount": code.co_posonlyargcount,
            "kwonlyargcount": code.co_kwonlyargcount, "nlocals": code.co_nlocals,
            "stacksize": code.co_stacksize, "names": list(code.co_names),
            "varnames": list(code.co_varnames), "freevars": list(code.co_freevars),
            "cellvars": list(code.co_cellvars),
        }
        observations[label] = {
            "module_name": module.__name__, "module_id": id(module),
            "module_file": str(path), "spec_origin": module.__spec__.origin,
            "loader_type": type(module.__loader__).__name__,
            "source_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
            "code_id": id(code), "code_filename": code.co_filename,
            "code_structure": structure,
        }
    return observations


def _candidate_binding(sources: dict[str, Any]) -> dict[str, Any]:
    root = Path(sources["env"]["module_file"]).parents[3]
    manifest = root / ".superpowers/sdd/2026-11-01-bac-remediation-upgrades/wave35/r144-candidate.json"
    raw = manifest.read_bytes()
    candidate = json.loads(raw)
    assert candidate["candidate_tree"] == "ed561dc04fcffba4f6617f343862d54db6a8dad0"
    for label in ("env", "helper"):
        relative = str(Path(sources[label]["module_file"]).relative_to(root))
        assert sources[label]["source_sha256"] == candidate["source_files"][relative]
    return {
        "manifest": str(manifest), "manifest_sha256": hashlib.sha256(raw).hexdigest(),
        "candidate_tree": candidate["candidate_tree"], "base_commit": candidate["base_commit"],
        "probe_binding": "external qualification source; separately hash-bound by root, not in candidate source tree",
    }


class Controller:
    def __init__(self) -> None:
        self.domain = process_rng_domain()
        self.participant = self.domain.register(owner=self, cuda_devices=(), entrypoints=frozenset({"component"}))

    def invocation(self) -> Any:
        return self.domain.device_invocation(participant=self.participant, operation_kind="component", lineage=None, cuda_devices=())


def _fork_refusals(env: Any, scope: Any, operation: Any, accepted: Any, participation: Any, predecessor: Any) -> dict[str, Any]:
    """Child performs no tensor work and never unwinds inherited contexts."""
    parent_pid = os.getpid()
    source_identity = {
        "env": id(env), "scope": id(scope), "operation": id(operation),
        "accepted": id(accepted), "reverse": id(env._env_custody.reverse),
    }
    parent_sources = _loaded_sources()
    candidate_binding = _candidate_binding(parent_sources)
    read_fd, write_fd = os.pipe()
    try:
        child_pid = os.fork()
    except BaseException:
        os.close(read_fd)
        os.close(write_fd)
        raise
    if child_pid == 0:
        # Never return from this branch: outer pytest/domain/RLock contexts belong
        # to the parent. Only inherited descriptors and child-local tripwires run.
        os.close(read_fd)
        exit_code = 91
        try:
            attempts: list[str] = []

            class LockTripwire:
                def __enter__(self) -> Any:
                    attempts.append("owner-lock")
                    raise AssertionError("inherited owner lock reached")

                def __exit__(self, *args: Any) -> None:
                    raise AssertionError("inherited owner lock exit reached")

            def domain_tripwire() -> Any:
                attempts.append("domain")
                raise AssertionError("inherited domain lookup reached")

            # No PID spoofing and no authority replacement: these child-only
            # negative tripwires establish the refusal cut before either lock.
            env._env_custody.lock = LockTripwire()
            parent_custody.process_rng_domain = domain_tripwire
            checks = {
                "ordinary_read": lambda: env.get_action_masks(),
                "phase_diagnostic": lambda: env.world_phase,
                "retention": lambda: env.retain_parent_custody().__enter__(),
                "binding": lambda: scope.bind_operation(operation=operation).__enter__(),
                "reverse_validation": lambda: env.validate_initial_participation(
                    accepted=accepted, participation=participation, predecessor=predecessor
                ),
                "reverse_cleanup": lambda: env.abandon_episode(accepted=accepted, reason="runtime_failure"),
            }
            outcomes = {}
            for name, check in checks.items():
                try:
                    check()
                except ValueError as error:
                    outcomes[name] = {"type": type(error).__name__, "message": str(error)}
                else:
                    raise AssertionError(f"inherited {name} succeeded")
            report = {
                "parent_pid": parent_pid, "child_pid": os.getpid(), "child_ppid": os.getppid(),
                "source_identity": source_identity,
                "child_identity": {
                    "env": id(env), "scope": id(scope), "operation": id(operation),
                    "accepted": id(accepted), "reverse": id(env._env_custody.reverse),
                },
                "creator_pid": env._env_custody.pid, "scope_pid": scope._pid,
                "refusals": outcomes, "lock_or_domain_attempts": attempts,
                "code_provenance": "inherited already-loaded parent modules and code objects; no fresh child APP import",
                "parent_before_fork_sources": parent_sources,
                "child_after_refusals_sources": _loaded_sources(),
                "parent_candidate_binding": candidate_binding,
                "child_candidate_binding": _candidate_binding(_loaded_sources()),
            }
            payload = json.dumps(report, sort_keys=True).encode()
            assert len(payload) < 65536
            offset = 0
            while offset < len(payload):
                written = os.write(write_fd, payload[offset:])
                assert written > 0
                offset += written
            exit_code = 0
        except BaseException:
            exit_code = 92
        finally:
            os.close(write_fd)
            os._exit(exit_code)
    os.close(write_fd)
    reaped = False
    payload = bytearray()
    status = None
    deadline = time.monotonic() + 8
    try:
        eof = False
        while time.monotonic() < deadline and not (reaped and eof):
            if not eof:
                ready, _, _ = select.select([read_fd], [], [], 0.05)
                if ready:
                    chunk = os.read(read_fd, 4096)
                    payload.extend(chunk)
                    eof = not chunk
                    assert len(payload) < 65536
            if not reaped:
                waited_pid, observed_status = os.waitpid(child_pid, os.WNOHANG)
                if waited_pid == child_pid:
                    reaped, status = True, observed_status
        assert reaped and eof, "owned fork did not terminate and close its evidence pipe"
        assert status is not None and os.WIFEXITED(status) and os.WEXITSTATUS(status) == 0
        report = json.loads(payload)
        report.update(waitpid=child_pid, wait_status=status, exit_code=os.WEXITSTATUS(status), pipe_eof=eof)
        # Preserve actual pipe contents even if a subsequent identity assertion fails.
        # This is an observation, not a successful qualification declaration.
        print("ENV_C2_FORK_OBSERVED=" + json.dumps(report, sort_keys=True))
        assert report["child_pid"] == child_pid != parent_pid
        assert report["parent_pid"] == report["child_ppid"] == parent_pid
        assert report["creator_pid"] == report["scope_pid"] == parent_pid
        assert report["source_identity"] == report["child_identity"] == source_identity
        assert len(report["refusals"]) == 6
        assert all("another process" in outcome["message"] for outcome in report["refusals"].values())
        assert report["lock_or_domain_attempts"] == []
        assert report["parent_before_fork_sources"] == report["child_after_refusals_sources"] == parent_sources
        assert report["parent_candidate_binding"] == report["child_candidate_binding"] == candidate_binding
        return report
    finally:
        os.close(read_fd)
        if not reaped:
            # Only this exact, unreaped child can be killed; PID cannot be reused
            # before our wait. Never kill a process group or an unowned worker.
            try:
                os.kill(child_pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
            os.waitpid(child_pid, 0)


@pytest.mark.skipif(not hasattr(os, "fork"), reason="requires an actual POSIX fork")
def test_actual_inherited_pid_refuses_before_domain_and_owner_lock(compiled: Any, monkeypatch: Any) -> None:
    claim = VectorizedPopulation.claim_vector_step
    commit = VectorizedHamletEnv.commit_initial_participation
    state: dict[str, Any] = {}
    reports = []

    def capture_commit(self: Any, **kwargs: Any) -> Any:
        result = commit(self, **kwargs)
        state["participation"] = result
        return result

    def observed_claim(self: Any, **kwargs: Any) -> int:
        assert self.env._env_custody.reverse is not None
        reports.append(_fork_refusals(
            self.env, state["scope"], state["operation"], kwargs["accepted"],
            state["participation"], kwargs["predecessor"],
        ))
        return claim(self, **kwargs)

    monkeypatch.setattr(VectorizedHamletEnv, "commit_initial_participation", capture_commit)
    monkeypatch.setattr(VectorizedPopulation, "claim_vector_step", observed_claim)
    rig = make_parent(compiled, "none", "shared")
    rig.enter()
    controller = Controller()
    with rig.env.retain_parent_custody() as scope, rig.runtime.retain_parent_custody() as runtime_scope:
        invocation = controller.invocation()
        with invocation as operation:
            state.update(scope=scope, operation=operation)
            with scope.bind_operation(operation=operation), runtime_scope.bind_operation(operation=operation):
                candidate = rig.candidate()
                prepared = rig.population.prepare_vector_step(**candidate)
                result = rig.env.step(candidate["actions"], vector_step=prepared)
                assert result[3]["intrinsic_transition"].accepted is candidate["accepted"]
                assert rig.env.global_tick == 1
        assert invocation.completion_failed() is False
    assert len(reports) == 1
    assert scope._custody is scope._thread is scope._operation is None and scope._pins == ()
    assert rig.env._env_custody.reverse is rig.env._env_custody.active is rig.env._env_custody.scope is None
    print("ENV_C2_ACTUAL_FORK=" + json.dumps(reports[0], sort_keys=True))


@pytest.mark.parametrize("failure_cut", ["none", "before", "after"])
def test_callback_lifecycle_refusal_and_record_reference_expiry(compiled: Any, monkeypatch: Any, failure_cut: str) -> None:
    claim = VectorizedPopulation.claim_vector_step
    primary = RuntimeError("independent callback cut")
    records = []
    denied = []

    def observer(self: Any, **kwargs: Any) -> int:
        env = self.env
        records.append(env._env_custody.reverse)
        checks = {
            "reset": lambda: env.reset(reservation=None),
            "restore": lambda: env.install_world_restore(prepared=None, release=None),
            "activation": lambda: env.activate_episode(accepted=kwargs["accepted"], population_receipt=None, curiosity_receipt=None),
            "close": lambda: env.close_episode(None, dispositions=None),
            "step": lambda: env.step(kwargs["actions"], vector_step=kwargs["prepared"]),
        }
        for name, check in checks.items():
            with pytest.raises(ValueError, match="no permission"):
                check()
            denied.append(name)
        if failure_cut == "before":
            raise primary
        result = claim(self, **kwargs)
        if failure_cut == "after":
            raise primary
        return result

    monkeypatch.setattr(VectorizedPopulation, "claim_vector_step", observer)
    rig = make_parent(compiled, "none", "shared")
    rig.enter()
    candidate = rig.candidate()
    prepared = rig.population.prepare_vector_step(**candidate)
    if failure_cut == "none":
        rig.env.step(candidate["actions"], vector_step=prepared)
        assert rig.env.global_tick == 1
    else:
        with pytest.raises(RuntimeError) as caught:
            rig.env.step(candidate["actions"], vector_step=prepared)
        assert caught.value is primary
        assert rig.env.global_tick == 0 and not bool(rig.env.step_counts.any())
    assert len(denied) == 5 and len(records) == 1 and records[0] == {}
    assert rig.env._env_custody.reverse is rig.env._env_custody.active is None


@pytest.mark.parametrize("fault", ["layout", "device"])
@pytest.mark.parametrize("swallow", [False, True])
def test_wrong_layout_or_device_reaches_real_validator_and_stays_denied(
    compiled: Any, monkeypatch: Any, fault: str, swallow: bool
) -> None:
    """Negative real tensor conversions; meta is device refusal, not CUDA proof."""
    claim = VectorizedPopulation.claim_vector_step
    validate = VectorizedHamletEnv.validate_initial_participation
    reading = VectorizedHamletEnv._participation_reading
    state: dict[str, Any] = {"inside": False, "primary": None, "reached": [], "claims": 0}

    def observe_reading(self: Any, value: Any, **kwargs: Any) -> None:
        if state["inside"]:
            state["reached"].append((value.layout, value.device))
        return reading(self, value, **kwargs)

    def change_reading(self: Any, **kwargs: Any) -> None:
        if not state["inside"]:
            return validate(self, **kwargs)
        original = kwargs["predecessor"]
        assert original.device.type == "cpu" and original.layout == torch.strided
        changed = original.to_sparse() if fault == "layout" else original.to(device="meta")
        assert changed.shape == original.shape and changed.dtype == original.dtype
        try:
            return validate(self, **{**kwargs, "predecessor": changed})
        except BaseException as error:
            state["primary"] = error
            raise

    def observer(self: Any, **kwargs: Any) -> Any:
        state["inside"] = True
        state["claims"] += 1
        try:
            return claim(self, **kwargs)
        except ValueError:
            if not swallow:
                raise
            return None  # Negative return; never a fabricated successful tick.
        finally:
            state["inside"] = False

    monkeypatch.setattr(VectorizedPopulation, "claim_vector_step", observer)
    monkeypatch.setattr(VectorizedHamletEnv, "validate_initial_participation", change_reading)
    monkeypatch.setattr(VectorizedHamletEnv, "_participation_reading", observe_reading)
    rig = make_parent(compiled, "none", "shared")
    rig.enter()
    candidate = rig.candidate()
    prepared = rig.population.prepare_vector_step(**candidate)
    with pytest.raises(ValueError, match="invalid geometry, dtype or device") as caught:
        rig.env.step(candidate["actions"], vector_step=prepared)
    assert caught.value is state["primary"] and state["claims"] == 1
    assert len(state["reached"]) == 1
    layout, device = state["reached"][0]
    assert (layout != torch.strided) if fault == "layout" else (device.type == "meta")
    assert rig.env.global_tick == 0 and not bool(rig.env.step_counts.any())
    assert rig.env._env_custody.reverse is rig.env._env_custody.active is None


def test_real_prior_revision_reading_refuses_with_same_accepted_and_participation(compiled: Any, monkeypatch: Any) -> None:
    """A genuine POP step ages the reading; no private revision/phase is forged."""
    commit = VectorizedHamletEnv.commit_initial_participation
    committed = []

    def capture(self: Any, **kwargs: Any) -> Any:
        result = commit(self, **kwargs)
        committed.append(result)
        return result

    monkeypatch.setattr(VectorizedHamletEnv, "commit_initial_participation", capture)
    rig = make_parent(compiled, "none", "shared")
    begin(rig, retired=IDS[1:])
    accepted, participation = rig.record.accepted, committed[-1]
    old_reading = rig.candidate()["predecessor"].detach().clone()
    prior_revision = rig.env._world_revision
    rig.env.validate_initial_participation(accepted=accepted, participation=participation, predecessor=old_reading)
    result = rig.population.step_population(rig.env)
    assert result.info["intrinsic_transition"].accepted is accepted
    assert rig.env._world_revision > prior_revision
    assert rig.env.global_tick == 1 and rig.env.step_counts.tolist() == [1, 0, 0, 0, 0, 0, 0, 0]
    current_reading = rig.candidate()["predecessor"]
    assert not torch.equal(old_reading, current_reading), "fixture must establish a genuinely changed physical reading"
    rig.env.validate_initial_participation(accepted=accepted, participation=participation, predecessor=current_reading)
    with pytest.raises(ValueError, match="predecessor disagrees with current physical reading"):
        rig.env.validate_initial_participation(accepted=accepted, participation=participation, predecessor=old_reading)
    # Failed stale input does not poison a still-current genuine reading or move physics.
    rig.env.validate_initial_participation(accepted=accepted, participation=participation, predecessor=current_reading)
    assert rig.env.global_tick == 1 and rig.env._env_custody.reverse is rig.env._env_custody.active is None
