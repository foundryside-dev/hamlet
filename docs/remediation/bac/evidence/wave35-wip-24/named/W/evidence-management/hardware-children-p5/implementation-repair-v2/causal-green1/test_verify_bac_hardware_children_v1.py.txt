def seed(rule):
    """Populate all P0 fields, then fixtures override behavioral values explicitly."""
    if isinstance(rule, dict):
        return {k: seed(v) for k, v in rule.items()}
    if isinstance(rule, list):
        if rule[0] == "enum":
            return rule[1]
        if rule[0] == "list":
            return [seed(rule[1]) for _ in range(rule[2])]
        if rule[0] == "nullable":
            return None
        obj = {"state": "measured", "value": seed(rule[1]), "reason": ""}
        if rule[0] == "fact":
            obj["provenance"] = copy.deepcopy(OPAQUE)
        return obj
    if rule == "reference":
        return copy.deepcopy(OPAQUE)
    if rule in p0.TYPES:
        return seed(p0.TYPES[rule])
    return {
        "path": "/synthetic/path",
        "uint": 0,
        "positive-int": 1,
        "int": 0,
        "number": 0,
        "positive-number": 1,
        "bool": True,
        "text": "synthetic",
        "string": "",
        "sha256": "a" * 64,
        "git-id": "b" * 40,
        "id": "synthetic",
        "reference-name": "synthetic",
        "utc": STAMP,
        "env": {},
    }[rule]

                name

                    "payload": "producer-result.json",
                    "preflight": "producer-preflight.json",
                    "outcome": "producer-outcome.json",
                    "baseline": "producer-parent-baseline.json",
                }.get(name, basename)
            for trial in ACTIONS:
                if name == trial + "-terminal":
                    basename = "core-fork-result-" + trial + ".json"
                if name == trial + "-observation":
                    basename = "core-fork-observation-" + trial + ".json"
            if basename:
                path = self.job["output"] + "/" + basename
        path = path or (self.job["output"] if hasattr(self, "job") else self.base + "/meta") + "/" + name + ".json"
        ref = {"path": path, "bytes": 0, "sha256": "0" * 64}
        self.items.append((name, obj, ref))
        self.named[name] = obj
        return ref

    def fork(self, trial, index):

        end = self.process(trial + "-end", role, 110 + index, 101, self.start, launch, "end", 9100000000)
        rows = []
        for phase, pr, entry, returned in [("start", start, 3000000000, 4000000000), ("end", end, 9000000000, 10900000000)]:

            rawref = proc["debug"]["provenance"]
            raw = self.obj(rawref)
            sampler = self.add(
                trial + "-" + phase + "-sampler",
                {
                    "schema": "bac-hardware-owner-sampler-observation-v1",

                    "identity": proc["identity"],

                trial + "-" + phase + "-emission",
                {
                    "schema": "bac-hardware-owner-sampler-emission-v1",

                update = {"path": ref["path"], "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()}

            ],
        }

def pair_bundle():
    outer = Bundle(kind="phase-pair", base="/synthetic/pair")
    # Pair wrapper has explicit subjects, not adjacent receipt discovery.
    subs = []
    for label in ("reference", "failure"):
        b = Bundle("invocation_phase_" + label, base="/synthetic/" + label)
        b.named["spec"]["attempt_id"] = label
        b.named["spec"]["authorized_root"] = "/synthetic/source"
        b.named["candidate"]["authorized_root"] = "/synthetic/source"
        b.named["candidate"]["import_roots"] = ["/synthetic/source/src"]
        for _, obj, _ in b.items:
            if obj.get("attempt_id") == "a":
                obj["attempt_id"] = label
        for rr in [b.owner_source, *b.imports.values()]:
            rr["path"] = rr["path"].replace("/synthetic/" + label + "/source", "/synthetic/source")
        for rows in b.named["candidate"]["inventory"].values():
            for row in rows:
                row["path"] = row["reference"]["path"]
        subject = b.subject()
        expected = {
            "schema": "bac-hardware-strict-validation-result-v1",
            "obligation_id": "strict",
            "verdict": "satisfied",
            "reasons": [],
            "cases": [b.node],
            "evidence": [b.evidence],
            "payloads": [b.payload],
            "producer_processes": [b.parent],
            "related_attempts": [],
        }
        receipt = {
            "schema": "bac-hardware-strict-child-receipt-v1",
            "spec": b.spec,
            "job_id": b.jobid,
            "obligation_id": "strict",
            "validator": b.validator,
            "validator_dependencies": [copy.deepcopy(OPAQUE)],
            "contract": b.contract,
            "witness_schema": b.named["contract"]["witness_schema"],
            "documents": [r for _, _, r in b.items],
            "result": expected,
            "validation_identity": {"pid": 500, "ppid": 1, "start_identity": "500", "host": "synthetic", "boot_id": "synthetic-boot"},
            "validated_at": STAMP,
        }
        receipt_ref = b.add("strict-receipt", receipt)
        subs.append((b, receipt_ref))
    keep = {
        "witness",
        "contract",
        "candidate",
        "spec",
        "evidence",
        "completion",
        "result",
        "report-setup",
        "report-call",
        "report-teardown",
    }
    outer.items = [item for item in outer.items if item[0] in keep]

        "schema": "bac-hardware-core-phase-pair-v1",
        "spec": outer.spec,
        "job_id": outer.jobid,
        "obligation_id": "strict",

                "job_id": b.jobid,
                "strict_receipt": rr,

            for b, rr in subs

    outer.named["evidence"]["payload"] = wr
    outer.named["evidence"]["related_attempts"] = [b.spec for b, _ in subs]
    outer.named["completion"]["process_records"] = []
    outer.named["completion"]["source_witnesses"] = []
    outer.named["completion"]["inventory"] = [
        r for n, o, r in outer.items if n not in ("completion", "result") and r["path"].startswith(outer.job["output"] + "/")
    ]
    for k in ("process_records", "source_witnesses", "inventory"):
        outer.named["result"][k] = outer.named["completion"][k]
    for b, rr in subs:
        outer.items += b.items
    return outer, subs

def materialize(b, root):
    """Write inert nonexecuted source/data files, never import or run fake sources."""
    old = b.base

    def replace(obj):
        if isinstance(obj, dict):
            for k, v in list(obj.items()):
                if isinstance(v, str) and v.startswith(old):
                    obj[k] = str(root) + v[len(old) :]
                else:
                    replace(v)
        elif isinstance(obj, list):
            for i, v in enumerate(obj):
                if isinstance(v, str) and v.startswith(old):
                    obj[i] = str(root) + v[len(old) :]
                else:
                    replace(v)

    for _, obj, ref in b.items:
        replace(obj)
        replace(ref)
    b.job["cwd"] = str(root / "source")
    b.job["budget"] = {k: 100 for k in b.job["budget"]}
    b.named["spec"]["budget"] = dict(b.job["budget"])
    b.named["candidate"].update(authorized_root=str(root / "source"), import_roots=[str(root / "source/src")])
    selfref = p0.file_reference(Path(api().__file__).absolute())
    protocol = p0.file_reference(Path(p0.__file__).absolute())
    b.validator.clear()
    b.validator.update(selfref)
    b.named["spec"]["tools"] = [{"name": n, "reference": protocol if n == "protocol" else selfref} for n in p0.TOOLS]
    for kind in ("process-custody", "source-freeze", "required-cases"):
        b.job["obligations"].append({**b.obligation, "id": kind, "kind": kind})
    document_paths = {r["path"] for _, _, r in b.items}

    def leaves(obj):
        if isinstance(obj, dict):
            if set(obj) == {"path", "bytes", "sha256"} and obj["path"] not in document_paths and obj != selfref and obj != protocol:
                path = Path(obj["path"])
                path.parent.mkdir(parents=True, exist_ok=True)
                if not path.exists():
                    path.write_bytes(b"")
                obj.update(p0.file_reference(path))
            else:
                for v in obj.values():
                    leaves(v)
        elif isinstance(obj, list):
            for v in obj:
                leaves(v)

    for _, obj, _ in b.items:
        leaves(obj)
    subject = b.subject()
    for row in subject["documents"]:
        path = Path(row["reference"]["path"])
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(row["content"])
    return Path(b.spec["path"])


def test_standalone_exclusive_receipt_and_live_hash_validation(tmp_path):

def add_canonical_owner(b, selected_roles=None):
    canonical = copy.deepcopy(b.named["contract"])
    canonical.update(obligation_kind="strict-child-core", roles=b.roles)
    cr = b.add("canonical-contract", canonical)
    co = {**b.obligation, "id": "owner", "kind": "strict-child-core", "contract": cr}
    b.job["obligations"].append(co)
    if selected_roles is not None:
        b.named["contract"]["roles"] = selected_roles
    for _, obj, _ in b.items:
        if obj.get("schema") == "bac-hardware-owner-launch-v1":
            obj["owner_contract"] = cr
    ce = {**b.named["evidence"], "obligation_id": "owner", "obligation_kind": "strict-child-core", "contract": cr}
    er = b.add("canonical-evidence", ce)
    b.named["completion"]["obligation_records"].append(er)
    b.named["completion"]["inventory"].extend([cr, er])
    return cr


@pytest.mark.parametrize("kind", ["rng-custody", "fork-three-trials"])

    add_canonical_owner(b, subset)
