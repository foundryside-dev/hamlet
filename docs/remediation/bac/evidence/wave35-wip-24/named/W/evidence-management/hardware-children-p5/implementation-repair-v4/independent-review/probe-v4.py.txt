def sha(data):
    return hashlib.sha256(data).hexdigest()


def pin_sources(folder, when):
    rows = []
    for name, expected in PINS.items():
        data = (Q / name).read_bytes()
        (folder / (when + '-' + name + '.txt')).write_bytes(data)
        rows.append({'path': str(Q / name), 'sha256': sha(data), 'bytes': len(data), 'lines': len(data.splitlines())})
        assert sha(data) == expected, (name, when, sha(data))
    return rows


def save_subject(folder, subject):
    folder.mkdir()
    manifest = {k: v for k, v in subject.items() if k != 'documents'}
    manifest['documents'] = []
    for index, row in enumerate(subject['documents']):
        leaf = f'document-{index:03d}.json'
        (folder / leaf).write_bytes(row['content'])
        manifest['documents'].append({'reference': row['reference'], 'retained_content': leaf})
    (folder / 'subject.json').write_text(json.dumps(manifest, indent=2) + '\n')

                result = verifier.validate_child_obligation(subject)
            record.update(observed=result['verdict'], result=result)
        except ValueError as exc:
            record.update(observed='refused', error=str(exc))
        record['matched_expectation'] = record['observed'] == record['expected']
        record['receipt_exists'] = (case_dir / 'receipt.json').exists()

    pair, _ = fixture.pair_bundle()

def main():
    mode, attempt = sys.argv[1:]
    folder = HERE / attempt
    folder.mkdir(exist_ok=False)
    os.sched_setaffinity(0, {min(os.sched_getaffinity(0))})
    resource.setrlimit(resource.RLIMIT_AS, (4 * 1024**3, 4 * 1024**3))
    for key in ('OMP_NUM_THREADS', 'MKL_NUM_THREADS', 'OPENBLAS_NUM_THREADS'):
        os.environ[key] = '1'
    os.environ['PYTEST_DISABLE_PLUGIN_AUTOLOAD'] = '1'
    sys.dont_write_bytecode = True
    sys.path.insert(0, str(Q))
    record = {
        'schema': 'bac-p5-independent-probe-execution-v1', 'mode': mode, 'attempt': attempt,
        'argv': sys.argv, 'actual_argv': [x.decode() for x in Path('/proc/self/cmdline').read_bytes().split(b'\0') if x],
        'interpreter': sys.executable, 'cwd': os.getcwd(), 'pid': os.getpid(), 'ppid': os.getppid(),
        'start_identity': Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()[19],
        'host': os.uname().nodename, 'boot_id': Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
        'affinity': sorted(os.sched_getaffinity(0)), 'ram_limit': 4 * 1024**3,
        'start': datetime.datetime.now(datetime.timezone.utc).isoformat(),
        'probe': {'path': str(Path(__file__)), 'sha256': sha(Path(__file__).read_bytes())},
    }
    before = time.monotonic_ns()
    record['source_before'] = pin_sources(folder, 'before')
    code = 2
    with (folder / 'stdout.log').open('x') as out, (folder / 'stderr.log').open('x') as err:
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            try:
                if mode == 'probe':
                    code = run_probes(folder)
                elif mode == 'suite':
                    import pytest
                    args = ['-q', '--noconftest', '-c', '/dev/null', '--basetemp=' + str(folder / 'pytest'), '-p', 'no:cacheprovider', str(Q / 'test_verify_bac_hardware_children_v1.py')]
                    record['pytest_argv'] = args
                    code = int(pytest.main(args))
                else:
                    raise ValueError('unknown mode')
            except BaseException:
                traceback.print_exc()
                code = 2
    record.update(exit=code, duration_ns=time.monotonic_ns() - before, end=datetime.datetime.now(datetime.timezone.utc).isoformat())
    record['source_after'] = pin_sources(folder, 'after')
    record['unchanged'] = record['source_before'] == record['source_after']
    inventory = []
    for path in sorted(folder.rglob('*')):
        if path.is_file() and not path.is_symlink():
            data = path.read_bytes()
            inventory.append({'path': str(path), 'bytes': len(data), 'sha256': sha(data)})
    (folder / 'retained-inventory.json').write_text(json.dumps(inventory, indent=2) + '\n')
    (folder / 'execution.json').write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps(record))
    return code


if __name__ == '__main__':
    raise SystemExit(main())
