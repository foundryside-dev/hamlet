import json

import test_run_bac_hardware_v1 as fixtures

NEW_FIELDS = ['python_debug', 'python_optimize', 'torch_anomaly', 'source_method', 'recorder_mode',
              'module_names_before_sha256', 'module_names_after_sha256', 'owner_claim', 'sampler']

@pytest.mark.parametrize('key', NEW_FIELDS)
def test_independent_task1_every_current_field_required(tmp_path, key):
    r, c, job, rec, role, raw, env, put, seal = fixtures.owner_transport_fixture(tmp_path)
    del raw[key]
    seal()
    with pytest.raises(ValueError, match='closed private'):
        c.process_context(job, rec, role, env)

@pytest.mark.parametrize('key,value', [('python_debug', 1), ('python_debug', 'true'),
    ('python_optimize', 0.0), ('torch_loaded', 0), ('cuda_marker', 0),
    ('owner_claim', {}), ('sampler', None), ('module_names_after_sha256', 'A'*64),
    ('source_method', ['direct-git']), ('recorder_mode', ['ordinary'])])
def test_independent_task1_exact_types(tmp_path, key, value):
    r, c, job, rec, role, raw, env, put, seal = fixtures.owner_transport_fixture(tmp_path)
    raw[key] = value
    seal()
    with pytest.raises(ValueError): c.process_context(job, rec, role, env)

@pytest.mark.parametrize('value,reason,accept', [(False, '', True), (True, '', True),
    (0, '', False), (1, '', False), (None, '', False), ('false', '', False),
    (False, 'loaded Torch initialization state only, not independent proof of native device work', False)])
def test_independent_task1_work_anomaly_exact_boolean(tmp_path, value, reason, accept):
    r, c, job, rec, role, raw, env, put, seal = fixtures.owner_transport_fixture(tmp_path)
    role.update(observer='work', torch_policy='work')
    raw.update(torch_loaded=True, cuda_marker=False,
        observation_limit='loaded Torch initialization state only, not independent proof of native device work',
        torch_anomaly={'state':'measured','value':value,'reason':reason})
    rec['torch_observation'] = 'work'
    for key in ('intra_op','inter_op'): rec[key] = {'state':'measured','value':1,'reason':''}
    seal()
    r.validate_record('process', rec)
    if accept: c.process_context(job, rec, role, env)
    else:
        with pytest.raises(ValueError, match='work anomaly'): c.process_context(job, rec, role, env)

@pytest.mark.parametrize('mutation', ['reason', 'forbidden-loaded', 'cuda-no-initialize'])
def test_independent_task1_observer_policy(tmp_path, mutation):
    r, c, job, rec, role, raw, env, put, seal = fixtures.owner_transport_fixture(tmp_path)
    limit = 'loaded Torch initialization marker only, not every native CUDA context; runtime calls prohibited'
    raw.update(torch_loaded=True,cuda_marker=False,observation_limit=limit,
        torch_anomaly={'state':'unavailable','value':None,'reason':limit})
    role['torch_policy']='observe-loaded'; rec['torch_observation']='loaded-without-calls'
    for key in ('intra_op','inter_op'): rec[key]['reason']=limit
    if mutation=='reason': raw['torch_anomaly']['reason']='arbitrary no-call assertion'
    elif mutation=='forbidden-loaded': role['torch_policy']='forbidden'
    else: raw['cuda_marker']=True; rec['cuda_observation']='initialized'
    seal()
    with pytest.raises(ValueError): c.process_context(job, rec, role, env)

# Task2 additions to the same fifth logical review body. Prior exact bytes retained inertly.
@pytest.mark.parametrize('authority', ['frozen', 'unfrozen', 'outside-root'])
def test_independent_task2_actual_child_import_authority(tmp_path, monkeypatch, authority):
    captured = []
    original = fixtures.spawn_owner_transport_fixture
    def capture(*args, **kwargs):
        value = original(*args, **kwargs); captured.append(value); return value
    monkeypatch.setattr(fixtures, 'spawn_owner_transport_fixture', capture)
    fixtures.test_owner_spawn_completion_requires_one_launch_and_claim_across_phases(tmp_path, False)
    r,c,job,rec,role,raw,env,put,seal,launch,claim = captured[0]
    allowed = c.candidate['inventory']['source'][0]['reference']
    if authority == 'unfrozen':
        extra = Path(job['cwd']) / 'unfrozen-child-module.txt'
        extra.write_text('inert bytes, never imported')
        allowed = r.file_reference(extra)
    elif authority == 'outside-root': allowed = c.inputs[1]
    def mutation(value):
        if value.get('kind') == 'process' and value.get('role') == 'spawn-child':
            value['imports'] = [{'name': '_observed_child_extra', 'reference': allowed}]
        return value
    path = Path(job['output']) / 'completion.json'
    completion = fixtures.reseal_owner_fixture(json.loads(path.read_text()), Path(job['output']), mutation)
    path.write_text(json.dumps(completion, sort_keys=True))
    if authority == 'frozen': assert c.evidence(job) == completion
    else:
        with pytest.raises(ValueError): c.evidence(job)


@pytest.mark.parametrize('change', ['parent-source-identity','launch-source-identity','child-source-identity','parent-argv'])
def test_independent_task2_resealed_ancestry_sources(tmp_path, change):
    r,c,job,rec,role,raw,env,put,seal,launch,claim = fixtures.spawn_owner_transport_fixture(tmp_path)
    reference = seal()
    parent = json.loads(Path(launch['parent_process']['path']).read_text())
    chosen = {'parent-source-identity':parent['source_witness']['path'],
              'launch-source-identity':launch['source_witness']['path'],
              'child-source-identity':rec['source_witness']['path']}
    if change in chosen:
        path=Path(chosen[change]); body=json.loads(path.read_text())
        body['process_identity']={**body['process_identity'], 'start_identity':'foreign'}
        path.write_text(json.dumps(body))
    else:
        path=Path(launch['parent_process']['path']);body=json.loads(path.read_text())
        body['argv']=['/foreign'];path.write_text(json.dumps(body))
    env=fixtures.reseal_owner_fixture(env,Path(job['output']),lambda value:value)
    for obligation in job['obligations']:
        for key in ('contract','validator'):
            if Path(obligation[key]['path']).parent==Path(job['output']):
                obligation[key]=r.file_reference(Path(obligation[key]['path']))
    rec=json.loads(Path(reference['path']).read_text())
    with pytest.raises(ValueError):c.process_context(job,rec,role,env)
