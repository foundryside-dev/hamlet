"""Root materialization of reviewed WIP24 metadata; no commit or network action."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import subprocess

R = Path('/workspace/hamlet')
W = R / '.superpowers/sdd/2026-11-01-bac-remediation-upgrades'
E = Path(__file__).parent
archive = R / 'docs/remediation/bac/evidence/wave35-wip-24'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
git = lambda *args: subprocess.check_output(['git', *args], cwd=R).decode().strip()
now = lambda: datetime.now(timezone.utc).isoformat()

def read(path):
    return json.loads(path.read_bytes())

def ref(path):
    return dict(path=str(path.relative_to(R)), bytes=path.stat().st_size, sha256=sha(path))

def write(path, data):
    with path.open('x') as stream:
        json.dump(data, stream, indent=2, sort_keys=True)
        stream.write('\n')
    assert path.stat().st_size < 100 * 1024**2, path

def copy(source, destination):
    with destination.open('xb') as stream:
        stream.write(source.read_bytes())
    assert sha(source) == sha(destination)
    assert destination.stat().st_size < 100 * 1024**2

m = read(manifest_path)
