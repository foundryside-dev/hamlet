def module():

def assess(tmp_path, obs=None, req=None):
    obs = observation() if obs is None else obs
    ref = reference(tmp_path / "discovery.json", json.dumps(obs))
    return module().assess_resources(request() if req is None else req, obs, ref)

def gpu_probe(tmp_path, monkeypatch, change):
    from types import SimpleNamespace
    tool = module()
    assert callable(getattr(tool, "collect_gpu", None)), "pinned CUDA backend missing"
    driver = reference(tmp_path / "nvidia-smi", "inert driver executable")
    torch = reference(tmp_path / "torch.py", "inert Torch module")
    env = {"dependencies": [{"name": "nvidia-smi", "version": "580.1", "reference": driver},
                            {"name": "torch", "version": "2.9+cu128", "reference": torch}]}
    obs = observation(); obs.update(profile="cuda", effective_visibility="1,0")
    responses = {"--version": "NVIDIA-SMI version  : 580.1\n",
                 "--query-gpu=index,uuid,name,memory.total,driver_version,mig.mode.current":
                 "0, GPU-a, Simulated A, 2000, 580.1, Disabled\n1, GPU-b, Simulated B, 1000, 580.1, Disabled\n",
                 "--query-compute-apps=gpu_uuid,pid,process_name": "", "topo": "synthetic topology"}
    # Actual logical UUIDs intentionally do NOT follow guessed numeric driver indices.
    props = [SimpleNamespace(uuid="GPU-a", name="Simulated A", total_memory=1990*1024**2),
             SimpleNamespace(uuid="GPU-b", name="Simulated B", total_memory=990*1024**2)]
    fake = SimpleNamespace(__file__=torch["path"], __version__="2.9+cu128", version=SimpleNamespace(cuda="12.8"),
                           cuda=SimpleNamespace(is_available=lambda: True, device_count=lambda: len(props),
                                                get_device_properties=lambda i: props[i]))
    if change == "unrelated":
        responses["--query-gpu=index,uuid,name,memory.total,driver_version,mig.mode.current"] += "2, GPU-c, Simulated C, 1000, 580.1, Disabled\n"
        responses["--query-compute-apps=gpu_uuid,pid,process_name"] = "GPU-c, 999, unrelated-work\n"
    if change == "mig": responses[next(k for k in responses if k.startswith("--query-gpu"))] = responses[next(k for k in responses if k.startswith("--query-gpu"))].replace("Disabled", "Enabled")
    if change == "unknown": props[0].uuid = "GPU-unknown"
    if change == "duplicate": props[1].uuid = "GPU-a"
    if change == "cpu_build": fake.version.cuda = None
    if change == "torch_origin": fake.__file__ = str(tmp_path / "wrong.py")
    if change == "visibility": obs["effective_visibility"] = "GPU-b,GPU-a"
    if change == "mixed": obs["effective_visibility"] = "0,GPU-a"
    if change == "vram": props[0].total_memory = 3000*1024**2
    calls = []
    def query(executable, tail, commands):
        assert executable == driver
        calls.append(tail)
        return responses[tail[0]]
    monkeypatch.setattr(tool, "run_query", query)
    monkeypatch.setattr(tool, "load_torch", lambda reference: fake)
    return tool, obs, env, calls


def test_real_cuda_backend_constructs_pinned_queries_and_joins_actual_uuid(tmp_path, monkeypatch):
    tool, obs, env, calls = gpu_probe(tmp_path, monkeypatch, "none")
    tool.collect_gpu(obs, env, lambda: None)

    assert obs["driver"] == "580.1" and obs["cuda_runtime"] == "12.8"

                     ["--query-compute-apps=gpu_uuid,pid,process_name", "--format=csv,noheader,nounits"], ["topo", "-m"]]
    assert "torch" not in sys.modules

def test_driver_query_retains_executable_actual_child_and_exact_argv(tmp_path, monkeypatch):
    from types import SimpleNamespace
    tool = module()
    assert callable(getattr(tool, "run_query", None)), "pinned query subprocess missing"
    executable = reference(tmp_path / "driver", "inert executable")
    calls = []
    class Child:
        pid = 4321
        returncode = 0
        def communicate(self, timeout): return "result", "diagnostic"
    def popen(argv, **kwargs):
        calls.append((argv, kwargs)); return Child()
    monkeypatch.setattr(tool.subprocess, "Popen", popen)
    identity = dict(observation()["identity"], ppid=tool.os.getpid())
    monkeypatch.setattr(tool, "process_identity", lambda pid: identity)
    monkeypatch.setattr(tool.os, "readlink", lambda path: executable["path"])
    records = []
    token = tool.LEASE_FDS.set((101, 102))  # Inert descriptors; Popen is stubbed.
    try: assert tool.run_query(executable, ["--version"], records) == "result"
    finally: tool.LEASE_FDS.reset(token)
    assert calls[0][1]["pass_fds"] == (101, 102)
    assert calls[0][0] == [executable["path"], "--version"] and not calls[0][1].get("shell", False)
    assert records[0]["identity"] == identity and records[0]["executable"] == executable
    assert records[0]["stdout"] == "result" and records[0]["stderr"] == "diagnostic"

def native_git_stub(tmp_path, monkeypatch, change="none"):
    tool = module(); executable = reference(tmp_path / "git", "inert native git bytes")
    dependency = {"name": "git", "version": "git version 2.50.0", "reference": executable}
    calls, events = [], []
    class Child:
        pid = 4321
        returncode = 7 if change == "exit" else 0
        def communicate(self, timeout):
            events.append("wait")
            if change in ("timeout", "cancel") and len(events) == 1:
                if change == "cancel": raise KeyboardInterrupt()
                raise tool.subprocess.TimeoutExpired("stub Git", timeout)
            if change == "mutated": Path(executable["path"]).write_text("different bytes")
            text = {"multiline": "git version 2.50.0\nextra\n", "empty": "", "not_git": "2.50.0\n",
                    "mismatch": "git version 2.51.0\n", "crlf": "git version 2.50.0\r\n"}.get(change, "git version 2.50.0\n")
            return text.encode(), b"stub diagnostic"
        def kill(self): events.append("kill"); self.returncode = -9
    def popen(argv, **kwargs): calls.append((argv, kwargs)); return Child()
    monkeypatch.setattr(tool.subprocess, "Popen", popen)
    identity = dict(observation()["identity"], ppid=tool.os.getpid(), pid=9999 if change == "pid" else 4321)
    monkeypatch.setattr(tool, "process_identity", lambda pid: identity)
    monkeypatch.setattr(tool.os, "readlink", lambda path: "/wrong/git" if change == "executable" else executable["path"])
    monkeypatch.setattr(tool.metadata, "distribution", lambda name: pytest.fail("native Git was treated as a Python distribution"))
    monkeypatch.setattr(tool, "load_torch", lambda ref: pytest.fail("CPU Git dependency imported Torch"))
    monkeypatch.setattr(tool.shutil, "which", lambda name: pytest.fail("declared Git was replaced by PATH lookup"))
    return tool, dependency, calls, events


def test_native_git_dependency_observes_actual_version_and_pinned_process(tmp_path, monkeypatch):
    tool, dependency, calls, events = native_git_stub(tmp_path, monkeypatch)
    records = []; token = tool.LEASE_FDS.set((101, 102))
    try: observed = tool.dependency_observations({"dependencies": [dependency]}, records)
    finally: tool.LEASE_FDS.reset(token)
    assert observed == [dependency] and len(records) == 1

    assert record["schema"] == "bac-hardware-git-command-v1"
