@pytest.mark.parametrize('output', [b'git version 2.50.0', b'git version 2.50.0\n'])
def test_exact_version_forms(tmp_path, monkeypatch, output):
    tool, dep, calls, events = fixtures.native_git_stub(tmp_path, monkeypatch)
    original = tool.subprocess.Popen
    def launch(*args, **kwargs):
        child = original(*args, **kwargs)
        child.communicate = lambda timeout: (output, b'\xff\r\n')
        return child
    monkeypatch.setattr(tool.subprocess, 'Popen', launch)
    records = []
    assert tool.dependency_observations({'dependencies': [dep]}, records) == [dep]
    assert records[0]['stderr'].encode('utf-8', 'surrogateescape') == b'\xff\r\n'


@pytest.mark.parametrize('stage', ['spawn', 'postref', 'ppid'])
def test_additional_refusals(tmp_path, monkeypatch, stage):
    tool, dep, calls, events = fixtures.native_git_stub(tmp_path, monkeypatch)
    if stage == 'spawn':
        def fail(*args, **kwargs): raise PermissionError('inert nonexecutable')
        monkeypatch.setattr(tool.subprocess, 'Popen', fail)
    elif stage == 'postref':
        original = tool.subprocess.Popen
        def launch(*args, **kwargs):
            child = original(*args, **kwargs)
            def communicate(timeout):
                Path(dep['reference']['path']).write_text('mutated independent fixture')
                return b'git version 2.50.0\n', b''
            child.communicate = communicate
            return child
        monkeypatch.setattr(tool.subprocess, 'Popen', launch)
    else:
        monkeypatch.setattr(tool, 'process_identity', lambda pid: dict(fixtures.observation()['identity'], pid=pid, ppid=0))
    records = []
    with pytest.raises(ValueError): tool.dependency_observations({'dependencies': [dep]}, records)
    assert len(records) == 1 and records[0]['errors']


@pytest.mark.parametrize('secondary', ['timeout', 'kill_error'])
def test_cleanup_failure_retains_native_command(tmp_path, monkeypatch, secondary):
    tool, dep, calls, events = fixtures.native_git_stub(tmp_path, monkeypatch)
    class Child:
        pid = 4321
        returncode = None
        def communicate(self, timeout):
            raise tool.subprocess.TimeoutExpired('inert stuck pipe', timeout, output=b'partial', stderr=b'partial-error')
        def kill(self):
            if secondary == 'kill_error': raise ProcessLookupError('inert exit race')
            self.returncode = -9
    monkeypatch.setattr(tool.subprocess, 'Popen', lambda *args, **kwargs: Child())
    records = []
    try: tool.dependency_observations({'dependencies': [dep]}, records)
    except (ValueError, OSError, tool.subprocess.SubprocessError): pass
    assert len(records) == 1, 'cleanup failure lost the launched native command and all diagnostic bytes'
    assert records[0]['errors']
