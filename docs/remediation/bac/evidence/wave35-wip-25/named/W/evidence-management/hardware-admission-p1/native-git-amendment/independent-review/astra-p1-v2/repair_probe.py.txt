import test_capture_bac_hardware_v1 as f


@pytest.mark.parametrize('completion', ['growth', 'absent', 'inconsistent'])
def test_independent_partial_bytes_and_terminal_refusal(tmp_path, monkeypatch, completion):
    candidate, request_path, source, req, env, registry = f.capture_fixture(tmp_path)
    executable = env['dependencies'][0]['reference']
    monkeypatch.setattr(tool.os, 'sched_getaffinity', lambda pid: {0, 1, 2, 3})
    monkeypatch.setattr(tool, 'process_identity', lambda pid: dict(f.observation()['identity'], pid=pid, ppid=tool.os.getpid()))
    monkeypatch.setattr(tool.os, 'readlink', lambda path: executable['path'])
    monkeypatch.setattr(tool, 'launch_discovery', lambda *args: pytest.fail('source failure must stop before discovery'))
    events = []
    class Child:
        pid = 4321
        returncode = None
        def communicate(self, timeout):
            events.append('wait')
            if len(events) == 1:
                raise tool.subprocess.TimeoutExpired('inert', timeout, output=b'A\xff', stderr=b'err\r\n')
            if completion == 'absent':
                raise tool.subprocess.TimeoutExpired('inert', timeout, output=None, stderr=None)
            return (b'A\xffB' if completion == 'growth' else b'OTHER'), b'err\r\nmore'
        def kill(self):
            events.append('kill')
            self.returncode = -9
    monkeypatch.setattr(tool.subprocess, 'Popen', lambda *args, **kwargs: Child())
    result = tool.capture(candidate, request_path, tmp_path/'capture')
    assert result['disposition'] == 'refused' and result['spec'] is None and result['reasons']
    assert json.loads((tmp_path/'capture/result.json').read_text()) == result
    assert not (tmp_path/'capture/spec.json').exists()
    records = json.loads((tmp_path/'capture/source-before-git.json').read_text())
    assert len(records) == 1 and events == ['wait', 'kill', 'wait']
    command = records[0]
    assert command['executable_after'] == executable and command['exit_code'] == -9
    assert command['errors'][0] == 'Git query interrupted: TimeoutExpired'
    assert command['stdout'].encode('utf-8', 'surrogateescape') == (b'A\xffB' if completion == 'growth' else b'A\xff')
    assert command['stderr'] == ('err\r\n' if completion == 'absent' else 'err\r\nmore')
    if completion == 'inconsistent': assert any(b'OTHER'.hex() in e for e in command['errors'])
    if completion == 'absent': assert any('reap/EOF not established' in e for e in command['errors'])

    return [{'path': str(p), 'bytes': p.stat().st_size, 'lines': len(p.read_bytes().splitlines()), 'sha256': hashlib.sha256(p.read_bytes()).hexdigest()} for p in paths]
