@pytest.mark.parametrize('mutation', ['valid', 'pre-mutation', 'post-mutation', 'failed-query'])
def test_current_p1_source_git_binding(tmp_path, monkeypatch, mutation):
    from types import SimpleNamespace
    import bac_hardware_protocol_v1 as p
    r = implementation(); calls = []
    git = tmp_path / 'chosen-git'; git.write_text('synthetic sealed executable')
    git_ref = p.file_reference(git)
    def command(argv, **kwargs):
        calls.append((argv, kwargs))
        assert argv == [str(git), '--no-optional-locks', '-c', 'core.fsmonitor=false', '-c', 'core.pager=cat', '-C', str(tmp_path), 'rev-parse', 'HEAD']
        assert kwargs['cwd'] == str(tmp_path) or kwargs['cwd'] == tmp_path
        assert kwargs['env'] == GIT_ENV
        if mutation == 'post-mutation': git.write_text('changed after query')
        return SimpleNamespace(returncode=1 if mutation == 'failed-query' else 0, stdout='observed-head\n', stderr='')
    monkeypatch.setattr(r.subprocess, 'run', command)
    if mutation == 'pre-mutation': git.write_text('changed before query')
    if mutation == 'valid': assert r._git(str(tmp_path), ['rev-parse', 'HEAD'], git_ref=git_ref) == 'observed-head\n'
    else:
        with pytest.raises(ValueError): r._git(str(tmp_path), ['rev-parse', 'HEAD'], git_ref=git_ref)
    assert len(calls) == (0 if mutation == 'pre-mutation' else 1)

def job(tmp_path, name, *, cpu=1, devices=(), domains=(), timing='ordinary', code=None, timeout=4, grace=.05):
    return {'id': name, 'argv': [sys.executable, '-c', code or 'print("observed", flush=True)'],
            'interpreter': sys.executable, 'cwd': str(tmp_path),
            'environment': {'OMP_NUM_THREADS': '1', 'MKL_NUM_THREADS': '1', 'OPENBLAS_NUM_THREADS': '1',
                            'CUDA_VISIBLE_DEVICES': '', 'PYTHONHASHSEED': '17'},
            'seed': {'policy': 'explicit', 'python': 17, 'numpy': 18, 'torch': 19, 'cuda': []},
            'roles': [{'count': 1, 'replacement_limit': 0, 'intra_op': 1, 'inter_op': 1, 'peak_cpu_slots': 1}],
            'budget': {'cpu_slots': cpu, 'ram_bytes': 2, 'scratch_bytes': 2, 'scratch_inodes': 2},
            'devices': [{'physical_parent_uuid': d} for d in devices], 'serialization_domains': list(domains),
            'timing': timing, 'output': str(tmp_path / 'out' / 'attempt' / name),
            'scratch': str(tmp_path / 'scratch' / 'attempt' / name),
            'timeout_seconds': timeout, 'termination_grace_seconds': grace}


def spec(tmp_path, jobs, *, cpu=4, concurrency=4):
    (tmp_path / 'out').mkdir(exist_ok=True)
    (tmp_path / 'scratch').mkdir(exist_ok=True)
    (tmp_path / 'leases').mkdir(exist_ok=True)
    return {'attempt_id': 'attempt', 'output_root': str(tmp_path / 'out'), 'scratch_root': str(tmp_path / 'scratch'),
            'lease_root': str(tmp_path / 'leases'), 'job_concurrency': concurrency,
            'budget': {'cpu_slots': cpu, 'ram_bytes': 100, 'scratch_bytes': 100, 'scratch_inodes': 100}, 'jobs': jobs}

def test_exclusive_attempt_race(tmp_path):
    r = implementation()
    s = spec(tmp_path, [job(tmp_path, 'a')])
    barrier = threading.Barrier(2)
    results = []
    def reserve():
        barrier.wait()
        try:
            a = r.Attempt(s)
            results.append(a)
        except (ValueError, FileExistsError):
            results.append(None)
    threads = [threading.Thread(target=reserve) for _ in range(2)]
    for t in threads: t.start()
    for t in threads: t.join()
    assert sum(v is not None for v in results) == 1
    for value in results:
        if value is not None: value.close()

    code = ('import json,sys; from pathlib import Path; '
            'sys.path.insert(0,sys.argv[1]); import run_bac_hardware_v1 as r; '
            'results=r.execute_jobs(json.loads(Path(sys.argv[2]).read_text())); '
            'sys.exit(1 if any(x["disposition"]!="successful" for x in results) else 0)')
    child = subprocess.Popen([sys.executable, '-c', code, str(Q), str(source)],

def test_process_group_cleanup_precedes_pid_reaping(tmp_path, monkeypatch):
    r = implementation(); events = []
    actual_poll, actual_wait, actual_killpg = subprocess.Popen._internal_poll, subprocess.Popen.wait, os.killpg
    def poll(process, *args, **kwargs):
        result = actual_poll(process, *args, **kwargs)
        if result is not None: events.append('reaped')
        return result
    def wait(process, *args, **kwargs):
        result = actual_wait(process, *args, **kwargs); events.append('reaped'); return result
    def killpg(pid, signum):
        if signum == signal.SIGTERM: events.append('cleanup')
        return actual_killpg(pid, signum)
    monkeypatch.setattr(r.subprocess.Popen, '_internal_poll', poll)
    monkeypatch.setattr(r.subprocess.Popen, 'wait', wait)
    monkeypatch.setattr(r.os, 'killpg', killpg)
    result = r.execute_jobs(spec(tmp_path, [job(tmp_path, 'short')]))[0]
    assert result['exit_code'] == 0
    assert events.index('cleanup') < events.index('reaped'), 'PID was released before process-group ownership cleanup'

def reseal_owner_fixture(envelope, output, change):
    """Reseal authored acyclic local data so semantic negatives reach consumers."""
    import bac_hardware_protocol_v1 as p
    cache = {}; active = set()
    def visit(value):
        if type(value) is list: return [visit(item) for item in value]
        if type(value) is not dict: return value
        if set(value) == {'path', 'bytes', 'sha256'}:
            path = Path(value['path'])
            if path.parent != output: return value
            if str(path) in cache: return cache[str(path)]
            assert str(path) not in active, 'fixture author introduced a reference cycle'
            active.add(str(path))
            body = visit(json.loads(path.read_text()))
            path.write_text(json.dumps(body, sort_keys=True))
            result = p.file_reference(path); cache[str(path)] = result; active.remove(str(path))
            return result
        return change({key: visit(item) for key, item in value.items()})
    return visit(envelope)

    'first-git-is-guard', 'second-git-is-guard'])
def test_owner_sampler_review_correspondence_gaps(tmp_path, mutation):
    def alter(parts):
        for part in parts:
            rows = part['sampler']['commands']
            if mutation == 'sequential-pid-reuse': rows[1]['pid'] = rows[0]['pid']
            elif mutation in ('first-git-is-guard', 'second-git-is-guard'):
                rows[0 if mutation == 'first-git-is-guard' else 1]['pid'] = part['record']['identity']['pid']
    r, custody, selected, completion, put, parts = owner_sampler_fixture(tmp_path, alter)
    if mutation.startswith('diagnostic-'):
        name = 'sampler-spawn-end-0.json'; diagnostic = json.loads(Path(selected['output'], name).read_text())
        if mutation == 'diagnostic-exit-bool': diagnostic['exit_code'] = False
        elif mutation == 'diagnostic-truncation-int': diagnostic['stdout_truncated'] = 0
        else: diagnostic['stdout']['bytes'] = float(diagnostic['stdout']['bytes'])
        changed = put(name, diagnostic)
        completion['inventory'] = [changed if ref['path'] == changed['path'] else ref for ref in completion['inventory']]
        put('completion.json', completion)
    elif mutation in ('wait-inclusive', 'sampler-after-wait', 'emission-after-wait'):
        last = parts[1]
        endpoint = last['emission']['emission_end_ns']
        if mutation == 'sampler-after-wait': endpoint = last['sampler']['end_ns'] - 1
        elif mutation == 'emission-after-wait': endpoint -= 1
        custody.observations[selected['id']]['monotonic_end_ns'] = endpoint
    if mutation in ('valid', 'wait-inclusive', 'sequential-pid-reuse'):
        assert custody.evidence(selected) == completion
    else:
        with pytest.raises(ValueError): custody.evidence(selected)


# Actual Task4 cases are deliberately gated; no source/Git substitute is allowed.
def _task4_stat(pid):
    fields = Path(f'/proc/{pid}/stat').read_text().rsplit(')', 1)[1].split()
    return dict(pid=pid, state=fields[0], ppid=int(fields[1]), pgid=int(fields[2]),
                sid=int(fields[3]), start_identity=fields[19])


def _task4_live(pid):
    import hashlib
    before = _task4_stat(pid)
    assert before['state'] not in ('Z', 'X'), 'actual process is not live'
    executable = Path(f'/proc/{pid}/exe')
    data = executable.read_bytes()
    result = {**before, 'exe': os.readlink(executable), 'exe_sha256': hashlib.sha256(data).hexdigest(),
              'exe_bytes': len(data), 'argv': [x.decode() for x in Path(f'/proc/{pid}/cmdline').read_bytes().split(b'\0') if x]}
    assert _task4_stat(pid)['start_identity'] == before['start_identity'], 'process identity changed'
    return result


def _task4_write(path, value):
    path = Path(path); temporary = path.with_name(path.name + '.pending')
    with temporary.open('x') as stream:
        json.dump(value, stream, sort_keys=True); stream.flush(); os.fsync(stream.fileno())
    assert not path.exists(), 'Task4 evidence cannot be overwritten'
    temporary.rename(path)


def _task4_wait(predicate, seconds):
    end = time.monotonic() + seconds
    while time.monotonic() < end:
        value = predicate()
        if value: return value
        time.sleep(.005)
    raise AssertionError('bounded Task4 observation did not arrive')


def _task4_driver(mode):
    # Imports and primitive preparation precede the actual P3 handle snapshot.
    import select
    import bac_hardware_identity_v1 as producer
    import bac_hardware_protocol_v1 as protocol
    snapshot = producer.current_process_context()
    assert snapshot is not None and set(snapshot) == {'context', 'start'}
    context = snapshot['context']; spec = json.loads(Path(context['spec']['path']).read_text())
    job = next(item for item in spec['jobs'] if item['id'] == context['job_id'])
    output = Path(job['output']); case = f'test_run_bac_hardware_v1.py::test_owner_actual_git_backstop[{mode}]'
    assert job['required_collected_ids'] == [case] and os.environ['PYTEST_CURRENT_TEST'] == case + ' (call)'
    guard_role = next(role for role in job['roles'] if role['role'] == 'fork-guard')
    obligation = next(item for item in job['obligations'] if item['kind'] == 'fork-three-trials')
    environment = json.loads(Path(spec['dependencies']['path']).read_text())
    git_ref = next(item['reference'] for item in environment['dependencies'] if item['name'] == 'git')
    protocol.validate_record('reference', git_ref)
    expected_argv = [git_ref['path'], '-C', spec['authorized_root'], 'rev-parse', 'HEAD']
    read_fd, write_fd = os.pipe(); real_spawn = os.posix_spawn; armed = [False]
    # Preload the live observer's hashlib before prepare; no imports in guard.
    controller = _task4_live(os.getpid()); signal.signal(signal.SIGTERM, signal.SIG_DFL)
    def spawn_observer(executable, argv, env, **kwargs):
        if not armed[0]: return real_spawn(executable, argv, env, **kwargs)
        assert executable == git_ref['path'] and list(argv) == expected_argv
        pid = real_spawn(executable, argv, env, **kwargs)
        os.kill(pid, signal.SIGSTOP)
        os.write(write_fd, str(pid).encode('ascii') + b'\n')
        while True: signal.pause()  # Deliberately killed before returning to P3.
    os.posix_spawn = spawn_observer
    prepared = producer.prepare_owner_launch(context, snapshot['start'], obligation['contract'], case,
        guard_role['id'], 'task4-' + mode, 'fork')
    guard_pid = os.fork()
    if guard_pid == 0:
        os.close(read_fd); armed[0] = True
        try: producer.record_fork_process(prepared['handle'], 'start')
        except BaseException: os._exit(82)
        os._exit(83)  # A returned recorder did not exercise this backstop.
    os.close(write_fd)
    assert select.select([read_fd], [], [], 5)[0], 'real Git spawn handshake absent'
    message = os.read(read_fd, 64); os.close(read_fd)
    assert message.endswith(b'\n') and message[:-1].isdigit(), 'invalid real PID handshake'
    git_pid = int(message)
    def stopped_git():
        observed = _task4_live(git_pid)
        return observed if observed['state'] in ('T', 't') else None
    git = _task4_wait(stopped_git, 1)
    guard = _task4_live(guard_pid)
    assert git['ppid'] == guard_pid and guard['ppid'] == controller['pid']
    assert git['pgid'] == guard['pgid'] == controller['pgid'] == controller['sid'] == controller['pid']
    assert git['sid'] == guard['sid'] == controller['sid']
    assert (git['exe'], git['exe_bytes'], git['exe_sha256']) == (git_ref['path'], git_ref['bytes'], git_ref['sha256'])
    assert git['argv'] == expected_argv
    claim_path = output / ('owner-claim-task4-' + mode + '.json'); claim_ref = protocol.file_reference(claim_path)
    claim = json.loads(claim_path.read_text())
    assert claim['launch'] == prepared['reference'] and claim['identity']['pid'] == guard_pid
    assert claim['identity']['start_identity'] == guard['start_identity']
    ready = dict(controller=controller, guard=guard, git=git, git_reference=git_ref, claim=claim_ref,
                 launch=prepared['reference'], at_ns=time.monotonic_ns(), host=os.uname().nodename,
                 boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip())
    _task4_write(output / 'task4-live.json', ready)
    approval = output / 'task4-stop-approved.json'
    _task4_wait(approval.exists, 5)
    approved = json.loads(approval.read_text())
    assert approved['git']['pid'] == git_pid and approved['git']['start_identity'] == git['start_identity']
    assert _task4_stat(guard_pid)['start_identity'] == guard['start_identity']
    os.kill(guard_pid, signal.SIGKILL)
    waited = _task4_wait(lambda: (result if (result := os.waitpid(guard_pid, os.WNOHANG))[0] else None), 2)
    assert waited[0] == guard_pid and os.WIFSIGNALED(waited[1]) and os.WTERMSIG(waited[1]) == signal.SIGKILL
    still_live = stopped_git(); assert still_live is not None and still_live['start_identity'] == git['start_identity']
    _task4_write(output / 'task4-parent.json', dict(guard_wait_pid=waited[0], guard_wait_status=waited[1],
        git_after_guard_death=still_live, at_ns=time.monotonic_ns()))
    if mode == 'controller-exit': os._exit(0)
    while True: signal.pause()  # Real sealed P2 timeout must terminate this controller.


def _task4_lease(mode):
    import datetime
    import resource
    import bac_hardware_protocol_v1 as protocol
    assert os.environ.get('BAC_P2_TASK4_LEASE') and os.environ.get('BAC_P2_TASK4_LEASE_SHA256'), 'actual Task4 root lease required'
    path = Path(os.environ['BAC_P2_TASK4_LEASE']); reference = protocol.file_reference(path)
    assert reference['sha256'] == os.environ['BAC_P2_TASK4_LEASE_SHA256']
    value = json.loads(path.read_text())
    assert set(value) == set('schema authority start end recipe source_freezes pins cpu_ids ram_limit_bytes execution_root cases'.split())
    assert value['schema'] == 'bac-p2-task4-execution-lease-v1' and value['authority'] == 'root'
    now = datetime.datetime.now(datetime.timezone.utc)
    assert datetime.datetime.fromisoformat(value['start']) <= now
    assert now + datetime.timedelta(seconds=35) <= datetime.datetime.fromisoformat(value['end'])
    assert type(value['cpu_ids']) is list and all(type(cpu) is int for cpu in value['cpu_ids'])
    assert value['cpu_ids'] == sorted(os.sched_getaffinity(0)) and len(value['cpu_ids']) == 4
    assert type(value['source_freezes']) is list and len(value['source_freezes']) == 2
    assert type(value['ram_limit_bytes']) is int and value['ram_limit_bytes'] == 2 * 1024**3
    assert resource.getrlimit(resource.RLIMIT_AS) == (value['ram_limit_bytes'], value['ram_limit_bytes'])
    assert set(value['cases']) == {'controller-exit', 'job-timeout'}
    assert value['recipe']['sha256'] == 'c8fd6245cf2b80f423121fe3bf7047d19c8ade07dcdef3445da73c716167999d'
    for item in value['pins'] + [value['recipe']] + value['source_freezes']: protocol.validate_record('reference', item)
    required = [RUNNER, Path(__file__), Q / 'bac_hardware_protocol_v1.py', Q / 'bac_hardware_identity_v1.py', Path(sys.executable).resolve()]
    assert all(protocol.file_reference(p) in value['pins'] for p in required)
    assert protocol.file_reference(Q / 'bac_hardware_identity_v1.py')['sha256'] == 'dee9852055bdc1427567f217ae83f8bcc837f158aa753c2d8a95e2f701f08b07'
    selected = value['cases'][mode]; assert set(selected) == {'spec', 'capture_custody_ref'}
    protocol.validate_record('reference', selected['capture_custody_ref'])
    protocol.validate_record('reference', selected['spec'])
    spec = json.loads(Path(selected['spec']['path']).read_text()); protocol.validate_spec(spec)
    assert spec['source_freeze'] in value['source_freezes'] and len(spec['jobs']) == 1 and spec['profile'] == 'cpu'
    job = spec['jobs'][0]; case = f'test_run_bac_hardware_v1.py::test_owner_actual_git_backstop[{mode}]'
    assert job['required_collected_ids'] == [case] and job['environment']['BAC_P2_TASK4_DRIVER'] == mode
    assert job['timing'] == 'exclusive-host' and job['timeout_seconds'] == 15 and job['termination_grace_seconds'] == .1
    assert job['budget']['cpu_slots'] == 3 and job['devices'] == []
    assert job['argv'][0] == str(Path(sys.executable).resolve()) and str(Q / case) in job['argv']
    assert '--rootdir=' + str(Q) in job['argv'] and '--confcutdir=' + str(Q) in job['argv']
    assert 'bac_hardware_identity_v1' in job['argv'] and '--bac-identity-policy' in job['argv']
    for key in ('output_root', 'scratch_root'): assert Path(spec[key]).is_relative_to(Path(value['execution_root']))
    environment = json.loads(Path(spec['dependencies']['path']).read_text())
    git = next(item['reference'] for item in environment['dependencies'] if item['name'] == 'git')
    assert git in value['pins']
    return value, selected['spec'], spec, job


@pytest.mark.parametrize('mode', ['controller-exit', 'job-timeout'])
def test_owner_actual_git_backstop(tmp_path, monkeypatch, mode):
    if os.environ.get('BAC_P2_TASK4_DRIVER'):
        assert os.environ['BAC_P2_TASK4_DRIVER'] == mode
        _task4_driver(mode)
        raise AssertionError('negative-path controller unexpectedly returned')
    import ctypes
    import hashlib
    lease, spec_ref, plan, selected = _task4_lease(mode)
    r = implementation(); events = []; errors = []; ready = {}; stopping = threading.Event(); active = [True]
    output = Path(selected['output']); packet = tmp_path / 'task4-events.json'
    real_close, real_wait, real_killpg = r.HeldLeases.close, subprocess.Popen.wait, os.killpg
    job_names = sorted(hashlib.sha256(key.encode()).hexdigest() + '.lock' for key in
                       ['host-timing', *('domain:' + x for x in selected['serialization_domains'])])
    libc = ctypes.CDLL(None, use_errno=True); prior_subreaper = ctypes.c_int()
    assert libc.prctl(37, ctypes.byref(prior_subreaper), 0, 0, 0) == 0
    assert libc.prctl(36, 1, 0, 0, 0) == 0
    def event(kind, **fields): events.append(dict(kind=kind, at_ns=time.monotonic_ns(), **fields))
    def probe(expect_blocked):
        handle = r.Leases(Path(plan['lease_root']))
        try:
            held = handle.try_acquire(selected)
            event('lease-probe', blocked=held is None)
            if held is not None: real_close(held)
            if (held is None) != expect_blocked: errors.append('lease contention/reuse mismatch')
        finally: handle.close()
    def monitor():
        try:
            while not stopping.is_set():
                path = output / 'task4-live.json'
                if path.exists(): break
                stopping.wait(.005)
            if stopping.is_set(): return
            ready.update(json.loads(path.read_text()))
            git = _task4_live(ready['git']['pid']); controller = _task4_live(ready['controller']['pid'])
            assert git == ready['git'] and git['state'] in ('T', 't')
            assert controller['pgid'] == controller['sid'] == controller['pid'] == git['pgid']
            event('supervisor-live-git', observation=git); probe(True)
            _task4_write(output / 'task4-stop-approved.json', dict(git=git, at_ns=time.monotonic_ns()))
        except BaseException as exc: errors.append('monitor: ' + repr(exc))
    def close(held):
        relevant = active[0] and sorted(held.names) == job_names
        if relevant:
            try:
                assert ready, 'no verified live Git setup before lease close'
                identity = _task4_stat(ready['git']['pid'])
                assert identity['start_identity'] == ready['git']['start_identity']
                status = os.waitid(os.P_PID, identity['pid'], os.WEXITED | os.WNOHANG | os.WNOWAIT)
                event('before-job-lease-close', git=identity, wait_status=None if status is None else list(status),
                      locks=[dict(name=name, device=os.fstat(fd).st_dev, inode=os.fstat(fd).st_ino)
                             for fd, name in zip(held.descriptors, held.names)])
                if identity['state'] != 'Z' or status is None: errors.append('live Git at lease release')
                elif status.si_code != os.CLD_KILLED or status.si_status != signal.SIGKILL: errors.append('unexpected Git termination')
                probe(True)
            except BaseException as exc: errors.append('lease observation: ' + repr(exc))
        real_close(held)
        if relevant: event('after-job-lease-close')
    def wait(process, *args, **kwargs):
        code = real_wait(process, *args, **kwargs); event('actual-popen-wait', pid=process.pid, code=code); return code
    def killpg(pgid, signum):
        event('actual-group-signal', pgid=pgid, signal=signum); return real_killpg(pgid, signum)
    def alarm(signum, frame):
        event('watchdog-failure'); raise TimeoutError('actual Task4 exceeded30s; not a passing cleanup')
    thread = threading.Thread(target=monitor, daemon=True)
    previous_alarm = signal.signal(signal.SIGALRM, alarm)
    monkeypatch.setattr(r.HeldLeases, 'close', close); monkeypatch.setattr(subprocess.Popen, 'wait', wait)
    monkeypatch.setattr(r.os, 'killpg', killpg)
    thread.start(); signal.setitimer(signal.ITIMER_REAL, 30)
    try:
        code = r.run(Path(spec_ref['path']), capture_custody_ref=lease['cases'][mode]['capture_custody_ref']); active[0] = False
        assert code == 1 and ready, 'negative actual case was successful or lacked setup'
        parent = json.loads((output / 'task4-parent.json').read_text())
        assert parent['guard_wait_pid'] == ready['guard']['pid'] and os.WIFSIGNALED(parent['guard_wait_status'])
        assert os.WTERMSIG(parent['guard_wait_status']) == signal.SIGKILL
        reaped = _task4_wait(lambda: (v if (v := os.waitpid(ready['git']['pid'], os.WNOHANG))[0] else None), 2)
        event('supervisor-actual-git-reap', pid=reaped[0], wait_status=reaped[1])
        assert os.WIFSIGNALED(reaped[1]) and os.WTERMSIG(reaped[1]) == signal.SIGKILL
        result = json.loads((output / 'result.json').read_text())
        if mode == 'controller-exit': assert result['exit_code'] == 0 and result['disposition'] == 'completed-unverified'
        else: assert result['disposition'] == 'interrupted' and 'job timeout' in result['reason']
        assert result['disposition'] != 'successful'
        for who in ('controller', 'guard', 'git'): assert not Path(f"/proc/{ready[who]['pid']}").exists()
        with pytest.raises(ProcessLookupError): real_killpg(ready['controller']['pgid'], 0)
        before = next(item for item in events if item['kind'] == 'before-job-lease-close')
        leader_wait = next(item for item in events if item['kind'] == 'actual-popen-wait' and item['pid'] == ready['controller']['pid'])
        kill = next(item for item in events if item['kind'] == 'actual-group-signal' and item['signal'] == signal.SIGKILL and item['pgid'] == ready['controller']['pgid'])
        assert kill['at_ns'] <= leader_wait['at_ns'] <= before['at_ns']
        probe(False); assert not errors, errors
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0); signal.signal(signal.SIGALRM, previous_alarm)
        stopping.set(); thread.join(timeout=1); active[0] = False
        # Failure cleanup is separate evidence and cannot manufacture PASS.
        for who in ('git', 'guard', 'controller'):
            observed = ready.get(who)
            if observed is None: continue
            try:
                current = _task4_stat(observed['pid'])
                if current['start_identity'] != observed['start_identity']: continue
                event('emergency-cleanup', process=current); errors.append('emergency cleanup was required')
                os.kill(observed['pid'], signal.SIGKILL)
                try: _task4_wait(lambda: (v if (v := os.waitpid(observed['pid'], os.WNOHANG))[0] else None), 2)
                except (ChildProcessError, AssertionError): pass
            except ProcessLookupError: pass
            except FileNotFoundError: pass
        # Include any adopted descendant whose live-group setup failed early.
        launch_path = output / 'launch.json'
        if launch_path.exists():
            controller_pid = json.loads(launch_path.read_text())['identity']['pid']
            children = Path(f'/proc/self/task/{os.getpid()}/children').read_text().split()
            for child in children:
                try:
                    observed = _task4_stat(int(child))
                    if observed['pgid'] != controller_pid: continue
                    event('emergency-adopted-group-cleanup', process=observed); errors.append('adopted group required emergency cleanup')
                    os.kill(observed['pid'], signal.SIGKILL)
                    _task4_wait(lambda: (v if (v := os.waitpid(observed['pid'], os.WNOHANG))[0] else None), 2)
                except (ProcessLookupError, FileNotFoundError, ChildProcessError): pass
        restored = libc.prctl(36, prior_subreaper.value, 0, 0, 0)
        unchanged = [r.file_reference(Path(pin['path'])) == pin for pin in lease['pins']]
        _task4_write(packet, dict(events=events, errors=errors, lease=lease, spec=spec_ref,
                                 source_unchanged=unchanged, failure=repr(sys.exc_info()[1]), subreaper_restore=restored))
    assert restored == 0 and all(unchanged) and not errors and not thread.is_alive()
