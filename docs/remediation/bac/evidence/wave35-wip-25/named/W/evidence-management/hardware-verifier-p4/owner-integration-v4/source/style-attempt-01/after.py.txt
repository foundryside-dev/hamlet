"""Independent BAC evidence verification, with no APP/hardware/executable dispatch.

Only the root-authorized exact spec under no-writer custody is an authority input.
Hashes detect observed changes; they are not proof of authorship. Common checks do
not implement strict child/fork/phase, application or real CUDA case validators.
"""

import datetime
import hashlib
import itertools
import json
import os
import re
import stat
from pathlib import Path

import bac_hardware_protocol_v1 as protocol

COMMON = frozenset(("process-custody", "source-freeze", "required-cases"))
PHASES = ("setup", "call", "teardown")
STRICT = frozenset(("strict-child-core", "strict-child-producer", "rng-custody", "fork-three-trials", "phase-pair"))
P5_SHA256 = "ecc7b3c32716adfb19bf8e115f835dffb12468549d877561079db57a8a37e83c"
PROTOCOL_SHA256 = "e4f31eacb0e4499655a19ec91a4f5618a7b2ad3f4090e7ed068816b311e00860"


class IncompleteEvidence(ValueError):
    """A terminal attempt lacks a witness; this is never qualification."""


def require(condition, message):
    if not condition:
        raise ValueError(message)


def exact(value, fields, label):
    require(type(value) is dict and set(value) == set(fields.split()), label + ": exact fields required")


def pairs(items):
    result = {}
    for key, value in items:
        require(key not in result, "duplicate JSON key")
        result[key] = value
    return result


def read_bytes(reference):
    protocol.validate_record("reference", reference)
    descriptors = _safe_parent(reference["path"])
    try:
        fd = os.open(Path(reference["path"]).name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=descriptors[-1])
        with os.fdopen(fd, "rb") as stream:
            before = os.fstat(stream.fileno())
            require(stat.S_ISREG(before.st_mode), "nonregular referenced file")
            data = stream.read()
            after = os.fstat(stream.fileno())
        require(
            (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns, before.st_ctime_ns)
            == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns),
            "reference changed while reading",
        )
        require(
            len(data) == reference["bytes"] and hashlib.sha256(data).hexdigest() == reference["sha256"], "reference changed while reading"
        )
        check = _safe_parent(reference["path"])
        try:
            require(
                len(check) == len(descriptors)
                and all(
                    (os.fstat(a).st_dev, os.fstat(a).st_ino) == (os.fstat(b).st_dev, os.fstat(b).st_ino) for a, b in zip(check, descriptors)
                ),
                "read namespace changed",
            )
            current = os.stat(Path(reference["path"]).name, dir_fd=check[-1], follow_symlinks=False)
            require((current.st_dev, current.st_ino) == (before.st_dev, before.st_ino), "read leaf changed")
        finally:
            for fd in reversed(check):
                os.close(fd)
        protocol.validate_record("reference", reference)
        return data
    except OSError as exc:
        raise ValueError("unsafe referenced file") from exc
    finally:
        for fd in reversed(descriptors):
            os.close(fd)


def decode(data):
    try:
        value = json.loads(data, object_pairs_hook=pairs, parse_constant=lambda _: (_ for _ in ()).throw(ValueError("nonfinite JSON")))
        require(type(value) is dict, "JSON object required")
        return value
    except (UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError("unreadable JSON record") from exc


def load(reference, kind=None):
    value = decode(read_bytes(reference))
    if kind:
        protocol.validate_record(kind, value)
    return value


def ref(path):
    return protocol.file_reference(Path(path))


def key(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"))


def unique(values, label):
    require(len(values) == len({key(v) for v in values}), label + ": duplicate")


def same_set(first, second, label):
    unique(first, label)
    unique(second, label)
    require({key(v) for v in first} == {key(v) for v in second}, label + ": set mismatch")


def bound(value, spec_ref, spec, job=None):
    require(value["spec"] == spec_ref and value["attempt_id"] == spec["attempt_id"], "spec/attempt binding")
    if job:
        require(value["job_id"] == job["id"], "job binding")


def utc():
    return datetime.datetime.now(datetime.UTC).strftime("%Y-%m-%dT%H:%M:%S.%fZ")


def identity():
    raw = Path("/proc/self/stat").read_text().rsplit(")", 1)[1].split()
    return {
        "pid": os.getpid(),
        "ppid": os.getppid(),
        "start_identity": raw[19],
        "host": os.uname().nodename,
        "boot_id": Path("/proc/sys/kernel/random/boot_id").read_text().strip(),
    }


def _safe_parent(path):
    raw = str(path)
    require(
        raw.startswith("/") and "\\" not in raw and "\x00" not in raw and all(p not in ("", ".", "..") for p in raw[1:].split("/")),
        "unsafe destination",
    )
    fds = [os.open("/", os.O_RDONLY | os.O_DIRECTORY)]
    try:
        for part in raw[1:].split("/")[:-1]:
            fds.append(os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fds[-1]))
        return fds
    except OSError as exc:
        for fd in reversed(fds):
            os.close(fd)
        raise ValueError("unsafe receipt parent") from exc


def publish(path, value):
    fds = _safe_parent(path)
    try:
        fd = os.open(Path(path).name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600, dir_fd=fds[-1])
        with os.fdopen(fd, "wb") as output:
            output.write((json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + "\n").encode())
            output.flush()
            os.fsync(output.fileno())
        # Revalidate namespace after publication; preserve incomplete artifacts.
        check = _safe_parent(path)
        try:
            require(
                all((os.fstat(a).st_dev, os.fstat(a).st_ino) == (os.fstat(b).st_dev, os.fstat(b).st_ino) for a, b in zip(fds, check)),
                "receipt namespace changed",
            )
        finally:
            for descriptor in reversed(check):
                os.close(descriptor)
        return ref(path)
    except FileExistsError:
        raise
    except OSError as exc:
        raise ValueError("receipt publication failed") from exc
    finally:
        for descriptor in reversed(fds):
            os.close(descriptor)


def source_check(reference, candidate_ref, candidate, freeze_ref, freeze, grant, expected_identity=None):
    witness = load(reference, "source-witness")
    require(witness["candidate"] == candidate_ref and witness["source_freeze"] == freeze_ref, "source binding")
    require(witness["head"] == candidate["head"] and witness["tree"] == candidate["tree"], "source head/tree")
    inputs = [row["reference"] for rows in candidate["inventory"].values() for row in rows]
    same_set(witness["inventory"], inputs, "source inventory")
    require(witness["at"] >= freeze["start"] and witness["at"] >= grant["start"], "source witness before freeze")
    if freeze["end"]["state"] == "measured":
        require(witness["at"] <= freeze["end"]["value"], "source freeze expired")
    if grant["end"] is not None:
        require(witness["at"] <= grant["end"], "grant expired")
    if expected_identity is not None:
        require(witness["process_identity"] == expected_identity, "source observer identity")
    return witness


def initial(spec_path, job_id):
    spec_ref = ref(spec_path)
    spec = load(spec_ref, "spec")
    jobs = [j for j in spec["jobs"] if j["id"] == job_id]
    require(len(jobs) == 1, "unknown job")
    job = jobs[0]
    tools = {t["name"]: t["reference"] for t in spec["tools"]}
    require(tools["protocol"] == ref(protocol.__file__) and tools["protocol"]["sha256"] == PROTOCOL_SHA256, "unpinned protocol")
    require(tools["verifier"] == ref(__file__), "verifier source mismatch")
    require(tools["capture"]["sha256"] == "d91d9a4eac393d1283de0fa2a894a6f7a837aa9e28aa854c4795462f5229a1bb", "unpinned capture tool")
    capture_result_ref = ref(Path(spec_path).parent / "result.json")
    captured = load(capture_result_ref)
    exact(captured, "schema disposition reasons admission spec", "capture terminal")
    require(
        captured
        == {
            "schema": "bac-hardware-capture-result-v1",
            "disposition": "admitted",
            "reasons": [],
            "admission": spec["admission"],
            "spec": spec_ref,
        },
        "capture terminal did not admit exact spec",
    )
    admission = load(spec["admission"], "admission")
    require(
        admission["disposition"] == "admitted"
        and admission["profile"] == spec["profile"]
        and admission["admitted_budget"] == spec["budget"],
        "admission/spec binding",
    )
    require(admission["environment"] == spec["dependencies"], "dependency admission binding")
    dependencies = load(spec["dependencies"], "environment")
    require(dependencies["interpreter"]["path"] == job["interpreter"], "interpreter binding")
    request = load(admission["request"], "admission-request")
    for field in (
        "attempt_id",
        "profile",
        "authorized_root",
        "output_root",
        "scratch_root",
        "lease_root",
        "source_freeze",
        "tools",
        "dependencies",
    ):
        require(request[field] == spec[field], "request/spec " + field)
    require(request["requested_job_concurrency"] == spec["job_concurrency"], "concurrency binding")
    require(len(request["jobs"]) == len(spec["jobs"]), "job inventory binding")
    for requested, normalized in zip(request["jobs"], spec["jobs"]):
        a, b = dict(requested), dict(normalized)
        a["devices"] = [{k: v for k, v in d.items() if k != "mapping_provenance"} for d in a["devices"]]
        b["devices"] = [{k: v for k, v in d.items() if k != "mapping_provenance"} for d in b["devices"]]
        require(a == b, "normalized job differs from requested contract")
        require(all(d in admission["devices"] for d in normalized["devices"]), "device assignment not admitted")
    candidate = load(spec["candidate"], "candidate")
    freeze = load(spec["source_freeze"], "source-freeze")
    require(
        candidate["authorized_root"] == freeze["authorized_root"] == spec["authorized_root"]
        and candidate["source_freeze"] == spec["source_freeze"],
        "candidate/freeze binding",
    )
    grant = load(freeze["grant"])
    exact(
        grant,
        "schema authorized_root head tree import_roots inputs start end no_writer_barrier includes_untracked authority allocation_grant cpu_reservation_evidence",
        "source grant",
    )
    require(
        grant["schema"] == "bac-hardware-source-grant-v1" and grant["no_writer_barrier"] is True and grant["includes_untracked"] is True,
        "root no-writer grant required",
    )
    for field in ("authorized_root", "head", "tree", "import_roots"):
        require(grant[field] == candidate[field], "grant/candidate " + field)
    inputs = [r["reference"] for rows in candidate["inventory"].values() for r in rows]
    same_set(grant["inputs"], inputs, "grant inputs")
    protocol.validate_record("reference", grant["allocation_grant"])
    for item in grant["cpu_reservation_evidence"]:
        protocol.validate_record("reference", item)
    require(type(grant["authority"]) is str and grant["authority"].strip(), "missing authority statement")
    require(grant["start"] <= freeze["start"], "freeze starts before grant")
    capture_files = capture_custody(spec_ref, spec, request, admission, candidate, dependencies, tools)
    return spec_ref, spec, job, tools, candidate, freeze, grant, dependencies, capture_result_ref, capture_files


def capture_custody(spec_ref, spec, request, admission, candidate, dependencies, tools):
    """Fixed capture siblings require the caller's external namespace custody."""
    directory = Path(spec_ref["path"]).parent
    files = {
        name: ref(directory / name)
        for name in (
            "start.json",
            "source-before.json",
            "source-after.json",
            "source-before-git.json",
            "source-after-git.json",
            "discovery.json",
            "discovery-command.json",
        )
    }
    start, wrapper, raw = (load(files[name]) for name in ("start.json", "discovery-command.json", "discovery.json"))
    exact(start, "at candidate request tools dependencies", "capture start")
    require(
        start["candidate"] == spec["candidate"]
        and start["request"] == admission["request"]
        and start["tools"] == spec["tools"]
        and start["dependencies"] == spec["dependencies"],
        "capture start binding",
    )
    exact(
        wrapper,
        "schema request capture executable capture_identity argv cwd environment start end identity exit_code stdout stderr errors",
        "discovery wrapper",
    )
    require(
        wrapper["schema"] == "bac-hardware-discovery-command-v1"
        and admission["discovery_process"] == files["discovery-command.json"]
        and wrapper["request"] == admission["request"]
        and wrapper["capture"] == tools["capture"]
        and wrapper["executable"] == request["interpreter"],
        "discovery wrapper pins",
    )
    require(type(wrapper["stdout"]) is str and type(wrapper["stderr"]) is str, "discovery output types")
    exact(
        raw,
        "schema profile identity at argv interpreter cwd inherited_visibility effective_visibility tmpdir cgroup_root cgroup_leaf "
        "ancestry_complete cgroups affinity meminfo scratch_free_bytes scratch_free_inodes cpu_topology numa_topology gpu_topology "
        "competing_workloads torch_build cuda_runtime driver devices probe_errors commands python_version dependencies torch_origin",
        "raw discovery",
    )
    require(type(raw["commands"]) is list and all(type(row) is dict for row in raw["commands"]), "discovery commands type")
    require(raw["inherited_visibility"] is None or type(raw["inherited_visibility"]) is str, "inherited visibility type")
    require(
        raw["schema"] == "bac-hardware-discovery-v1"
        and decode(wrapper["stdout"].encode()) == raw
        and wrapper["identity"] == raw["identity"]
        and raw["profile"] == spec["profile"]
        and type(wrapper["exit_code"]) is int
        and wrapper["exit_code"] == 0
        and wrapper["errors"] == []
        and admission["reasons"] == admission["probe_errors"] == raw["probe_errors"] == [],
        "discovery successful binding",
    )
    for ident in (wrapper["capture_identity"], raw["identity"]):
        protocol.validate_record("identity", ident)
    require(
        raw["identity"]["ppid"] == wrapper["capture_identity"]["pid"]
        and all(raw["identity"][k] == wrapper["capture_identity"][k] for k in ("host", "boot_id")),
        "discovery actual parent",
    )
    bootstrap = (
        "import sys;sys.path.insert(0,"
        + repr(str(Path(tools["capture"]["path"]).parent))
        + ");from capture_bac_hardware_v1 import discovery_main;discovery_main()"
    )
    expected_argv = [
        request["interpreter"]["path"],
        "-I",
        "-B",
        "-c",
        bootstrap,
        admission["request"]["path"],
        json.dumps(raw["inherited_visibility"]),
    ]
    environment = dict(dependencies["environment"], CUDA_VISIBLE_DEVICES=request["requested_visibility"], TMPDIR=request["scratch_root"])
    require("BAC_CAPTURE_LEASE_FDS" not in environment, "caller supplied capture lease descriptors")
    lease_text = wrapper["environment"].get("BAC_CAPTURE_LEASE_FDS")
    try:
        leases = json.loads(lease_text)
    except (TypeError, ValueError) as exc:
        raise ValueError("discovery lease descriptors") from exc
    require(
        type(leases) is list
        and leases
        and all(type(fd) is int and fd > 2 for fd in leases)
        and len(set(leases)) == len(leases)
        and json.dumps(leases) == lease_text,
        "discovery lease descriptor shape",
    )
    environment["BAC_CAPTURE_LEASE_FDS"] = lease_text
    require(
        wrapper["argv"] == raw["argv"] == expected_argv
        and wrapper["environment"] == environment
        and wrapper["cwd"] == raw["cwd"] == spec["authorized_root"]
        and raw["interpreter"] == request["interpreter"]["path"]
        and raw["tmpdir"] == request["scratch_root"]
        and raw["effective_visibility"] == request["requested_visibility"]
        and raw["python_version"] == dependencies["python_version"],
        "discovery actual context",
    )
    same_set(raw["dependencies"], dependencies["dependencies"], "discovery dependencies")
    git = [row for row in dependencies["dependencies"] if row["name"] == "git"]
    require(len(git) == 1, "capture one Git dependency")
    git_ref = git[0]["reference"]
    read_bytes(git_ref)
    git_env = {"PATH": "/usr/bin:/bin", "LC_ALL": "C", "GIT_CONFIG_NOSYSTEM": "1", "GIT_CONFIG_GLOBAL": "/dev/null"}

    def timestamp(value):
        require(
            type(value) is str and re.fullmatch(r"[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}\.[0-9]{6}Z", value),
            "capture UTC type",
        )
        datetime.datetime.strptime(value, "%Y-%m-%dT%H:%M:%S.%fZ").replace(tzinfo=datetime.UTC)
        return value

    for value in (start["at"], wrapper["start"], wrapper["end"], raw["at"], admission["at"]):
        timestamp(value)

    def native(row, argv, parent, earliest, latest):
        exact(
            row,
            "schema argv executable executable_after cwd environment identity start end exit_code stdout stderr errors",
            "native Git command",
        )
        protocol.validate_record("identity", row["identity"])
        timestamp(row["start"])
        timestamp(row["end"])
        require(row["identity"]["start_identity"].isdigit() and row["identity"]["pid"] != parent["pid"], "native Git birth identity")
        require(
            row["schema"] == "bac-hardware-git-command-v1"
            and row["argv"] == argv
            and row["executable"] == row["executable_after"] == git_ref
            and row["cwd"] == spec["authorized_root"]
            and row["environment"] == git_env
            and type(row["exit_code"]) is int
            and row["exit_code"] == 0
            and row["errors"] == []
            and type(row["stdout"]) is str
            and type(row["stderr"]) is str,
            "native Git successful command",
        )
        require(
            row["identity"]["ppid"] == parent["pid"]
            and all(row["identity"][k] == parent[k] for k in ("host", "boot_id"))
            and earliest <= row["start"] <= row["end"] <= latest,
            "native Git parent/interval",
        )

    require(wrapper["start"] <= raw["at"] <= wrapper["end"], "discovery interval")
    versions = [row for row in raw["commands"] if row.get("schema") == "bac-hardware-git-command-v1"]
    require(len(versions) == 1, "discovery one native version command")
    native(versions[0], [git_ref["path"], "--version"], raw["identity"], raw["at"], wrapper["end"])
    version = versions[0]["stdout"].removesuffix("\n")
    require(version and "\n" not in version and "\r" not in version and version == git[0]["version"], "observed Git version")
    if spec["profile"] == "cpu":
        require(
            raw["commands"] == versions
            and not raw["devices"]
            and raw["effective_visibility"] == ""
            and all(raw[name] is None for name in ("torch_origin", "torch_build", "cuda_runtime", "driver", "gpu_topology")),
            "CPU discovery native-only",
        )
    prefix = [git_ref["path"], "--no-optional-locks", "-c", "core.fsmonitor=false", "-c", "core.pager=cat", "-C", spec["authorized_root"]]
    tails = [["rev-parse", "HEAD"], ["diff-index", "--cached", "--quiet", candidate["tree"], "--"], ["worktree", "list", "--porcelain"]]
    inputs = [
        row["reference"]
        for name in ("source", "tests", "configs", "scripts", "ratchet", "artifacts")
        for row in candidate["inventory"][name]
    ]
    for phase, earliest, latest in (("before", start["at"], wrapper["start"]), ("after", wrapper["end"], admission["at"])):
        source = load(files["source-" + phase + ".json"])
        exact(source, "at head tree inventory registry availability", "capture source")
        timestamp(source["at"])
        require(
            source["head"] == candidate["head"]
            and source["tree"] == candidate["tree"]
            and source["inventory"] == inputs
            and earliest <= source["at"] <= latest,
            "capture source binding",
        )
        commands = json.loads(
            read_bytes(files["source-" + phase + "-git.json"]),
            object_pairs_hook=pairs,
            parse_constant=lambda _: (_ for _ in ()).throw(ValueError("nonfinite capture command")),
        )
        require(type(commands) is list and len(commands) == 3, "capture three native source commands")
        previous = earliest
        for row, tail in zip(commands, tails):
            native(row, prefix + tail, wrapper["capture_identity"], previous, source["at"])
            previous = row["end"]
        require(
            commands[0]["stdout"].strip() == candidate["head"]
            and commands[2]["stdout"] == source["registry"] == read_bytes(candidate["registry"]).decode()
            and source["availability"] == load(candidate["availability"])["worktrees"],
            "capture source query observations",
        )
    for reference in files.values():
        read_bytes(reference)
    return list(files.values())


def _policy(job, spec_ref):
    args = job["argv"]
    values = []
    for flag in ("--bac-identity-policy", "--bac-identity-policy-sha256"):
        require(args.count(flag) == 1 and not any(a.startswith(flag + "=") for a in args), "identity policy argv")
        index = args.index(flag)
        require(index + 1 < len(args), "missing policy argument")
        values.append(args[index + 1])
    policy_ref = ref(values[0])
    require(policy_ref["sha256"] == values[1], "policy hash")
    policy = load(policy_ref)
    exact(policy, "schema spec_path command_path runtime_path job_id role_id required_imports worker_roles environment_delta", "policy")
    require(
        policy["schema"] == "bac-hardware-identity-policy-v1" and policy["spec_path"] == spec_ref["path"] and policy["job_id"] == job["id"],
        "policy binding",
    )
    require(
        policy["command_path"] == str(Path(job["output"]) / "command.json")
        and policy["runtime_path"] == str(Path(job["output"]) / "runtime-context.json"),
        "policy paths",
    )
    exact(policy["environment_delta"], "common worker", "environment delta")
    for group, allowed in (("common", {"PYTEST_VERSION", "LC_CTYPE"}), ("worker", {"COLUMNS", "LINES"})):
        values = policy["environment_delta"][group]
        require(
            type(values) is dict
            and all(
                type(k) is str and re.fullmatch(r"[A-Za-z_][A-Za-z0-9_]*", k) and type(v) is str and "\x00" not in v
                for k, v in values.items()
            ),
            "environment delta types",
        )
        require(set(values) <= allowed and not (set(values) & set(job["environment"])), "environment additions")
    roles = {r["id"]: r for r in job["roles"]}
    require(policy["role_id"] in roles and roles[policy["role_id"]]["role"] == "controller", "policy controller")
    require(type(policy["worker_roles"]) is list and len(policy["worker_roles"]) == job["xdist_workers"], "policy worker inventory")
    for role in roles.values():
        if role["role"] == "worker":
            require(policy["worker_roles"].count(role["id"]) == role["count"], "worker role policy count")
    unique([r["name"] for r in policy["required_imports"]], "required imports")
    for required in policy["required_imports"]:
        exact(required, "name reference phases", "required import")
        protocol.validate_record("named-reference", {k: required[k] for k in ("name", "reference")})
        unique(required["phases"], "import phases")
        require(bool(required["phases"]) and set(required["phases"]) <= {"start", "end"}, "required import phases")
    return policy_ref, policy


def _worker(value, job):
    exact(value, "workerid workercount testrunuid mainargv", "worker identity")
    require(type(value["workerid"]) is str and re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.-]{0,127}", value["workerid"]), "worker ID")
    require(type(value["workercount"]) is int and value["workercount"] == job["xdist_workers"], "worker count")
    require(
        type(value["testrunuid"]) is str and len(value["testrunuid"]) == 32 and all(c in "0123456789abcdef" for c in value["testrunuid"]),
        "worker run UID",
    )
    require(
        type(value["mainargv"]) is list and value["mainargv"] and all(type(x) is str and x for x in value["mainargv"]), "worker main argv"
    )


def processes(result, spec_ref, spec, job, command_ref, candidate, freeze, grant, dependencies):
    policy_ref, policy = _policy(job, spec_ref)
    require(policy_ref not in grant["inputs"], "policy cannot be candidate input")
    runtime_ref = ref(policy["runtime_path"])
    runtime = load(runtime_ref)
    exact(runtime, "schema spec command trusted_grant policy runner_identity runner_source_start", "runtime")
    require(
        runtime["schema"] == "bac-hardware-identity-runtime-v1"
        and runtime["spec"] == spec_ref
        and runtime["command"] == command_ref
        and runtime["trusted_grant"] == freeze["grant"]
        and runtime["policy"] == policy_ref,
        "runtime delegation",
    )
    protocol.validate_record("identity", runtime["runner_identity"])
    runner_start = source_check(
        runtime["runner_source_start"], spec["candidate"], candidate, spec["source_freeze"], freeze, grant, runtime["runner_identity"]
    )
    by_id, by_ref, raw_by_ref = {}, {}, {}
    unique(result["process_records"], "process records")
    roles = {r["id"]: r for r in job["roles"]}
    allowed_imports = [r["reference"] for rows in candidate["inventory"].values() for r in rows] + [
        d["reference"] for d in dependencies["dependencies"]
    ]
    for reference in result["process_records"]:
        process = load(reference, "process")
        bound(process, spec_ref, spec, job)
        require(process["role_id"] in roles and process["role"] == roles[process["role_id"]]["role"], "process role")
        role = roles[process["role_id"]]
        require(
            process["cwd"] == job["cwd"] and process["interpreter"] == dependencies["interpreter"] and process["devices"] == job["devices"],
            "process execution context",
        )
        require(process["at"] >= runner_start["at"] and process["at"] >= result["start"], "process before launch")
        if result["end"]["state"] == "measured":
            require(process["at"] <= result["end"]["value"], "process after terminal")
        require(
            process["source_witness"]["path"] == str(Path(job["output"]) / f"source-{process['process_id']}-{process['phase']}.json"),
            "process source path",
        )
        source = source_check(
            process["source_witness"], spec["candidate"], candidate, spec["source_freeze"], freeze, grant, process["identity"]
        )
        require(
            source["at"] == process["at"] if role["role"] == "fork-guard" else source["at"] >= process["at"],
            "source/process observation chronology",
        )
        require(all(i["reference"] in allowed_imports for i in process["imports"]), "unexpected import origin")
        for required in policy["required_imports"]:
            if process["phase"] in required["phases"]:
                require({k: required[k] for k in ("name", "reference")} in process["imports"], "required import absent")
        provenances = [process[k]["provenance"] for k in ("intra_op", "inter_op", "debug")]
        require(provenances[0] == provenances[1] == provenances[2], "fact provenance differs")
        raw_ref = provenances[0]
        require(
            raw_ref["path"] == str(Path(job["output"]) / f"process-{process['process_id']}-observation-{process['phase']}.json"),
            "raw observation path",
        )
        raw = load(raw_ref)
        exact(
            raw,
            "schema context identity phase environment worker_identity torch_loaded cuda_marker observation_limit at "
            "python_debug python_optimize torch_anomaly source_method recorder_mode module_names_before_sha256 "
            "module_names_after_sha256 owner_claim sampler",
            "raw",
        )
        require(raw["schema"] == "bac-hardware-process-observation-v2", "current raw schema")
        require(
            raw["python_debug"] is True and type(raw["python_optimize"]) is int and raw["python_optimize"] == 0,
            "Python assertions disabled or optimization mistyped",
        )
        require(raw["source_method"] == "direct-git", "raw source method")
        require(raw["recorder_mode"] == ("fork-guard" if role["role"] == "fork-guard" else "ordinary"), "raw recorder mode")
        for name in ("module_names_before_sha256", "module_names_after_sha256"):
            require(type(raw[name]) is str and re.fullmatch("[a-f0-9]{64}", raw[name]), "raw module-name digest")
        for name in ("owner_claim", "sampler"):
            require(type(raw[name]) is list and len(raw[name]) <= 1, "raw owner reference cardinality")
            for item in raw[name]:
                protocol.validate_record("reference", item)
        if role["role"] in ("controller", "worker"):
            require(not raw["owner_claim"] and not raw["sampler"], "raw unexpected owner claim/sampler")
        exact(raw["torch_anomaly"], "state value reason", "raw Torch anomaly")
        for field in ("identity", "phase", "environment", "at"):
            require(raw[field] == process[field], "raw/process " + field)
        require(type(raw["torch_loaded"]) is bool and (raw["cuda_marker"] is None or type(raw["cuda_marker"]) is bool), "raw bool types")
        require(type(raw["observation_limit"]) is str and raw["observation_limit"].strip(), "observation limitation required")
        context = raw["context"]
        exact(
            context,
            "schema spec command trusted_grant job_id role_id required_imports worker_roles environment_delta parent_start replacement_start lifecycle worker_identity policy runtime",
            "context",
        )
        require(
            context["schema"] == "bac-hardware-identity-context-v1"
            and context["spec"] == spec_ref
            and context["command"] == command_ref
            and context["trusted_grant"] == freeze["grant"]
            and context["job_id"] == job["id"]
            and context["role_id"] == process["role_id"],
            "context binding",
        )
        if process["role"] in ("controller", "worker"):
            require(
                context["policy"] == [policy_ref] and context["runtime"] == [runtime_ref], "missing delegated policy/runtime provenance"
            )
        for field in ("required_imports", "worker_roles", "environment_delta"):
            require(context[field] == policy[field], "derived policy mismatch")
        for field in ("parent_start", "replacement_start", "lifecycle"):
            require(type(context[field]) is list and len(context[field]) <= 1, "context reference cardinality")
            for item in context[field]:
                protocol.validate_record("reference", item)
        expected_env = {**job["environment"], **policy["environment_delta"]["common"]}
        if process["role"] == "controller":
            require(process["argv"] == job["argv"] and process["role_id"] == policy["role_id"], "controller argv/role")
            require(
                not any(context[f] for f in ("parent_start", "replacement_start", "lifecycle")) and context["worker_identity"] is None,
                "controller context ancestry",
            )
            require(not process["parent_identity"] and not process["replacement_of"], "controller parent fields")
            require(
                process["identity"]["ppid"] == runtime["runner_identity"]["pid"]
                and all(process["identity"][k] == runtime["runner_identity"][k] for k in ("host", "boot_id")),
                "controller runner identity",
            )
        elif process["role"] == "worker":
            _worker(context["worker_identity"], job)
            worker = context["worker_identity"]
            expected_env.update(policy["environment_delta"]["worker"])
            expected_env.update(
                PYTEST_XDIST_WORKER=worker["workerid"],
                PYTEST_XDIST_WORKER_COUNT=str(worker["workercount"]),
                PYTEST_XDIST_TESTRUNUID=worker["testrunuid"],
            )
            require(len(context["parent_start"]) == len(context["lifecycle"]) == 1, "worker missing parent lifecycle")
        else:
            expected_env = owner_context(process, raw, result, spec_ref, spec, job, command_ref, candidate, freeze, grant)
        require(
            raw["worker_identity"] == context["worker_identity"] and process["environment"] == expected_env,
            "actual environment/context mismatch",
        )
        if role["observer"] == "guard-only":
            require(
                process["torch_observation"] == "forbidden-inherited-module-witness" and process["cuda_observation"] == "forbidden",
                "guard observations",
            )
        elif not raw["torch_loaded"]:
            require(
                process["torch_observation"] == "not-loaded"
                and process["cuda_observation"] == "uninitialized"
                and raw["cuda_marker"] is None,
                "absent Torch observations",
            )
            require(
                not (process["phase"] == "end" and (role["observer"] == "work" or role["torch_policy"] == "work")),
                "work end has no Torch observations",
            )
        else:
            require(
                type(raw["cuda_marker"]) is bool
                and process["cuda_observation"] == ("initialized" if raw["cuda_marker"] else "uninitialized"),
                "CUDA marker",
            )
            require(role["cuda_policy"] != "no-initialize" or raw["cuda_marker"] is False, "observer initialized CUDA")
            require(
                process["torch_observation"]
                == ("work" if role["observer"] == "work" or role["torch_policy"] == "work" else "loaded-without-calls"),
                "Torch observation mode",
            )
        measured = raw["torch_loaded"] and process["torch_observation"] == "work"
        anomaly = raw["torch_anomaly"]
        require(anomaly["state"] == ("measured" if measured else "unavailable"), "raw Torch anomaly availability")
        require(
            (
                (type(anomaly["value"]) is bool and anomaly["reason"] == "")
                if measured
                else (anomaly["value"] is None and type(anomaly["reason"]) is str and bool(anomaly["reason"].strip()))
            ),
            "raw Torch anomaly observation",
        )
        for field in ("intra_op", "inter_op", "debug"):
            observed = role["role"] != "fork-guard" if field == "debug" else measured
            fact = process[field]
            require(fact["state"] == ("measured" if observed else "unavailable"), "role fact availability")
            if observed:
                require(type(fact["value"]) is type(role[field]) and fact["value"] == role[field], "role fact mismatch")
            if field == "debug" and observed:
                require(fact["value"] is raw["python_debug"], "Python debug observation mismatch")
        entry = by_id.setdefault(process["process_id"], {})
        require(process["phase"] not in entry, "duplicate process phase")
        entry[process["phase"]] = (reference, process, raw)
        by_ref[key(reference)] = process
        raw_by_ref[key(reference)] = raw
    require(by_id, "missing process observations")
    starts = []
    for phases in by_id.values():
        if set(phases) != {"start", "end"}:
            raise IncompleteEvidence("missing process start/end")
        sr, start, raw_start = phases["start"]
        _er, end, raw_end = phases["end"]
        for field in (
            "identity",
            "role_id",
            "role",
            "parent_identity",
            "replacement_of",
            "interpreter",
            "argv",
            "cwd",
            "environment",
            "devices",
        ):
            require(start[field] == end[field], "start/end " + field)
        require(raw_start["context"] == raw_end["context"], "start/end context")
        require(
            end["at"] >= start["at"]
            and end["monotonic_ns"] >= start["monotonic_ns"]
            and end["elapsed_ns"]["value"] == end["monotonic_ns"] - start["monotonic_ns"],
            "process time",
        )
        require(load(start["source_witness"], "source-witness")["at"] <= end["at"], "START source after END entry")
        starts.append((sr, start, raw_start))
    unique([s[1]["identity"] for s in starts], "process identity")
    unique([{k: s[1]["identity"][k] for k in ("pid", "start_identity", "host", "boot_id")} for s in starts], "process start token")
    pid_lifetimes = {}
    for _, process, _ in starts:
        ident = process["identity"]
        require(ident["start_identity"].isdigit(), "Linux process start token")
        pid_lifetimes.setdefault((ident["pid"], ident["host"], ident["boot_id"]), []).append(by_id[process["process_id"]])
    for lifetimes in pid_lifetimes.values():
        ordered = sorted(lifetimes, key=lambda row: row["start"][1]["monotonic_ns"])
        for earlier, later in itertools.pairwise(ordered):
            require(
                earlier["end"][1]["monotonic_ns"] <= later["start"][1]["monotonic_ns"]
                and earlier["end"][1]["at"] <= later["start"][1]["at"],
                "overlapping reused PID",
            )
    for role in roles.values():
        rows = [s for s in starts if s[1]["role_id"] == role["id"]]
        initial_rows = [s for s in rows if not s[1]["replacement_of"]]
        require(
            len(initial_rows) == role["count"] and len(rows) <= role["count"] + role["replacement_limit"], "role count/replacement limit"
        )
    owner_pairs(starts, by_ref, raw_by_ref, job, result)
    _lifecycles(starts, by_ref, job, result)
    return by_ref, raw_by_ref, runtime, policy, [s[0] for s in starts]


def uint(value, label):
    require(type(value) is int and value >= 0, label + ": exact nonnegative integer")
    return value


def owner_context(process, raw, result, spec_ref, spec, job, command_ref, candidate, freeze, grant):
    context = raw["context"]
    require(process["role"] in ("spawn-child", "work-child", "fork-guard"), "unsupported owner role")
    require(len(context["parent_start"]) == len(context["lifecycle"]) == len(raw["owner_claim"]) == 1, "owner ancestry/claim")
    parent_ref, launch_ref = context["parent_start"][0], context["lifecycle"][0]
    require(parent_ref in result["process_records"], "owner parent not observed")
    parent = load(parent_ref, "process")
    parent_context = load(parent["debug"]["provenance"])["context"]
    require(parent["phase"] == "start" and parent["identity"] != process["identity"], "owner parent START")
    expected = dict(
        parent_context,
        role_id=process["role_id"],
        parent_start=[parent_ref],
        lifecycle=[launch_ref],
        replacement_start=[],
        worker_identity=None,
        policy=[],
        runtime=[],
    )
    require(context == expected and not process["replacement_of"], "owner exact derived context")
    launch = load(launch_ref)
    exact(
        launch,
        "schema spec command owner_contract parent_process source_witness parent_identity parent_argv case_id pytest_phase "
        "child_role_id instance_id start_method environment python_assertions_enabled inherited_imports module_names_sha256 "
        "sampler_policy owner_deadline_ns sampler_cutoff_ns at monotonic_ns",
        "owner launch",
    )
    require(
        launch["schema"] == "bac-hardware-owner-launch-v1"
        and launch["spec"] == spec_ref
        and launch["command"] == command_ref
        and launch["parent_process"] == parent_ref
        and launch["parent_identity"] == parent["identity"]
        and launch["parent_argv"] == parent["argv"]
        and launch["child_role_id"] == process["role_id"]
        and launch["pytest_phase"] == "call"
        and launch["python_assertions_enabled"] is True,
        "owner launch bindings",
    )
    require(
        type(launch["instance_id"]) is str and re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.-]{0,127}", launch["instance_id"]), "owner instance"
    )
    require(launch_ref["path"] == str(Path(job["output"]) / ("owner-launch-" + launch["instance_id"] + ".json")), "owner launch path")
    require(
        type(launch["module_names_sha256"]) is str and re.fullmatch("[a-f0-9]{64}", launch["module_names_sha256"]), "owner module digest"
    )
    canonical = []
    for obligation in job["obligations"]:
        if obligation["kind"] in ("strict-child-core", "strict-child-producer") and launch["case_id"] in obligation["required_cases"]:
            contract = load(obligation["contract"], "case-contract")
            require(all(r in job["roles"] for r in contract["roles"]), "owner canonical role subset")
            for left, right in (
                ("obligation_kind", "kind"),
                ("required_cases", "required_cases"),
                ("validator", "validator"),
                ("evidence_class", "evidence_class"),
                ("minimum_devices", "minimum_devices"),
            ):
                require(contract[left] == obligation[right], "owner canonical obligation binding")
            if any(r["id"] == process["role_id"] for r in contract["roles"]):
                canonical.append(obligation["contract"])
    require(canonical == [launch["owner_contract"]], "owner canonical contract")
    require(
        process["parent_identity"] == [parent["identity"]]
        and process["identity"]["ppid"] == parent["identity"]["pid"]
        and all(process["identity"][f] == parent["identity"][f] for f in ("host", "boot_id")),
        "owner immediate ancestry",
    )
    environment = dict(parent["environment"])
    if parent["role"] in ("controller", "worker"):
        require("PYTEST_CURRENT_TEST" not in environment, "owner unexpected parent phase")
        environment["PYTEST_CURRENT_TEST"] = launch["case_id"] + " (call)"
    else:
        parent_launch = load(parent_context["lifecycle"][0])
        require(
            parent_launch["case_id"] == launch["case_id"] and parent_launch["owner_contract"] == launch["owner_contract"],
            "owner nested case",
        )
    require(launch["environment"] == environment, "owner launch full environment")
    require(
        parent["at"] <= launch["at"] <= process["at"]
        and parent["monotonic_ns"] <= uint(launch["monotonic_ns"], "owner launch") <= process["monotonic_ns"],
        "owner launch chronology",
    )
    require(
        launch["source_witness"]["path"] == str(Path(job["output"]) / ("source-launch-" + launch["instance_id"] + ".json")),
        "owner launch source path",
    )
    source = source_check(launch["source_witness"], spec["candidate"], candidate, spec["source_freeze"], freeze, grant, parent["identity"])
    require(parent["at"] <= source["at"] <= launch["at"], "owner launch source chronology")
    allowed = [r["reference"] for rows in candidate["inventory"].values() for r in rows]
    require(type(launch["inherited_imports"]) is list, "owner inherited imports")
    unique([item["name"] for item in launch["inherited_imports"]], "owner inherited imports")
    for item in launch["inherited_imports"]:
        protocol.validate_record("named-reference", item)
        require(item["reference"] in allowed, "owner inherited import source")
    claim_ref = raw["owner_claim"][0]
    claim = load(claim_ref)
    exact(claim, "schema launch identity role_id at monotonic_ns", "owner claim")
    require(
        claim["schema"] == "bac-hardware-owner-claim-v1"
        and claim["launch"] == launch_ref
        and claim["identity"] == process["identity"]
        and claim["role_id"] == process["role_id"]
        and launch["at"] <= claim["at"] <= process["at"]
        and launch["monotonic_ns"] <= uint(claim["monotonic_ns"], "owner claim"),
        "owner claim binding",
    )
    require(claim_ref["path"] == str(Path(job["output"]) / ("owner-claim-" + launch["instance_id"] + ".json")), "owner claim path")
    fork = process["role"] == "fork-guard"
    require(fork or claim["monotonic_ns"] <= process["monotonic_ns"], "owner claim chronology")
    require(launch["start_method"] == ("fork" if fork else "spawn"), "owner start method")
    require(type(launch["sampler_policy"]) is list and len(launch["sampler_policy"]) == int(fork), "owner sampler policy")
    if fork:
        require(
            process["argv"] == launch["parent_argv"] and process["imports"] == launch["inherited_imports"], "owner fork inherited context"
        )
        require(
            raw["module_names_before_sha256"] == raw["module_names_after_sha256"] == launch["module_names_sha256"],
            "owner fork module change",
        )
        require(len(raw["sampler"]) == 1, "owner fork sampler missing")
    else:
        require(not raw["sampler"] and launch["owner_deadline_ns"] is None and launch["sampler_cutoff_ns"] is None, "owner spawn sampler")
        require(process["argv"] != parent["argv"], "owner spawn actual argv")
    owner_resources(process, launch, job, candidate, grant)
    return environment


def owner_resources(process, launch, job, candidate, grant):
    role = next(r for r in job["roles"] if r["id"] == process["role_id"])
    reference = role["cpu_reservation_evidence"]
    require(reference in grant["cpu_reservation_evidence"], "owner CPU bound outside grant")
    bound = load(reference)
    exact(
        bound,
        "schema basis role_id role observer intra_op inter_op debug torch_policy cuda_policy peak_cpu_slots argv environment "
        "required_cases head tree inputs justification authority bound_statement",
        "owner CPU bound",
    )
    require(
        bound["schema"] == "bac-hardware-cpu-bound-v1"
        and bound["basis"] == role["cpu_reservation_basis"]
        and bound["role_id"] == role["id"],
        "owner CPU bound identity",
    )
    for field in ("role", "observer", "intra_op", "inter_op", "debug", "torch_policy", "cuda_policy", "peak_cpu_slots"):
        require(type(bound[field]) is type(role[field]) and bound[field] == role[field], "owner CPU role binding")
    require(
        bound["argv"] == job["argv"]
        and bound["environment"] == job["environment"]
        and bound["required_cases"] == job["required_collected_ids"]
        and bound["head"] == candidate["head"]
        and bound["tree"] == candidate["tree"]
        and bound["inputs"] == grant["inputs"],
        "owner CPU workload binding",
    )
    require(
        type(bound["justification"]) is list
        and bound["justification"]
        and all(type(bound[f]) is str and bound[f].strip() for f in ("authority", "bound_statement")),
        "owner CPU authority",
    )
    for item in bound["justification"]:
        protocol.validate_record("reference", item)
    require(all(r in bound["justification"] for r in launch["sampler_policy"]), "owner sampler justification")


def owner_pairs(starts, by_ref, raws, job, result):
    launches, claims = [], []
    for reference, process, raw in starts:
        if process["role"] not in ("spawn-child", "work-child", "fork-guard"):
            continue
        launch_ref = raw["context"]["lifecycle"][0]
        launch = load(launch_ref)
        launches.append(launch_ref)
        claims.extend(raw["owner_claim"])
        end_ref, end = next((r, p) for r, p in by_ref.items() if p["process_id"] == process["process_id"] and p["phase"] == "end")
        require(raw["owner_claim"] == raws[end_ref]["owner_claim"], "owner start/end claim")
        parent = by_ref[key(launch["parent_process"])]
        parent_end = next(p for p in by_ref.values() if p["process_id"] == parent["process_id"] and p["phase"] == "end")
        require(load(end["source_witness"], "source-witness")["at"] <= parent_end["at"], "owner source after parent end")
        require(end["at"] <= parent_end["at"] and end["monotonic_ns"] <= parent_end["monotonic_ns"], "owner parent lifetime")
    for prefix, used in (("owner-launch-", launches), ("owner-claim-", claims)):
        actual = [
            r for r in result["inventory"] if Path(r["path"]).parent == Path(job["output"]) and Path(r["path"]).name.startswith(prefix)
        ]
        same_set(used, actual, "owner launch/claim inventory")


def owner_sampler(process_ref, process, raw, launch_ref, launch, spec, job, candidate, dependencies, prior=0):
    policy_ref = launch["sampler_policy"][0]
    require(policy_ref in job["prerequisites"], "sampler policy not sealed prerequisite")
    policies = [r for r in job["prerequisites"] if load(r).get("schema") == "bac-hardware-owner-sampler-policy-v1"]
    require(policies == [policy_ref], "sampler unique policy")
    policy = load(policy_ref)
    fixed = {
        "owner_deadline_ns": 10_000_000_000,
        "sampler_cutoff_ns": 8_000_000_000,
        "sampler_total_ns": 6_000_000_000,
        "phase_ns": 3_000_000_000,
        "lock_wait_ns": 250_000_000,
        "command_run_ns": 500_000_000,
        "term_grace_ns": 100_000_000,
        "kill_reap_ns": 400_000_000,
        "emission_reserve_ns": 250_000_000,
        "stdout_limit_bytes": 8192,
        "stderr_limit_bytes": 8192,
    }
    exact(policy, "schema role_ids max_input_bytes cpu_slots scratch_bytes scratch_inodes " + " ".join(fixed), "sampler policy")
    for field in set(policy) - {"schema", "role_ids"}:
        require(uint(policy[field], "sampler policy") > 0, "sampler positive bound")
    require(all(policy[k] == v for k, v in fixed.items()), "sampler policy fixed constants")
    require(type(policy["role_ids"]) is list and process["role_id"] in policy["role_ids"], "sampler policy role")
    unique(policy["role_ids"], "sampler policy roles")
    role = next(r for r in job["roles"] if r["id"] == process["role_id"])
    require(
        policy["cpu_slots"] <= role["peak_cpu_slots"]
        and policy["scratch_bytes"] <= job["budget"]["scratch_bytes"]
        and policy["scratch_inodes"] <= job["budget"]["scratch_inodes"],
        "sampler resource grant",
    )
    for field in ("owner_deadline_ns", "sampler_cutoff_ns"):
        require(uint(launch[field], "sampler absolute deadline") == launch["monotonic_ns"] + policy[field], "sampler absolute deadline")
    sampler_ref = raw["sampler"][0]
    prefix = "sampler-" + process["process_id"] + "-" + process["phase"]
    require(sampler_ref["path"] == str(Path(job["output"]) / (prefix + ".json")), "sampler path")
    sampler = load(sampler_ref)
    exact(
        sampler, "schema policy launch identity phase start_ns end_ns cumulative_ns lock_acquired_ns lock_released_ns commands", "sampler"
    )
    require(
        sampler["schema"] == "bac-hardware-owner-sampler-observation-v1"
        and sampler["policy"] == policy_ref
        and sampler["launch"] == launch_ref
        and sampler["identity"] == process["identity"]
        and sampler["phase"] == process["phase"],
        "sampler binding",
    )
    for name in ("start_ns", "end_ns", "cumulative_ns", "lock_acquired_ns", "lock_released_ns"):
        uint(sampler[name], "sampler clock")
    require(
        sampler["start_ns"] == process["monotonic_ns"]
        and sampler["start_ns"]
        <= sampler["lock_acquired_ns"]
        <= sampler["lock_released_ns"]
        <= sampler["end_ns"]
        <= launch["sampler_cutoff_ns"],
        "sampler source chronology",
    )
    require(
        sampler["end_ns"] - sampler["start_ns"] <= policy["phase_ns"]
        and sampler["cumulative_ns"] == prior + sampler["end_ns"] - sampler["start_ns"] <= policy["sampler_total_ns"],
        "sampler source duration",
    )
    source = load(process["source_witness"], "source-witness")
    require(source["at"] == raw["at"] == process["at"], "fork source UTC")
    claim = load(raw["owner_claim"][0])
    require(
        claim["monotonic_ns"] <= sampler["end_ns"] and (process["phase"] != "start" or sampler["start_ns"] <= claim["monotonic_ns"]),
        "fork claim chronology",
    )
    git = [row["reference"] for row in dependencies["dependencies"] if row["name"] == "git"]
    require(len(git) == 1, "sampler admitted Git")
    read_bytes(git[0])
    argv = [
        [git[0]["path"], "-C", spec["authorized_root"], "rev-parse", "HEAD"],
        [git[0]["path"], "-C", spec["authorized_root"], "diff-index", "--cached", "--quiet", candidate["tree"], "--"],
    ]
    require(type(sampler["commands"]) is list and len(sampler["commands"]) == 2, "sampler two commands")
    members, previous = [sampler_ref], sampler["lock_acquired_ns"]
    for index, command in enumerate(sampler["commands"]):
        exact(command, "argv pid start_ns end_ns exit_code stdout stderr stdout_truncated stderr_truncated", "sampler command")
        require(
            command["argv"] == argv[index]
            and uint(command["pid"], "sampler Git pid") > 0
            and type(command["exit_code"]) is int
            and command["exit_code"] == 0,
            "sampler command result",
        )
        require(
            previous
            <= uint(command["start_ns"], "sampler command")
            <= uint(command["end_ns"], "sampler command")
            <= sampler["lock_released_ns"],
            "sampler command order",
        )
        previous = command["end_ns"]
        for stream in ("stdout", "stderr"):
            require(
                command[stream + "_truncated"] is False
                and command[stream]["path"] == str(Path(job["output"]) / f"{prefix}-{index}.{stream}"),
                "sampler log binding",
            )
            content = read_bytes(command[stream])
            require(len(content) <= policy[stream + "_limit_bytes"], "sampler log cap")
            if index == 0 and stream == "stdout":
                require(content.decode().strip() == candidate["head"], "sampler observed HEAD")
            members.append(command[stream])
        diagnostic = ref(Path(job["output"]) / f"{prefix}-{index}.json")
        require(key(load(diagnostic)) == key(command), "sampler diagnostic correspondence")
        members.append(diagnostic)
    emission_ref = ref(Path(job["output"]) / ("sampler-emission-" + process["process_id"] + "-" + process["phase"] + ".json"))
    emission = load(emission_ref)
    exact(emission, "schema policy launch identity phase sampler raw process emission_end_ns module_names_after_sha256", "sampler emission")
    require(
        emission["schema"] == "bac-hardware-owner-sampler-emission-v1"
        and emission["policy"] == policy_ref
        and emission["launch"] == launch_ref
        and emission["identity"] == process["identity"]
        and emission["phase"] == process["phase"]
        and emission["sampler"] == sampler_ref
        and emission["raw"] == process["debug"]["provenance"]
        and emission["process"] == process_ref
        and emission["module_names_after_sha256"] == launch["module_names_sha256"]
        and sampler["end_ns"] <= uint(emission["emission_end_ns"], "sampler emission") <= launch["sampler_cutoff_ns"],
        "sampler emission correspondence",
    )
    return (process_ref, process, emission_ref, emission, sampler), members + [emission_ref]


def sampler_inputs(raw, launch_ref, launch, spec, job, candidate, dependencies, prior_refs):
    """P3's bounded metadata closure; opaque leaves are hashed, never parsed."""
    freeze = load(spec["source_freeze"], "source-freeze")
    grant = load(freeze["grant"])
    role = next(r for r in job["roles"] if r["id"] == raw["context"]["role_id"])
    policy = load(launch["sampler_policy"][0])
    context = dict(raw["context"], lifecycle=[])
    objects = [
        context,
        spec,
        job,
        candidate,
        dependencies,
        launch["owner_contract"],
        load(launch["owner_contract"], "case-contract"),
        load(role["cpu_reservation_evidence"]),
        policy,
        grant,
        freeze,
    ]
    cursor, seen = context, set()
    while True:
        for field in ("policy", "runtime", "lifecycle"):
            for reference in cursor[field]:
                value = load(reference)
                objects += [reference, value]
                if value.get("schema") == "bac-hardware-identity-runtime-v1":
                    objects.append(load(value["runner_source_start"], "source-witness"))
        if not cursor["parent_start"]:
            break
        parent_ref = cursor["parent_start"][0]
        parent = load(parent_ref, "process")
        require(parent["process_id"] not in seen, "sampler ancestry cycle")
        seen.add(parent["process_id"])
        parent_raw = load(parent["debug"]["provenance"])
        objects += [parent_ref, parent, parent_raw, load(parent["source_witness"], "source-witness")]
        cursor = parent_raw["context"]
    references = {}

    def collect(value):
        if type(value) is dict:
            if set(value) == {"path", "bytes", "sha256"}:
                require(value["path"] not in references or references[value["path"]] == value, "sampler conflicting input")
                references[value["path"]] = value
            else:
                for child in value.values():
                    collect(child)
        elif type(value) is list:
            for child in value:
                collect(child)

    collect(objects)
    require(sum(r["bytes"] for r in references.values()) <= policy["max_input_bytes"], "sampler prepared input cap")
    collect([launch_ref, *prior_refs])
    require(sum(r["bytes"] for r in references.values()) <= policy["max_input_bytes"], "sampler phase input cap")
    for reference in references.values():
        read_bytes(reference)


def owner_completion(completion, result, by_ref, raws, spec, job, candidate, dependencies):
    """Producer snapshot custody precedes the independent strict semantic gate."""
    members, sources, fork_rows, used_forks = [], [], {}, set()
    for refkey, process in by_ref.items():
        reference, raw = json.loads(refkey), raws[refkey]
        members += [reference, process["debug"]["provenance"], process["source_witness"]]
        sources.append(process["source_witness"])
        if process["role"] not in ("spawn-child", "work-child", "fork-guard"):
            continue
        launch_ref = raw["context"]["lifecycle"][0]
        launch = load(launch_ref)
        members += [launch_ref, launch["source_witness"], *raw["owner_claim"]]
        sources.append(launch["source_witness"])
        if process["role"] == "fork-guard" and process["phase"] == "start":
            pair = [(reference, process, raw)] + [
                (json.loads(k), v, raws[k]) for k, v in by_ref.items() if v["process_id"] == process["process_id"] and v["phase"] == "end"
            ]
            rows, prior, prior_refs = [], 0, []
            for pr, pv, rv in pair:
                sampler_inputs(rv, launch_ref, launch, spec, job, candidate, dependencies, prior_refs)
                row, artifacts = owner_sampler(pr, pv, rv, launch_ref, launch, spec, job, candidate, dependencies, prior)
                rows.append(row)
                members += artifacts
                prior = row[4]["cumulative_ns"]
                prior_refs += [pr, pv["source_witness"], pv["debug"]["provenance"], *rv["owner_claim"]]
                prior_refs += [r for r in artifacts if not re.search(r"-[01]\.json$", r["path"])]
            fork_rows[key(launch_ref)] = (launch, rows)
    for reference in completion["obligation_records"]:
        members.append(reference)
        evidence = load(reference, "obligation-evidence")
        payload_ref = evidence["payload"]
        members.append(payload_ref)
        payload = load(payload_ref)
        if payload.get("schema") == "bac-hardware-producer-result-v1":
            outcome_ref = payload["acknowledgement"]["outcome"]
            outcome = load(outcome_ref)
            members += [outcome_ref, outcome["preflight"], payload["parent_baseline"]]
        if payload.get("schema") == "bac-hardware-replay-result-v1":
            members.append(payload["acknowledgement"]["result"])
        if payload.get("schema") != "bac-hardware-core-result-v1":
            continue
        child_ref = payload["acknowledgement"]["result"]
        members.append(child_ref)
        child = load(child_ref)
        for observation_ref in child.get("forks", []):
            observation = load(observation_ref)
            terminal_ref = observation["acknowledgement"]["result"]
            members += [observation_ref, terminal_ref]
            terminal, custody = load(terminal_ref), observation["parent_custody"]
            launch_key = key(observation["owner_launch"])
            require(launch_key in fork_rows, "fork payload missing observed owner")
            used_forks.add(launch_key)
            launch, rows = fork_rows[launch_key]
            exact(
                custody, "launch child_identity terminal_payload acknowledgement_received_ns reaped_ns wait_status", "fork parent custody"
            )
            require(
                custody["launch"] == observation["owner_launch"]
                and custody["terminal_payload"] == terminal_ref
                and custody["child_identity"] == rows[0][1]["identity"] == rows[1][1]["identity"]
                and observation["parent_start"] == launch["parent_process"]
                and observation["acknowledgement"]["process_start"] == rows[0][0]
                and observation["acknowledgement"]["process_end"] == rows[1][0],
                "fork terminal custody correspondence",
            )
            require(
                all(type(v) is int and v == 0 for v in (custody["wait_status"], observation["wait_status"], observation["exit_code"])),
                "fork wait status",
            )
            ident = rows[0][1]["identity"]
            require(
                observation["fork_pid"] == observation["waitpid_pid"] == observation["acknowledgement"]["pid"] == ident["pid"]
                and observation["acknowledgement"]["ppid"] == ident["ppid"],
                "fork actual wait identity",
            )
            fork_call_timing(launch, rows, terminal, custody)
    require(used_forks == set(fork_rows), "fork observed owner missing terminal custody")
    for reference in members:
        require(
            reference in completion["inventory"] and Path(reference["path"]).is_relative_to(Path(job["output"])),
            "completion inventory missing owner artifact",
        )
    require(all(reference in completion["source_witnesses"] for reference in sources), "completion missing owner source witness")


def fork_call_timing(launch, rows, terminal, custody):
    returns = terminal["recorder_returns"]
    require(type(returns) is list and len(returns) == len(rows) == 2, "fork requires both actual call returns")
    total, previous = 0, launch["monotonic_ns"]
    for phase, returned, row in zip(("start", "end"), returns, rows):
        process_ref, process, emission_ref, emission, sampler = row
        exact(returned, "phase process emission entry_ns return_ns", "fork return")
        require(
            returned["phase"] == phase
            and returned["process"] == process_ref
            and returned["emission"] == emission_ref
            and emission["process"] == process_ref
            and emission["identity"] == process["identity"]
            and emission["phase"] == phase,
            "fork return/emission correspondence",
        )
        entry, end = uint(returned["entry_ns"], "fork call entry"), uint(returned["return_ns"], "fork call return")
        require(
            previous
            <= entry
            <= uint(sampler["start_ns"], "fork sampler")
            <= uint(sampler["end_ns"], "fork sampler")
            <= uint(emission["emission_end_ns"], "fork emission")
            <= end
            <= launch["sampler_cutoff_ns"],
            "fork full call chronology/cutoff",
        )
        require(end - entry <= 3_000_000_000, "fork complete call exceeds3s")
        total += end - entry
        previous = end
    require(total <= 6_000_000_000, "fork complete calls exceed6s")
    for field in ("acknowledgement_received_ns", "reaped_ns"):
        require(
            previous <= uint(custody[field], "fork parent endpoint") <= launch["owner_deadline_ns"],
            "fork parent endpoint outside returns/deadline",
        )


def _lifecycles(starts, by_ref, job, result):
    workers = [row for row in starts if row[1]["role"] == "worker"]
    if workers:
        first_identity = workers[0][2]["context"]["worker_identity"]
        gateways = {}
        for reference, _process, raw in sorted(workers, key=lambda row: row[1]["monotonic_ns"]):
            context = raw["context"]
            worker = context["worker_identity"]
            require(worker["testrunuid"] == first_identity["testrunuid"], "worker run UID differs across job")
            require(worker["mainargv"] == first_identity["mainargv"], "worker main argv differs across job")
            gateway = worker["workerid"]
            if gateway in gateways:
                require(context["replacement_start"] == [gateways[gateway]], "worker gateway reused without its predecessor")
            gateways[gateway] = reference
    consumed = set()
    lifecycle_records = {}
    for reference in result["inventory"]:
        if not reference["path"].endswith(".json"):
            continue
        item = load(reference)
        if type(item) is dict and item.get("schema") == "bac-hardware-worker-lifecycle-v1":
            exact(
                item,
                "schema phase controller_start worker_identity role_id predecessor_start process_start error at prior_lifecycle",
                "lifecycle",
            )
            require(item["phase"] in ("configure", "down") and type(item["error"]) is str, "worker lifecycle types")
            _worker(item["worker_identity"], job)
            for field in ("predecessor_start", "process_start", "prior_lifecycle"):
                require(type(item[field]) is list and len(item[field]) <= 1, "worker lifecycle reference cardinality")
                for child in item[field]:
                    protocol.validate_record("reference", child)
            lifecycle_records[key(reference)] = item
    used_lifecycles = set()
    for reference, process, raw in starts:
        if process["role"] != "worker":
            continue
        context = raw["context"]
        parent_ref = context["parent_start"][0]
        require(key(parent_ref) in by_ref, "worker parent not observed")
        parent = by_ref[key(parent_ref)]
        require(parent["role"] == "controller" and parent["phase"] == "start", "worker controller parent")
        require(
            process["parent_identity"] == [parent["identity"]] and process["identity"]["ppid"] == parent["identity"]["pid"],
            "worker actual parent",
        )
        configure_ref = context["lifecycle"][0]
        require(key(configure_ref) in lifecycle_records, "worker configure not inventoried")
        used_lifecycles.add(key(configure_ref))
        configure = load(configure_ref)
        downs = [(r, d) for r, d in lifecycle_records.items() if d["phase"] == "down" and d["process_start"] == [reference]]
        require(len(downs) == 1, "worker missing/duplicate down lifecycle")
        down_ref, down = downs[0]
        used_lifecycles.add(down_ref)
        require(
            down["controller_start"] == parent_ref
            and down["role_id"] == process["role_id"]
            and down["worker_identity"] == context["worker_identity"]
            and down["prior_lifecycle"] == [configure_ref]
            and down["predecessor_start"] == context["replacement_start"],
            "worker down lifecycle binding",
        )
        ends = [v for v in by_ref.values() if v["process_id"] == process["process_id"] and v["phase"] == "end"]
        parent_ends = [v for v in by_ref.values() if v["process_id"] == parent["process_id"] and v["phase"] == "end"]
        require(
            len(ends) == len(parent_ends) == 1 and process["at"] <= ends[0]["at"] <= down["at"] <= parent_ends[0]["at"],
            "worker down lifecycle chronology",
        )
        require(
            parent["monotonic_ns"] <= process["monotonic_ns"] <= ends[0]["monotonic_ns"] <= parent_ends[0]["monotonic_ns"],
            "worker monotonic lifetime outside controller interval",
        )
        require(all(process["identity"][k] == parent["identity"][k] for k in ("host", "boot_id")), "worker parent host/boot")
        exact(
            configure,
            "schema phase controller_start worker_identity role_id predecessor_start process_start error at prior_lifecycle",
            "lifecycle",
        )
        require(
            configure["schema"] == "bac-hardware-worker-lifecycle-v1"
            and configure["phase"] == "configure"
            and configure["controller_start"] == parent_ref
            and configure["worker_identity"] == context["worker_identity"]
            and configure["role_id"] == process["role_id"]
            and not configure["process_start"]
            and configure["error"] == ""
            and configure["predecessor_start"] == context["replacement_start"],
            "worker configure binding",
        )
        require(parent["at"] <= configure["at"] <= process["at"], "lifecycle chronology")
        if context["replacement_start"]:
            require(len(configure["prior_lifecycle"]) == 1, "replacement prior lifecycle")
            prior_ref = configure["prior_lifecycle"][0]
            predecessor_ref = context["replacement_start"][0]
            require(key(predecessor_ref) in by_ref, "replacement predecessor missing")
            predecessor = by_ref[key(predecessor_ref)]
            predecessor_downs = [
                (refkey, row)
                for refkey, row in lifecycle_records.items()
                if row["phase"] == "down" and row["process_start"] == [predecessor_ref]
            ]
            require(
                len(predecessor_downs) == 1 and key(prior_ref) == predecessor_downs[0][0],
                "replacement prior is not exact inventoried predecessor down",
            )
            require(key(predecessor_ref) not in consumed, "replacement predecessor reused")
            consumed.add(key(predecessor_ref))
            prior = predecessor_downs[0][1]
            require(
                prior["schema"] == "bac-hardware-worker-lifecycle-v1"
                and prior["phase"] == "down"
                and prior["controller_start"] == parent_ref
                and prior["role_id"] == process["role_id"]
                and prior["process_start"] == [predecessor_ref]
                and type(prior["error"]) is str
                and bool(prior["error"]),
                "replacement down binding",
            )
            require(
                predecessor["phase"] == "start"
                and predecessor["role_id"] == process["role_id"]
                and predecessor["identity"] != process["identity"]
                and process["replacement_of"] == [predecessor["identity"]],
                "replacement identity",
            )
            require(predecessor["at"] <= prior["at"] <= configure["at"], "replacement ordering")
            predecessor_end = next(
                row for row in by_ref.values() if row["process_id"] == predecessor["process_id"] and row["phase"] == "end"
            )
            require(predecessor_end["monotonic_ns"] <= process["monotonic_ns"], "replacement monotonic overlap")
            require(len(prior["prior_lifecycle"]) == 1, "down missing configure link")
            old = load(prior["prior_lifecycle"][0])
            require(
                old["phase"] == "configure"
                and old["worker_identity"] == prior["worker_identity"]
                and old["controller_start"] == parent_ref,
                "down configure ancestry",
            )
        else:
            require(not configure["prior_lifecycle"] and not process["replacement_of"], "initial worker has predecessor")
    require(used_lifecycles == set(lifecycle_records), "orphan worker lifecycle")


def outcomes(result, job, processes_by_ref):
    required = set(job["required_collected_ids"])
    reasons = []
    if set(result["collected_ids"]) != required:
        reasons.append("required collection set mismatch")
    if {o["case_id"] for o in result["outcomes"]} != required:
        reasons.append("required outcomes missing; collection-only is unverified")
    for outcome in result["outcomes"]:
        phases = {}
        for reference in outcome["evidence"]:
            report = load(reference)
            exact(report, "schema process case_id phase outcome longrepr at", "phase report")
            require(
                report["schema"] == "bac-hardware-pytest-report-v1"
                and report["case_id"] == outcome["case_id"]
                and report["phase"] in PHASES,
                "report binding",
            )
            require(report["phase"] not in phases, "duplicate report phase")
            require(key(report["process"]) in processes_by_ref, "report process not observed")
            process = processes_by_ref[key(report["process"])]
            require(process["phase"] == "start" and process["at"] <= report["at"], "report process lifecycle")
            ends = [v for v in processes_by_ref.values() if v["process_id"] == process["process_id"] and v["phase"] == "end"]
            require(len(ends) == 1 and report["at"] <= ends[0]["at"], "report outside process lifecycle")
            require(report["outcome"] == outcome[report["phase"]], "report/outcome mismatch")
            require(type(report["longrepr"]) is str, "report longrepr")
            phases[report["phase"]] = report
        for phase in PHASES:
            if outcome[phase] != "not-run":
                require(phase in phases, "phase report missing")
            else:
                require(phase not in phases, "not-run phase has report")
        ordered = [phases[phase] for phase in PHASES if phase in phases]
        require(len({key(r["process"]) for r in ordered}) <= 1, "case phase reports came from different processes")
        require([r["at"] for r in ordered] == sorted(r["at"] for r in ordered), "report phase chronology")
        if not all(outcome[phase] == "passed" for phase in PHASES):
            reasons.append("required case not all passed: " + outcome["case_id"])
    return reasons


def _inventory_exact(references, root, excluded):
    unique(references, "file inventory")
    seen = []
    for path in sorted(Path(root).rglob("*")):
        require(not path.is_symlink(), "symlink in sealed output")
        if path.is_file() and str(path) not in excluded:
            seen.append(ref(path))
        else:
            require(path.is_dir() or str(path) in excluded, "nonregular sealed output")
    same_set(references, seen, "sealed file inventory")


def parent_observations(result, spec_ref, spec, job, command_ref):
    launch_ref = ref(Path(job["output"]) / "launch.json")
    launch = load(launch_ref)
    exact(launch, "schema attempt_id job_id spec command identity affinity at monotonic_ns inherited_leases", "launch")
    require(launch["schema"] == "bac-hardware-runner-launch-v1", "launch schema")
    bound(launch, spec_ref, spec, job)
    require(launch["command"] == command_ref, "launch command")
    protocol.validate_record("identity", launch["identity"])
    require(
        type(launch["affinity"]) is list and bool(launch["affinity"]) and all(type(v) is int and v >= 0 for v in launch["affinity"]),
        "launch affinity",
    )
    require(launch["affinity"] == sorted(set(launch["affinity"])), "launch affinity duplicates/order")
    require(type(launch["monotonic_ns"]) is int and launch["monotonic_ns"] >= 0, "launch monotonic")
    require(type(launch["inherited_leases"]) is list and launch["inherited_leases"], "launch leases absent")
    for lease in launch["inherited_leases"]:
        exact(lease, "fd lock_name device inode", "inherited lease")
        require(
            type(lease["fd"]) is int
            and lease["fd"] > 2
            and type(lease["device"]) is int
            and lease["device"] >= 0
            and type(lease["inode"]) is int
            and lease["inode"] >= 0,
            "lease identity types",
        )
        require(type(lease["lock_name"]) is str and re.fullmatch(r"[0-9a-f]{64}\.lock", lease["lock_name"]), "lease lock name")
    unique([l["fd"] for l in launch["inherited_leases"]], "inherited descriptor")
    unique([l["lock_name"] for l in launch["inherited_leases"]], "inherited lock")
    waited = load(ref(Path(job["output"]) / "wait.json"))
    exact(
        waited,
        "schema attempt_id job_id spec command launch identity exit_code signal interruption_reason start end start_monotonic_ns end_monotonic_ns",
        "wait",
    )
    require(waited["schema"] == "bac-hardware-runner-wait-v1", "wait schema")
    bound(waited, spec_ref, spec, job)
    require(
        waited["command"] == command_ref and waited["launch"] == launch_ref and waited["identity"] == launch["identity"],
        "wait launch/identity binding",
    )
    require(
        type(waited["exit_code"]) is int and type(waited["signal"]) is int and type(waited["interruption_reason"]) is str, "wait exit types"
    )
    require(waited["exit_code"] == result["exit_code"] and waited["signal"] == result["signal"], "wait/result exit mismatch")
    require(waited["signal"] == max(0, -waited["exit_code"]), "wait signal mismatch")
    for field in ("start_monotonic_ns", "end_monotonic_ns"):
        require(type(waited[field]) is int and waited[field] >= 0, "wait monotonic type")
    require(waited["start"] == result["start"] and waited["end"] == result["end"]["value"], "wait absolute times")
    require(
        waited["start_monotonic_ns"] <= launch["monotonic_ns"] <= waited["end_monotonic_ns"]
        and result["elapsed_ns"]["value"] == waited["end_monotonic_ns"] - waited["start_monotonic_ns"],
        "wait elapsed mismatch",
    )
    require(waited["start"] <= launch["at"] <= waited["end"], "wait chronology")
    if waited["signal"] or waited["interruption_reason"]:
        require(result["disposition"] == "interrupted", "interrupted wait relabeled")
    elif waited["exit_code"]:
        require(result["disposition"] == "completed-failed", "nonzero wait relabeled")
    else:
        require(result["disposition"] in ("successful", "completed-unverified"), "zero wait relabeled")
    return launch, waited


def collections(result, job, by_ref):
    rows = []
    report_refs = []
    for reference in result["inventory"]:
        if not reference["path"].endswith(".json"):
            continue
        value = load(reference)
        if type(value) is dict and value.get("schema") == "bac-hardware-pytest-report-v1":
            report_refs.append(reference)
        if type(value) is dict and value.get("schema") == "bac-hardware-pytest-collection-v1":
            exact(value, "schema process ids", "collection")
            require(key(value["process"]) in by_ref and by_ref[key(value["process"])]["phase"] == "start", "collection process")
            require(type(value["ids"]) is list and all(type(x) is str and x for x in value["ids"]), "collection IDs")
            same_set(value["ids"], result["collected_ids"], "actual collection/result")
            rows.append(value)
    expected = [
        refkey
        for refkey, process in by_ref.items()
        if process["phase"] == "start" and process["role"] == ("worker" if job["xdist_workers"] else "controller")
    ]
    same_set([key(row["process"]) for row in rows], expected, "actual collection processes")
    same_set(report_refs, [r for o in result["outcomes"] for r in o["evidence"]], "actual phase report inventory")


def validator_inventory(job, tools):
    inventories = []
    for reference in job["prerequisites"]:
        value = load(reference)
        if type(value) is dict and value.get("schema") == "bac-hardware-validator-inventory-v1":
            inventories.append((reference, value))
    require(len(inventories) <= 1, "duplicate validator inventory")
    if not inventories:
        return None
    reference, inventory = inventories[0]
    exact(inventory, "schema authority validators", "validator inventory")
    require(type(inventory["authority"]) is str and inventory["authority"].strip(), "inventory authority statement")
    require(type(inventory["validators"]) is list and 1 <= len(inventory["validators"]) <= 2, "finite validator inventory")
    entries = {}
    for entry in inventory["validators"]:
        exact(entry, "id kinds validator dependencies contracts witness_schemas evidence_classes", "validator entry")
        name = entry["id"]
        require(
            name in ("p4-common-v1", "p5-strict-v1") and name not in entries, "unknown/duplicate validator ID; arbitrary dispatch forbidden"
        )
        same_set(entry["kinds"], list(COMMON if name == "p4-common-v1" else STRICT), "validator kinds")
        expected = tools["verifier"] if name == "p4-common-v1" else ref(Path(__file__).with_name("verify_bac_hardware_children_v1.py"))
        require(entry["validator"] == expected and entry["dependencies"] == [tools["protocol"]], "validator source/dependency pins")
        require(tools["protocol"]["sha256"] == PROTOCOL_SHA256, "validator protocol literal pin")
        require(name == "p4-common-v1" or expected["sha256"] == P5_SHA256, "validator P5 literal pin")
        require(entry["evidence_classes"] == ["cpu", "cpu-standin"], "validator evidence classes")
        for field in ("contracts", "witness_schemas"):
            unique(entry[field], "validator " + field)
            require(entry[field], "missing validator " + field)
            for item in entry[field]:
                protocol.validate_record("reference", item)
        entries[name] = entry
    return entries


def common_validation(
    obligation, contract, evidence, entry, result, by_ref, spec_ref, job, validation_identity, validation_start, outcome_reasons
):
    require(obligation["kind"] in COMMON, "unknown built-in common validator kind")
    require(load(contract["witness_schema"]) == {"schema": "bac-hardware-common-observation-v1"}, "common witness schema declaration")
    require(
        obligation["contract"] in entry["contracts"] and contract["witness_schema"] in entry["witness_schemas"],
        "common contract/schema not authorized",
    )
    require(
        obligation["validator"] == entry["validator"]
        and obligation["evidence_class"] in entry["evidence_classes"]
        and obligation["minimum_devices"] == 0,
        "common validator binding/class",
    )
    require(len(evidence) == len(obligation["required_cases"]), "common evidence case inventory")
    outcomes_by_case = {o["case_id"]: o for o in result["outcomes"]}
    payloads = []
    producers = []
    for reference in evidence:
        item = load(reference, "obligation-evidence")
        require(item["case_id"] in outcomes_by_case, "common outcome absent")
        outcome = outcomes_by_case[item["case_id"]]
        payload = load(item["payload"])
        exact(payload, "schema kind case_id reports processes source_witnesses", "common payload")
        require(
            payload["schema"] == "bac-hardware-common-observation-v1"
            and payload["kind"] == obligation["kind"]
            and payload["case_id"] == item["case_id"],
            "common payload binding",
        )
        same_set(payload["reports"], outcome["evidence"], "common payload reports")
        same_set(payload["processes"], result["process_records"], "common payload processes")
        same_set(payload["source_witnesses"], result["source_witnesses"], "common payload source witnesses")
        report_producers = {key(load(r)["process"]) for r in payload["reports"]}
        require(key(item["producer_process"]) in report_producers, "common producer did not execute case reports")
        require(all(by_ref[key(load(r)["process"])]["phase"] == "start" for r in payload["reports"]), "common report lifecycle")
        require(not item["related_attempts"], "common unrelated attempt")
        payloads.append(item["payload"])
        if item["producer_process"] not in producers:
            producers.append(item["producer_process"])
    verdict = "unsatisfied" if outcome_reasons or result["exit_code"] != 0 or result["signal"] != 0 else "satisfied"
    reasons = list(outcome_reasons)
    if result["exit_code"] != 0 or result["signal"] != 0:
        reasons.append("execution did not complete with ordinary zero exit")
    return {
        "schema": "bac-hardware-case-validator-receipt-v1",
        "validator_id": "p4-common-v1",
        "validator": entry["validator"],
        "dependencies": entry["dependencies"],
        "spec": spec_ref,
        "job_id": job["id"],
        "obligation_id": obligation["id"],
        "contract": obligation["contract"],
        "witness_schema": contract["witness_schema"],
        "evidence_class": obligation["evidence_class"],
        "cases": obligation["required_cases"],
        "evidence": evidence,
        "payloads": payloads,
        "producer_processes": producers,
        "related_attempts": [],
        "validation_process": validation_identity,
        "start": validation_start,
        "end": utc(),
        "verdict": verdict,
        "reasons": reasons,
    }


def private_leaves(value):
    """Explicit transport edges only: no path-shaped dictionary discovery."""
    schema = value.get("schema", "")
    fields = {
        "bac-hardware-owner-launch-v1": "spec command owner_contract parent_process source_witness",
        "bac-hardware-owner-claim-v1": "launch",
        "bac-hardware-owner-sampler-observation-v1": "policy launch",
        "bac-hardware-owner-sampler-emission-v1": "policy launch sampler raw process",
        "bac-hardware-strict-child-receipt-v1": "spec validator contract witness_schema",
    }
    refs = [value[name] for name in fields.get(schema, "").split()]
    if schema == "bac-hardware-process-observation-v2":
        context = value["context"]
        refs += [context[name] for name in ("spec", "command", "trusted_grant")]
        refs += [r["reference"] for r in context["required_imports"]]
        refs += [r for name in ("parent_start", "replacement_start", "lifecycle", "policy", "runtime") for r in context[name]]
        refs += value["owner_claim"] + value["sampler"]
    if schema == "bac-hardware-owner-launch-v1":
        refs += value["sampler_policy"] + [r["reference"] for r in value["inherited_imports"]]
    if schema == "bac-hardware-owner-sampler-observation-v1":
        refs += [row[name] for row in value["commands"] for name in ("stdout", "stderr")]
    if schema == "bac-hardware-strict-child-receipt-v1":
        refs += value["validator_dependencies"] + value["documents"]
        refs += [r for name in ("evidence", "payloads", "producer_processes", "related_attempts") for r in value["result"][name]]
    for reference in refs:
        read_bytes(reference)


def strict_subject(spec_ref, job_id, obligation_id, *, nested=False):
    """Collect the finite public consumed graph, never private P5 implementation."""
    documents = {}

    def take(reference, kind=None):
        prior = documents.get(reference["path"])
        require(prior is None or prior["reference"] == reference, "strict conflicting document reference")
        content = read_bytes(reference)
        value = decode(content)
        if kind:
            protocol.validate_record(kind, value)
        private_leaves(value)
        documents[reference["path"]] = {"reference": reference, "content": content}
        return value

    spec = take(spec_ref, "spec")
    jobs = [job for job in spec["jobs"] if job["id"] == job_id]
    require(len(jobs) == 1, "strict unknown job")
    job = jobs[0]
    if nested:
        require(len(spec["jobs"]) == 1 and all(o["kind"] != "phase-pair" for o in job["obligations"]), "nested phase pair")
    selected = [o for o in job["obligations"] if o["id"] == obligation_id]
    require(len(selected) == 1, "strict unknown obligation")
    obligation = selected[0]
    contract = take(obligation["contract"], "case-contract")
    take(contract["witness_schema"])
    take(spec["candidate"], "candidate")

    def process(reference):
        row = take(reference, "process")
        raw = take(row["debug"]["provenance"])
        take(row["source_witness"], "source-witness")
        for claim in raw["owner_claim"]:
            take(claim)
        for sampler in raw["sampler"]:
            take(sampler)
        return row

    def owner(payload):
        process(payload["parent_start"])
        launch = take(payload["owner_launch"])
        take(launch["source_witness"], "source-witness")
        for reference in launch["sampler_policy"]:
            take(reference)
        for name in ("process_start", "process_end"):
            process(payload["acknowledgement"][name])

    if obligation["kind"] in STRICT and obligation["evidence_class"] in ("cpu", "cpu-standin"):
        completion = take(ref(Path(job["output"]) / "completion.json"), "job-evidence")
        result = take(ref(Path(job["output"]) / "result.json"), "job-result")
        if result["disposition"] == "successful":
            for case in obligation["required_cases"]:
                outcomes = [row for row in completion["outcomes"] if row["case_id"] == case]
                require(len(outcomes) == 1, "strict selected outcome")
                for report in outcomes[0]["evidence"]:
                    take(report)
            evidence = [take(r, "obligation-evidence") for r in completion["obligation_records"]]
            selected_evidence = [row for row in evidence if row["obligation_id"] == obligation_id]
            for item in selected_evidence:
                payload = take(item["payload"])
                if obligation["kind"] == "phase-pair":
                    require(not nested, "nested phase pair")
                    for subject in payload["subjects"]:
                        receipt = take(subject["strict_receipt"])
                        inner = strict_subject(subject["spec"], subject["job_id"], receipt["obligation_id"], nested=True)
                        same_set(receipt["documents"], [r["reference"] for r in inner["documents"]], "phase receipt consumed documents")
                        for row in inner["documents"]:
                            take(row["reference"])
                    continue
                for reference in completion["process_records"]:
                    process(reference)
                family = "strict-child-producer" if payload.get("schema") == "bac-hardware-producer-result-v1" else "strict-child-core"
                for candidate in job["obligations"]:
                    if candidate["kind"] == family and item["case_id"] in candidate["required_cases"]:
                        take(candidate["contract"], "case-contract")
                owner(payload)
                acknowledgement = payload["acknowledgement"]
                if family == "strict-child-producer":
                    outcome = take(acknowledgement["outcome"])
                    take(outcome["preflight"])
                    take(payload["parent_baseline"])
                else:
                    child = take(acknowledgement["result"])
                    for reference in child.get("forks", []):
                        observation = take(reference)
                        owner(observation)
                        terminal = take(observation["acknowledgement"]["result"])
                        for returned in terminal["recorder_returns"]:
                            emission = take(returned["emission"])
                            take(emission["sampler"])
    for row in documents.values():
        read_bytes(row["reference"])
    return {
        "schema": "bac-hardware-strict-validation-subject-v1",
        "spec": spec_ref,
        "job_id": job_id,
        "obligation_id": obligation_id,
        "documents": list(documents.values()),
    }


def phase_common(spec_ref, job_id):
    initial_rows = initial(spec_ref["path"], job_id)
    actual, spec, job, tools, candidate, freeze, grant, dependencies, capture_ref, capture_files = initial_rows
    require(
        actual == spec_ref and len(spec["jobs"]) == 1 and all(o["kind"] != "phase-pair" for o in job["obligations"]), "phase common spec"
    )
    output = Path(job["output"])
    result_ref, completion_ref, command_ref = (ref(output / name) for name in ("result.json", "completion.json", "command.json"))
    result, completion, command = load(result_ref, "job-result"), load(completion_ref, "job-evidence"), load(command_ref, "command")
    for value in (result, completion, command):
        bound(value, actual, spec, job)
    require(
        result["disposition"] == "successful"
        and result["command"] == completion["command"] == command_ref
        and command["runner"] == tools["runner"],
        "phase common terminal",
    )
    for field in ("argv", "cwd", "environment", "seed"):
        require(command[field] == job[field], "phase common command")
    for field in ("collected_ids", "outcomes", "process_records", "obligation_records", "source_witnesses"):
        require(completion[field] == result[field], "phase common copied arrays")
    _inventory_exact(result["inventory"], output, {str(output / "result.json")})
    launch, waited = parent_observations(result, actual, spec, job, command_ref)
    by_ref, raws, runtime, _policy_record, _starts = processes(
        result, actual, spec, job, command_ref, candidate, freeze, grant, dependencies
    )
    controllers = [p for p in by_ref.values() if p["role"] == "controller" and p["phase"] == "start"]
    require(len(controllers) == 1 and controllers[0]["identity"] == launch["identity"], "phase controller launch")
    require(launch["identity"]["ppid"] == runtime["runner_identity"]["pid"], "phase runner ancestry")
    for process in by_ref.values():
        require(
            all(process["identity"][field] == runtime["runner_identity"][field] for field in ("host", "boot_id"))
            and waited["start_monotonic_ns"] <= process["monotonic_ns"] <= waited["end_monotonic_ns"],
            "phase process lifetime",
        )
    owner_completion(completion, result, by_ref, raws, spec, job, candidate, dependencies)
    collections(result, job, by_ref)
    require(not outcomes(result, job, by_ref), "phase required reports incomplete")
    start_ref, seal_ref = ref(output.parent / "start.json"), ref(output.parent / "seal.json")
    start, seal = load(start_ref, "attempt-start"), load(seal_ref, "attempt-seal")
    for row in (start, seal):
        bound(row, actual, spec)
    require(
        seal["start"] == start_ref
        and seal["job_results"] == [result_ref]
        and seal["disposition"] == "successful"
        and seal["source_freeze"] == start["source_freeze"] == spec["source_freeze"]
        and start["at"] <= command["at"] <= result["start"] <= result["end"]["value"] <= seal["at"],
        "phase attempt seal",
    )
    source_check(start["source_witness"], spec["candidate"], candidate, spec["source_freeze"], freeze, grant, runtime["runner_identity"])
    _inventory_exact([r for r in seal["inventory"] if r != capture_ref], output.parent, {str(output.parent / "seal.json")})
    require(capture_ref in seal["inventory"] and runtime["runner_source_start"] in seal["inventory"], "phase runner sealed sources")
    sources = [
        source_check(r, spec["candidate"], candidate, spec["source_freeze"], freeze, grant, runtime["runner_identity"])
        for r in seal["inventory"]
        if Path(r["path"]).parent == output.parent and re.fullmatch(r"source-[0-9]{4}\.json", Path(r["path"]).name)
    ]
    require(
        sources and min(row["at"] for row in sources) <= result["start"] and max(row["at"] for row in sources) >= result["end"]["value"],
        "phase runner pre/post sources",
    )
    for reference in [*capture_files, *result["inventory"], actual]:
        read_bytes(reference)


def strict_validation(obligation, contract, entry, spec_ref, job, validation_identity, validation_start):
    require(
        obligation["contract"] in entry["contracts"]
        and contract["witness_schema"] in entry["witness_schemas"]
        and obligation["validator"] == entry["validator"]
        and obligation["evidence_class"] in entry["evidence_classes"],
        "strict inventory binding",
    )
    expected = ref(Path(__file__).with_name("verify_bac_hardware_children_v1.py"))
    require(
        expected == entry["validator"]
        and expected["sha256"] == P5_SHA256
        and entry["dependencies"] == [ref(protocol.__file__)]
        and entry["dependencies"][0]["sha256"] == PROTOCOL_SHA256,
        "strict literal source pins",
    )
    import verify_bac_hardware_children_v1

    require(ref(verify_bac_hardware_children_v1.__file__) == expected, "strict imported source mismatch")
    subject = strict_subject(spec_ref, job["id"], obligation["id"])
    checked = verify_bac_hardware_children_v1.validate_child_obligation(subject)
    exact(checked, "schema obligation_id verdict reasons cases evidence payloads producer_processes related_attempts", "strict result")
    require(
        checked["schema"] == "bac-hardware-strict-validation-result-v1"
        and checked["obligation_id"] == obligation["id"]
        and checked["verdict"] in ("satisfied", "unsatisfied", "unrun"),
        "strict result binding",
    )
    if checked["verdict"] == "satisfied" and obligation["kind"] == "phase-pair":
        for reference in checked["evidence"]:
            wrapper = load(load(reference, "obligation-evidence")["payload"])
            for inner in wrapper["subjects"]:
                phase_common(inner["spec"], inner["job_id"])
    for row in subject["documents"]:
        value = decode(read_bytes(row["reference"]))
        if value.get("schema") == protocol.SCHEMA:
            protocol.validate_record(value["kind"], value)
        private_leaves(value)
    require(ref(verify_bac_hardware_children_v1.__file__) == expected, "strict source changed")
    return {
        "schema": "bac-hardware-case-validator-receipt-v1",
        "validator_id": "p5-strict-v1",
        "validator": expected,
        "dependencies": entry["dependencies"],
        "spec": spec_ref,
        "job_id": job["id"],
        "obligation_id": obligation["id"],
        "contract": obligation["contract"],
        "witness_schema": contract["witness_schema"],
        "evidence_class": obligation["evidence_class"],
        **{k: checked[k] for k in ("verdict", "reasons", "cases", "evidence", "payloads", "producer_processes", "related_attempts")},
        "validation_process": validation_identity,
        "start": validation_start,
        "end": utc(),
    }


def components(spec_ref, job, observed, obligations, validator_receipts):
    common = [o for o in obligations if next(j for j in job["obligations"] if j["id"] == o["id"])["kind"] in COMMON]
    strict_kinds = {"strict-child-core", "strict-child-producer", "rng-custody", "fork-three-trials", "phase-pair"}
    strict = [o for o in obligations if next(j for j in job["obligations"] if j["id"] == o["id"])["kind"] in strict_kinds]

    def state(rows):
        return (
            "satisfied"
            if rows and all(r["disposition"] == "satisfied" for r in rows)
            else "unsatisfied" if any(r["disposition"] == "unsatisfied" for r in rows) else "unrun"
        )

    hardware = [o for o in obligations if next(j for j in job["obligations"] if j["id"] == o["id"])["evidence_class"] == "real-cuda"]
    complete = all(o["disposition"] == "satisfied" for o in obligations) and observed == "successful"
    reason = (
        []
        if complete
        else [
            (
                "aggregate obligation gate incomplete; common source/process custody independently satisfied"
                if state(common) == "satisfied"
                else "common or required case obligations incomplete"
            )
        ]
    )
    return {
        "schema": "bac-hardware-component-classification-v1",
        "spec": spec_ref,
        "job_id": job["id"],
        "common_custody": state(common),
        "strict_custody": state(strict) if strict else "not-required",
        "test_outcome": observed,
        "hardware": state(hardware) if hardware else "unrun",
        "overall": "satisfied" if complete else "incomplete",
        "reasons": reason,
        "validator_receipts": validator_receipts,
    }


def verify(spec_path, job_id, destination):
    """Verify one exact root-authorized attempt; exclusive immutable receipt.

    Invalid evidence raises ValueError. Well-formed observed failures retain their
    terminal classification. Missing case validators never produce satisfaction.
    """
    require(type(spec_path) is str and type(job_id) is str and type(destination) is str, "string API arguments required")
    # Reject destination aliases before any artifact can be created.
    fds = _safe_parent(destination)
    for descriptor in reversed(fds):
        os.close(descriptor)
    require(not os.path.lexists(destination), "receipt destination exists")
    validation_start, validation_identity = utc(), identity()
    spec_ref, spec, job, tools, candidate, freeze, grant, dependencies, capture_ref, capture_files = initial(spec_path, job_id)
    accepted_validator = validator_inventory(job, tools)
    output = Path(job["output"])
    require(
        not Path(destination).is_relative_to(Path(spec["output_root"]))
        and not Path(destination).is_relative_to(Path(spec["authorized_root"])),
        "receipt inside sealed attempt/source",
    )
    result_ref = ref(output / "result.json")
    result = load(result_ref, "job-result")
    bound(result, spec_ref, spec, job)
    command_ref = ref(output / "command.json")
    command = load(command_ref, "command")
    bound(command, spec_ref, spec, job)
    require(result["command"] == command_ref and command["runner"] == tools["runner"], "command/runner reference")
    for field in ("argv", "cwd", "environment", "seed"):
        require(command[field] == job[field], "exact command " + field)
    require(result["log"] == ref(output / "stdout.log"), "log binding")
    start_ref = ref(output.parent / "start.json")
    start = load(start_ref, "attempt-start")
    bound(start, spec_ref, spec)
    require(start["source_freeze"] == spec["source_freeze"], "attempt freeze")
    seal_ref = ref(output.parent / "seal.json")
    seal = load(seal_ref, "attempt-seal")
    bound(seal, spec_ref, spec)
    require(
        seal["start"] == start_ref and seal["source_freeze"] == spec["source_freeze"] and result_ref in seal["job_results"], "seal binding"
    )
    require(seal["at"] >= start["at"] and command["at"] >= start["at"] and result["start"] >= command["at"], "attempt chronology")
    if result["end"]["state"] == "measured":
        require(seal["at"] >= result["end"]["value"], "seal before completion")
    _inventory_exact(result["inventory"], output, {str(output / "result.json")})
    require(capture_ref in seal["inventory"], "sealed capture terminal missing")
    _inventory_exact([r for r in seal["inventory"] if r != capture_ref], output.parent, {str(output.parent / "seal.json")})
    expected_results = [str(Path(j["output"]) / "result.json") for j in spec["jobs"]]
    same_set([r["path"] for r in seal["job_results"]], expected_results, "sealed jobs")
    statuses = []
    for sealed_result_ref in seal["job_results"]:
        sealed_result = load(sealed_result_ref, "job-result")
        bound(sealed_result, spec_ref, spec)
        require(
            sealed_result_ref["path"]
            == str(Path(next(j for j in spec["jobs"] if j["id"] == sealed_result["job_id"])["output"]) / "result.json"),
            "sealed result job binding",
        )
        statuses.append(sealed_result["disposition"])
    precedence = ("interrupted", "completed-failed", "completed-unverified", "refused", "unrun", "successful")
    require(seal["disposition"] == next(status for status in precedence if status in statuses), "seal disposition mismatch")
    source_check(start["source_witness"], spec["candidate"], candidate, spec["source_freeze"], freeze, grant)
    source_rows = [source_check(r, spec["candidate"], candidate, spec["source_freeze"], freeze, grant) for r in result["source_witnesses"]]
    require(source_rows or result["disposition"] in ("refused", "unrun"), "missing source witnesses")
    reasons = []
    by_ref = {}
    never_run = result["disposition"] in ("refused", "unrun")
    generic_complete = False
    if not never_run:
        try:
            launch, waited = parent_observations(result, spec_ref, spec, job, command_ref)
            if not (output / "completion.json").exists():
                raise IncompleteEvidence("missing process completion envelope")
            completion_ref = ref(output / "completion.json")
            completion = load(completion_ref, "job-evidence")
            for field in (
                "attempt_id",
                "job_id",
                "spec",
                "command",
                "collected_ids",
                "outcomes",
                "process_records",
                "obligation_records",
                "source_witnesses",
            ):
                require(completion[field] == result[field], "completion/result " + field)
            by_ref, _raws, runtime, _policy_record, _starts = processes(
                result, spec_ref, spec, job, command_ref, candidate, freeze, grant, dependencies
            )
            controllers = [row for row in by_ref.values() if row["role"] == "controller" and row["phase"] == "start"]
            require(len(controllers) == 1 and controllers[0]["identity"] == launch["identity"], "parent launch/controller identity")
            require(launch["identity"]["ppid"] == runtime["runner_identity"]["pid"], "launch/runtime parent")
            for process in by_ref.values():
                require(
                    all(process["identity"][field] == runtime["runner_identity"][field] for field in ("host", "boot_id"))
                    and waited["start_monotonic_ns"] <= process["monotonic_ns"] <= waited["end_monotonic_ns"],
                    "process monotonic lifetime outside parent wait interval",
                )
                require(
                    process["intra_op"]["provenance"] in completion["inventory"], "completion inventory missing process raw observation"
                )
            attempt_source = load(start["source_witness"], "source-witness")
            require(attempt_source["process_identity"] == runtime["runner_identity"], "attempt/runner source identity")
            runner_sources = [
                source_check(r, spec["candidate"], candidate, spec["source_freeze"], freeze, grant, runtime["runner_identity"])
                for r in seal["inventory"]
                if Path(r["path"]).parent == output.parent and re.fullmatch(r"source-[0-9]{4}\.json", Path(r["path"]).name)
            ]
            require(runtime["runner_source_start"] in seal["inventory"], "runner start source not sealed")
            require(
                runner_sources
                and min(row["at"] for row in runner_sources) <= result["start"]
                and max(row["at"] for row in runner_sources) >= result["end"]["value"],
                "runner pre/post source observations",
            )
            owner_completion(completion, result, by_ref, _raws, spec, job, candidate, dependencies)
            collections(result, job, by_ref)
            reasons += outcomes(result, job, by_ref)
            generic_complete = True
        except IncompleteEvidence as exc:
            reasons.append(str(exc))
    elif result["end"]["state"] == "unavailable":
        reasons.append("absolute finish and duration unavailable: " + result["end"]["reason"])
    observed = result["disposition"]
    if observed == "successful" and reasons:
        observed = "completed-unverified"
    if observed != "successful":
        reasons.append(result["reason"])
    obligations = []
    validation_receipts = []
    for obligation in job["obligations"]:
        contract = load(obligation["contract"], "case-contract")
        for left, right in (
            ("obligation_kind", "kind"),
            ("required_cases", "required_cases"),
            ("validator", "validator"),
            ("evidence_class", "evidence_class"),
            ("minimum_devices", "minimum_devices"),
        ):
            require(contract[left] == obligation[right], "case contract binding")
        if obligation["kind"] in COMMON:
            require(contract["roles"] == job["roles"], "case contract roles")
        else:
            require(all(role in job["roles"] for role in contract["roles"]), "case contract role subset")
        evidence = []
        for reference in result["obligation_records"]:
            item = load(reference, "obligation-evidence")
            bound(item, spec_ref, spec, job)
            require(item["obligation_id"] in {o["id"] for o in job["obligations"]}, "unknown obligation evidence")
            if item["obligation_id"] != obligation["id"]:
                continue
            for left, right in (
                ("obligation_kind", "kind"),
                ("contract", "contract"),
                ("validator", "validator"),
                ("evidence_class", "evidence_class"),
            ):
                require(item[left] == obligation[right], "obligation evidence binding")
            require(item["case_id"] in obligation["required_cases"], "obligation case")
            if generic_complete:
                require(
                    key(item["producer_process"]) in by_ref and by_ref[key(item["producer_process"])]["phase"] == "start",
                    "obligation producer missing",
                )
            evidence.append(reference)
        unique([load(r)["case_id"] for r in evidence], "obligation cases")
        disposition = {
            "id": obligation["id"],
            "disposition": "unrun" if not evidence else "unsatisfied",
            "reason": "required accepted case validator is unimplemented or lacks evidence",
            "contract": obligation["contract"],
            "validator": obligation["validator"],
            "evidence": evidence,
            "validator_receipt": [],
        }
        if accepted_validator and "p4-common-v1" in accepted_validator and generic_complete and obligation["kind"] in COMMON and evidence:
            validated = common_validation(
                obligation,
                contract,
                evidence,
                accepted_validator["p4-common-v1"],
                result,
                by_ref,
                spec_ref,
                job,
                validation_identity,
                validation_start,
                reasons,
            )
            validation_receipts.append((disposition, validated))
            disposition["disposition"] = validated["verdict"]
            disposition["reason"] = "" if validated["verdict"] == "satisfied" else "; ".join(validated["reasons"])
        if (
            accepted_validator
            and "p5-strict-v1" in accepted_validator
            and obligation["kind"] in STRICT
            and (evidence or observed != "successful")
            and (generic_complete or never_run or result["disposition"] != "successful")
        ):
            validated = strict_validation(
                obligation, contract, accepted_validator["p5-strict-v1"], spec_ref, job, validation_identity, validation_start
            )
            require(validated["verdict"] != "satisfied" or (generic_complete and observed == "successful"), "strict without common custody")
            validation_receipts.append((disposition, validated))
            disposition["disposition"] = validated["verdict"]
            disposition["reason"] = "; ".join(validated["reasons"])
        obligations.append(disposition)
    load(spec_ref, "spec")
    load(spec["candidate"], "candidate")
    load(spec["dependencies"], "environment")
    for reference in [*capture_files, *result["inventory"], seal_ref]:
        read_bytes(reference)
    assets = destination + ".components"
    parent_fds = _safe_parent(assets)
    try:
        os.mkdir(Path(assets).name, 0o700, dir_fd=parent_fds[-1])
    except OSError as exc:
        raise ValueError("component destination occupied or unsafe") from exc
    finally:
        for fd in reversed(parent_fds):
            os.close(fd)
    retained_receipts = []
    for disposition, validated in validation_receipts:
        retained = publish(str(Path(assets) / (disposition["id"] + ".json")), validated)
        retained_receipts.append(retained)
        if disposition["disposition"] == "satisfied":
            disposition["validator_receipt"] = [retained]
    component = components(spec_ref, job, observed, obligations, retained_receipts)
    component_ref = publish(str(Path(assets) / "classification.json"), component)
    reasons += component["reasons"]
    receipt = record_receipt(
        spec,
        job,
        spec_ref,
        tools["verifier"],
        result_ref,
        observed,
        obligations,
        reasons,
        [spec_ref, result_ref, command_ref, start_ref, seal_ref, capture_ref, *capture_files, component_ref, *retained_receipts],
    )
    protocol.validate_record("receipt", receipt)
    publish(destination, receipt)
    return receipt


def record_receipt(spec, job, spec_ref, verifier_ref, result_ref, observed, obligations, reasons, inventory):
    return {
        "schema": protocol.SCHEMA,
        "kind": "receipt",
        "attempt_id": spec["attempt_id"],
        "job_id": job["id"],
        "spec": spec_ref,
        "verifier": verifier_ref,
        "at": utc(),
        "result": result_ref,
        "custody": "satisfied" if all(o["disposition"] == "satisfied" for o in obligations) else "unsatisfied",
        "test_outcome": observed,
        "hardware": (
            "unsatisfied"
            if any(
                o["disposition"] == "unsatisfied"
                and next(j for j in job["obligations"] if j["id"] == o["id"])["evidence_class"] == "real-cuda"
                for o in obligations
            )
            else "unrun"
        ),
        "obligations": obligations,
        "reasons": [r for r in reasons if r],
        "inventory": inventory,
    }
