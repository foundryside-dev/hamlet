def verifier():

def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, sort_keys=True) + "\n")
    return p.file_reference(path)


def obs(value):
    return {"state": "measured", "value": value, "reason": ""}


def absent():
    return {"state": "unavailable", "value": None, "reason": "not observed"}


def record(kind, **fields):
    return {"schema": p.SCHEMA, "kind": kind, **fields}

    def seal(self):
        files = [x for x in self.output.iterdir() if x.is_file() and x.name not in ("result.json", "completion.json")]
        completion = record(
            "job-evidence",
            **{
                k: self.result[k]
                for k in (
                    "attempt_id",
                    "job_id",
                    "spec",
                    "command",
                    "collected_ids",
                    "outcomes",
                    "process_records",
                    "obligation_records",
                    "source_witnesses",
                )
            },
            inventory=[p.file_reference(x) for x in sorted(files)]
        )
        write(self.output / "completion.json", completion)
        self.result["inventory"] = [p.file_reference(x) for x in sorted(self.output.iterdir()) if x.is_file() and x.name != "result.json"]
        self.result_ref = write(self.output / "result.json", self.result)
        refs = [p.file_reference(x) for x in sorted(self.output.parent.rglob("*")) if x.is_file() and x.name != "seal.json"]
        write(
            self.output.parent / "seal.json",
            record(
                "attempt-seal",
                attempt_id="attempt",
                spec=self.spec_ref,
                start=self.start_ref,
                disposition=self.result["disposition"],
                job_results=[self.result_ref],
                inventory=[*refs, p.file_reference(self.capture / "result.json")],
                at=T4,
                source_freeze=self.freeze,
            ),
        )

def coherent_change(evidence, mutate):
    """Alter inert observations, then rehash their reference DAG without stale-reference failures."""
    for path in sorted(evidence.root.rglob("*.json")):
        value = json.loads(path.read_text())
        mutate(path, value)
        write(path, value)
    for _ in range(30):
        changed = False

        def refresh(value):
            if type(value) is dict:
                if set(value) == {"path", "bytes", "sha256"}:
                    return p.file_reference(Path(value["path"]))
                return {k: refresh(v) for k, v in value.items()}
            return [refresh(v) for v in value] if type(value) is list else value

        for path in sorted(evidence.root.rglob("*.json")):
            value = json.loads(path.read_text())
            updated = refresh(value)
            if updated != value:
                write(path, updated)
                changed = True
        if not changed:
            return
    raise AssertionError("synthetic reference DAG did not converge")

        content = (Path(serialized["path"]).parent / row["retained_content"]).read_bytes()
        assert len(content) == row["reference"]["bytes"] and hashlib.sha256(content).hexdigest() == row["reference"]["sha256"]
        documents[row["reference"]["path"]] = content
        collect(json.loads(content))
        references[row["reference"]["path"]] = row["reference"]
    module = verifier()
    # Supplied pure P5 specs omit mandatory P0 common obligations: this isolates graph traversal only.
    monkeypatch.setattr(p, "validate_record", lambda kind, value: None)
    original_ref = p.file_reference
    monkeypatch.setattr(p, "file_reference", lambda path: references[str(path)] if str(path) in references else original_ref(path))
    monkeypatch.setattr(module, "read_bytes", lambda reference: documents.get(reference["path"], b""))
    actual = module.strict_subject(subject["spec"], subject["job_id"], subject["obligation_id"])
    assert {module.key(row["reference"]) for row in actual["documents"]} == {module.key(row["reference"]) for row in subject["documents"]}
    from verify_bac_hardware_children_v1 import validate_child_obligation
    assert validate_child_obligation(actual)["verdict"] == "satisfied"

@pytest.mark.parametrize("mutation", ["policy", "resource", "argv", "bool-time", "order", "cumulative", "truncation", "diagnostic", "emission", "claim"])
def test_sampler_live_transport_resealed_correspondence_mutations(tmp_path, mutation):
    evidence, args = sampler_fixture(tmp_path)
    _process_ref, _process, raw, _launch_ref, launch, _spec, job, _candidate, _dependencies = args
    if mutation in ("policy", "resource"):
        policy = json.loads(Path(launch["sampler_policy"][0]["path"]).read_text())
        policy["phase_ns" if mutation == "policy" else "cpu_slots"] += 1
        launch["sampler_policy"] = [write(Path(launch["sampler_policy"][0]["path"]), policy)]
        job["prerequisites"] = launch["sampler_policy"]
    elif mutation == "emission":
        path = evidence.output / "sampler-emission-guard-start.json"
        value = json.loads(path.read_text()); value["process"] = raw["sampler"][0]; write(path, value)
    elif mutation == "claim":
        raw["owner_claim"] = [write(Path(raw["owner_claim"][0]["path"]), {"monotonic_ns": 199})]
    else:
        sampler = json.loads(Path(raw["sampler"][0]["path"]).read_text())
        if mutation == "argv": sampler["commands"][0]["argv"].append("foreign")
        if mutation == "bool-time": sampler["start_ns"] = True
        if mutation == "order": sampler["commands"][1]["start_ns"] = 0
        if mutation == "cumulative": sampler["cumulative_ns"] += 1
        if mutation == "truncation": sampler["commands"][0]["stderr_truncated"] = True
        if mutation == "diagnostic":
            path = evidence.output / "sampler-guard-start-0.json"
            diagnostic = json.loads(path.read_text()); diagnostic["pid"] += 1; write(path, diagnostic)
        raw["sampler"] = [write(Path(raw["sampler"][0]["path"]), sampler)]
    with pytest.raises(ValueError, match="sampler|fork claim"):
        verifier().owner_sampler(*args)


@pytest.mark.parametrize("mutation", ["executable-after", "version", "git-parent", "git-time", "git-errors", "source-order", "source-head", "wrapper-argv", "lease-bool"])
def test_native_capture_custody_resealed_mutations(tmp_path, mutation):
    evidence = Evidence(tmp_path, common=True)
    def mutate(path, value):
        if path.name == "discovery.json":
            command = value["commands"][0]
            if mutation == "executable-after": command["executable_after"] = evidence.inputs[1]
            if mutation == "version": command["stdout"] = "git version foreign\n"
            if mutation == "git-parent": command["identity"]["ppid"] += 1
            if mutation == "git-time": command["start"] = T4
            if mutation == "git-errors": command["errors"] = ["EOF not established"]
        if path.name == "discovery-command.json":
            raw = json.loads((evidence.capture / "discovery.json").read_text())
            value["stdout"] = json.dumps(raw)
            if mutation == "wrapper-argv": value["argv"].append("foreign")
            if mutation == "lease-bool": value["environment"]["BAC_CAPTURE_LEASE_FDS"] = "[true]"
        if path.name == "source-before-git.json" and mutation == "source-order": value.reverse()
        if path.name == "source-after.json" and mutation == "source-head": value["head"] = "c" * 40
    coherent_change(evidence, mutate)
    # Sorted DAG mutation visits the wrapper first, so reseal its parsed stdout explicitly afterward.
    coherent_change(evidence, lambda path, row: row.update(stdout=json.dumps(json.loads((evidence.capture / "discovery.json").read_text())))
                    if path.name == "discovery-command.json" else None)
    with pytest.raises(ValueError, match="native Git|observed Git|capture source|discovery"):
        evidence.verify()
