def read(p):return json.loads(p.read_text())

def ref(p):
    if p.is_symlink() or any(x.is_symlink() for x in p.parents):raise ValueError('Symlink refused')
    k=next(k for k in ('W','Q','R') if p.is_relative_to(ROOTS[k])); b=p.read_bytes()
    return dict(root=k,path=str(p.relative_to(ROOTS[k])),bytes=len(b),sha256=hashlib.sha256(b).hexdigest())

def save(p,j):
    with p.open('x') as f:json.dump(j,f,indent=2,sort_keys=True);f.write('\n')
    return ref(p)
