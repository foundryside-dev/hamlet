**File**: `src/townlet/universe/raw_configs_v21.py` (new file)

**Structure**:

```python
"""Raw config loader for v2.1 hierarchical structure."""

from dataclasses import dataclass
from pathlib import Path
from typing import Dict

from townlet.config.experiment_config import ExperimentConfig
from townlet.config.stratum_config import StratumConfig
from townlet.config.environment_config import EnvironmentConfig
from townlet.config.actions_config import ActionsConfig
from townlet.config.agent_config import AgentConfig
from townlet.config.curriculum_config import CurriculumConfig
from townlet.config.bars_config import BarsConfig
from townlet.config.affordances_config import AffordancesConfig
from townlet.config.training_config import TrainingConfig
from townlet.universe.errors import CompilationErrorCollector


@dataclass(frozen=True)
class CurriculumLevel:
    """All configs for a single curriculum level."""
    name: str  # e.g., "L1_full_observability"
    curriculum: CurriculumConfig
    bars: BarsConfig
    affordances: AffordancesConfig
    training: TrainingConfig

    @property
    def level_dir(self) -> str:
        """Directory name for this level."""
        return self.name


@dataclass(frozen=True)
class RawConfigsV21:
    """
    Container of all config DTOs for v2.1 hierarchical structure.

    This is the v2.1 equivalent of RawConfigs, but with hierarchical structure.
    """

    # Shared experiment-level configs (WHAT exists)
    experiment: ExperimentConfig
    stratum: StratumConfig
    environment: EnvironmentConfig  # Vocabulary only (meters, affordances, VFS vars)
    actions: ActionsConfig
    agent: AgentConfig

    # Curriculum levels (HOW things behave)
    levels: Dict[str, CurriculumLevel]  # {level_name: CurriculumLevel}

    # Provenance
    experiment_dir: Path

    def __post_init__(self):
        """Validate vocabulary consistency across all levels."""
        if not self.levels:
            raise ValueError(f"No curriculum levels found in {self.experiment_dir}")

        # Get canonical vocabulary from environment.yaml
        env_meter_names = {m.name for m in self.environment.meters}
        env_affordance_names = {a.name for a in self.environment.affordances}

        # Validate each level has same vocabulary
        for level_name, level in self.levels.items():
            level_meter_names = {m.name for m in level.bars.meters}
            level_affordance_names = {a.name for a in level.affordances.affordances}

            if level_meter_names != env_meter_names:
                missing = env_meter_names - level_meter_names
                extra = level_meter_names - env_meter_names
                raise ValueError(
                    f"Meter vocabulary mismatch in {level_name}/bars.yaml\n"
                    f"  Expected: {sorted(env_meter_names)}\n"
                    f"  Actual: {sorted(level_meter_names)}\n"
                    f"  Missing: {sorted(missing) if missing else 'none'}\n"
                    f"  Extra: {sorted(extra) if extra else 'none'}\n"
                    f"\nAll levels must have same meters as environment.yaml"
                )

            if level_affordance_names != env_affordance_names:
                missing = env_affordance_names - level_affordance_names
                extra = level_affordance_names - env_affordance_names
                raise ValueError(
                    f"Affordance vocabulary mismatch in {level_name}/affordances.yaml\n"
                    f"  Expected: {sorted(env_affordance_names)}\n"
                    f"  Actual: {sorted(level_affordance_names)}\n"
                    f"  Missing: {sorted(missing) if missing else 'none'}\n"
                    f"  Extra: {sorted(extra) if extra else 'none'}\n"
                    f"\nAll levels must have same affordances as environment.yaml"
                )

    @classmethod
    def from_experiment_dir(cls, experiment_dir: Path) -> "RawConfigsV21":
        """
        Load all configs from v2.1 hierarchical structure.

        Validates:
        - All required files exist
        - All YAMLs parse correctly
        - Vocabulary is consistent across levels

        Raises:
            CompilationError: If any validation fails
        """
        experiment_dir = Path(experiment_dir).resolve()
        errors = CompilationErrorCollector(stage="Stage 1: Load v2.1 Configs")

        # Load shared configs
        try:
            experiment = ExperimentConfig.from_yaml(experiment_dir / "experiment.yaml")
        except Exception as e:
            errors.add(f"Failed to load experiment.yaml: {e}", code="LOAD_ERROR")

        try:
            stratum = StratumConfig.from_yaml(experiment_dir / "stratum.yaml")
        except Exception as e:
            errors.add(f"Failed to load stratum.yaml: {e}", code="LOAD_ERROR")

        try:
            environment = EnvironmentConfig.from_yaml(experiment_dir / "environment.yaml")
        except Exception as e:
            errors.add(f"Failed to load environment.yaml: {e}", code="LOAD_ERROR")

        try:
            actions = ActionsConfig.from_yaml(experiment_dir / "actions.yaml")
        except Exception as e:
            errors.add(f"Failed to load actions.yaml: {e}", code="LOAD_ERROR")

        try:
            agent = AgentConfig.from_yaml(experiment_dir / "agent.yaml")
        except Exception as e:
            errors.add(f"Failed to load agent.yaml: {e}", code="LOAD_ERROR")

        # Stop if shared configs failed
        if errors.errors:
            errors.check_and_raise()

        # Load curriculum levels
        levels_dir = experiment_dir / "levels"
        if not levels_dir.exists():
            errors.add(
                f"Missing levels/ directory",
                code="MISSING_LEVELS_DIR",
                location=str(experiment_dir)
            )
            errors.check_and_raise()

        levels = {}
        for level_dir in sorted(levels_dir.iterdir()):
            if not level_dir.is_dir():
                continue

            level_name = level_dir.name

            try:
                curriculum = CurriculumConfig.from_yaml(level_dir / "curriculum.yaml")
                bars = BarsConfig.from_yaml(level_dir / "bars.yaml")
                affordances = AffordancesConfig.from_yaml(level_dir / "affordances.yaml")
                training = TrainingConfig.from_yaml(level_dir / "training.yaml")

                levels[level_name] = CurriculumLevel(
                    name=level_name,
                    curriculum=curriculum,
                    bars=bars,
                    affordances=affordances,
                    training=training
                )
            except Exception as e:
                errors.add(
                    f"Failed to load level {level_name}: {e}",
                    code="LEVEL_LOAD_ERROR",
                    location=level_name
                )

        errors.check_and_raise()

        # Create and validate (vocabulary check happens in __post_init__)
        return cls(
            experiment=experiment,
            stratum=stratum,
            environment=environment,
            actions=actions,
            agent=agent,
            levels=levels,
            experiment_dir=experiment_dir
        )
```

```python
def _phase_0_validate_yaml_syntax(self, experiment_dir: Path) -> None:
    """Phase 0 – validate all YAML files can be parsed before compilation begins."""
    errors = CompilationErrorCollector(stage="Phase 0: YAML Syntax Validation")

    # Shared config files (v2.1 structure)
    shared_files = [
        "experiment.yaml",
        "stratum.yaml",
        "environment.yaml",
        "actions.yaml",
        "agent.yaml",
    ]

    for file_name in shared_files:
        file_path = experiment_dir / file_name
        if not file_path.exists():
            errors.add(
                f"{file_name}: File not found",
                code="MISSING_FILE",
                location=str(file_path)
            )
            continue

        try:
            with file_path.open() as handle:
                yaml.safe_load(handle)
        except yaml.YAMLError as exc:
            error_msg = str(exc)
            if hasattr(exc, "problem_mark"):
                mark = exc.problem_mark
                problem = getattr(exc, "problem", None) or "syntax error"
                error_msg = f"line {mark.line + 1}, column {mark.column + 1}: {problem}"
            errors.add(error_msg, code="YAML_SYNTAX_ERROR", location=file_name)

    # Curriculum level files
    levels_dir = experiment_dir / "levels"
    if not levels_dir.exists():
        errors.add(
            "Missing levels/ directory",
            code="MISSING_LEVELS_DIR",
            location=str(experiment_dir)
        )
    else:
        for level_dir in levels_dir.iterdir():
            if not level_dir.is_dir():
                continue

            level_files = ["curriculum.yaml", "bars.yaml", "affordances.yaml", "training.yaml"]
            for file_name in level_files:
                file_path = level_dir / file_name
                if not file_path.exists():
                    errors.add(
                        f"{level_dir.name}/{file_name}: File not found",
                        code="MISSING_FILE",
                        location=str(file_path)
                    )
                    continue

                try:
                    with file_path.open() as handle:
                        yaml.safe_load(handle)
                except yaml.YAMLError as exc:
                    # ... same error handling ...
                    errors.add(error_msg, code="YAML_SYNTAX_ERROR", location=f"{level_dir.name}/{file_name}")

    if errors.errors:
        errors.add_hint("Check YAML indentation (use spaces, not tabs)")
        errors.add_hint("Ensure lists use proper '- item' syntax")
        errors.check_and_raise()
```

```python
@dataclass
class CompiledUniverse:
    """
    Compiled universe artifact with all validated configs and derived metadata.

    Supports both single-level and multi-level compilation:
    - Single-level: legacy mode (one curriculum level compiled at a time)
    - Multi-level: v2.1 mode (all levels compiled together, can switch at runtime)
    """

    # Core metadata (existing fields - keep these)
    metadata: UniverseMetadata
    observation_spec: ObservationSpec
    action_metadata: ActionSpaceMetadata
    meter_metadata: MeterMetadata
    affordance_metadata: AffordanceMetadata
    optimization_data: OptimizationData

    # NEW: Multi-level support (v2.1)
    experiment_dir: Path | None = None  # Only for v2.1
    all_levels: Dict[str, "LevelMetadata"] | None = None  # {level_name: metadata}

    # ... existing methods ...

    @dataclass
    class LevelMetadata:
        """Metadata for one curriculum level."""
        name: str
        curriculum: CurriculumConfig
        observation_spec: ObservationSpec
        # Add other per-level metadata as needed

    def get_level(self, level_name: str) -> "LevelMetadata":
        """Get metadata for a specific curriculum level."""
        if self.all_levels is None:
            raise ValueError("This CompiledUniverse was not compiled with multi-level support")
        if level_name not in self.all_levels:
            raise ValueError(f"Level {level_name} not found. Available: {list(self.all_levels.keys())}")
        return self.all_levels[level_name]

    @property
    def available_levels(self) -> list[str]:
        """Get list of available curriculum levels."""
        if self.all_levels is None:
            return []
        return list(self.all_levels.keys())
```

```python
def compile(self, experiment_dir: Path, primary_level: str | None = None, use_cache: bool = True) -> CompiledUniverse:
    experiment_dir = Path(experiment_dir).resolve()

    # Phase 0: Validate YAML syntax
    self._phase_0_validate_yaml_syntax(experiment_dir)

    # Stage 1: Load all configs
    logger.info("=== Stage 1: Loading v2.1 configs ===")
    raw_configs = self._stage_1_load_v21_configs(experiment_dir)

    logger.info(f"Loaded experiment: {raw_configs.experiment.name}")
    logger.info(f"Found {len(raw_configs.levels)} curriculum levels: {list(raw_configs.levels.keys())}")

    # Determine primary level
    if primary_level is None:
        primary_level = sorted(raw_configs.levels.keys())[0]
        logger.info(f"No primary level specified, using: {primary_level}")

    if primary_level not in raw_configs.levels:
        raise ValueError(
            f"Primary level '{primary_level}' not found. "
            f"Available: {list(raw_configs.levels.keys())}"
        )

    # Compile ALL levels (Stages 2-6 for each level)
    all_level_metadata = {}

    for level_name in sorted(raw_configs.levels.keys()):
        level = raw_configs.levels[level_name]
        logger.info(f"\n=== Compiling {level_name} ===")

        # Stage 2: Build symbol tables
        logger.info(f"  Stage 2: Building symbol tables for {level_name}")
        symbol_table = self._stage_2_build_symbol_tables(raw_configs, level)

        # Stage 3: Resolve references
        logger.info(f"  Stage 3: Resolving references for {level_name}")
        resolved = self._stage_3_resolve_references(symbol_table, raw_configs, level)

        # Stage 4: Cross-validate
        logger.info(f"  Stage 4: Cross-validating {level_name}")
        validated = self._stage_4_cross_validate(resolved, raw_configs, level)

        # Stage 5: Build metadata
        logger.info(f"  Stage 5: Building metadata for {level_name}")
        metadata = self._stage_5_build_rich_metadata(validated, raw_configs, level)

        # Stage 6: Optimize
        logger.info(f"  Stage 6: Optimizing {level_name}")
        optimized = self._stage_6_optimize(metadata, raw_configs, level)

        # Store metadata for this level
        all_level_metadata[level_name] = {
            'observation_spec': metadata['observation_spec'],
            'optimized': optimized,
            # ... other per-level data ...
        }

    # Stage 7: Emit CompiledUniverse (with multi-level support)
    logger.info(f"\n=== Stage 7: Emitting CompiledUniverse ===")
    primary_metadata = all_level_metadata[primary_level]

    compiled = self._stage_7_emit_compiled_universe(
        optimization_data=primary_metadata['optimized'],
        raw_configs=raw_configs,
        all_level_metadata=all_level_metadata
    )

    logger.info(f"✓ Compiled {len(all_level_metadata)} curriculum levels")
    logger.info(f"  Primary level: {primary_level}")
    logger.info(f"  All levels: {list(all_level_metadata.keys())}")

    # Cache handling
    if use_cache:
        cache_path = self._cache_artifact_path(experiment_dir, primary_level)
        compiled.save_to_cache(cache_path)

    return compiled
```
